   314→  /** اعمال مرتب‌سازی چندمرحله‌ای (عددی/متنی) */
   315→  ListsView.applySort = function (rows, pipeline) {
   316→    const sorted = [...rows];
   317→    for (let i = pipeline.length - 1; i >= 0; i--) {
   318→      const { field, direction } = pipeline[i];
   319→      const spec = RM.config.LIST_SORTABLE_FIELDS[field] || { numeric: true };
   320→      sorted.sort((a, b) => {
   321→        let r;
   322→        if (spec.numeric) {
   323→          r = U.compareNumeric(a[field], b[field], 'asc');
   324→        } else {
   325→          const ka = String(a[field] ?? '');
   326→          const kb = String(b[field] ?? '');
   327→          r = ka.localeCompare(kb, 'fa', { numeric: true });
   328→          if (r !== 0 && (ka === '' || kb === '')) r = ka === '' ? 1 : -1;
   329→        }
   330→        return direction === 'desc' ? -r : r;
   331→      });
   332→    }
   333→    return sorted;
   334→  };

   336→  /** رندر جدول لیست یک منبع */
   337→  ListsView.renderTable = async function (src) {
   338→    const st = ListsView.state[src];
   339→    const target = RM.config.IMPORT_TARGETS[src];
   340→    const wrap = document.getElementById(`list-table-${src}`);
   341→    const statsEl = document.getElementById(`list-stats-${src}`);
   342→    if (!wrap) return;
   343→
   344→    /* --- داده: رکوردهای فایل + اعمال ویرایش‌ها (جایگزینی اطلاعات فایل) --- */
   345→    const records = await RM.edits.applyToRecords(src, await db[target.table].toArray());
   346→    const editedCount = RM.edits._lastAppliedCount || 0;
   347→
   348→    const rows = ListsView.applySort(records, st.sortPipeline);
   349→
   350→    /* --- چیپ‌های آمار --- */
   351→    statsEl.innerHTML = `
   352→      <span class="chip">${U.faNum(rows.length)} رکورد</span>
   353→      ${editedCount ? `<span class="chip chip-warn" title="رکوردهایی که اطلاعات ویرایش‌شده دارند">✎ ویرایش‌شده: ${U.faNum(editedCount)}</span>` : ''}`;
   354→
   355→    /* --- پنل مرتب‌سازی --- */
   356→    UI.sortPanel(document.getElementById(`list-sort-pipeline-${src}`), st.sortPipeline,
   357→      RM.config.LIST_SORTABLE_FIELDS, {
   358→        onDir: (i) => {
   359→          st.sortPipeline[i].direction = st.sortPipeline[i].direction === 'asc' ? 'desc' : 'asc';
   360→          DB.setSetting(`list${cap(src)}Sort`, st.sortPipeline);
   361→          ListsView.renderTable(src);
   362→        },
   363→        onRemove: (i) => {
   364→          st.sortPipeline.splice(i, 1);
   365→          DB.setSetting(`list${cap(src)}Sort`, st.sortPipeline);
   366→          ListsView.renderTable(src);
   367→        },
   368→        onReorder: (from, to) => {
   369→          const [moved] = st.sortPipeline.splice(from, 1);
   370→          st.sortPipeline.splice(to, 0, moved);
   371→          DB.setSetting(`list${cap(src)}Sort`, st.sortPipeline);
   372→          ListsView.renderTable(src);
   373→        },
   374→        emptyText: 'مرتب‌سازی پیش‌فرض (شماره ستاپ، شماره رول)',
   375→      });
   376→
   377→    /* --- پنل ترتیب ستون‌ها + نمایش/مخفی + عرض --- */
   378→    const all = await ListsView.columnRegistry(src);
   379→    const hiddenSet = new Set(st.hiddenCols || []);
   380→    const items = (st.columnOrder || RM.config.LIST_DEFAULT_COLUMN_ORDER[src])
   381→      .map((key) => all[key])
   382→      .filter(Boolean)
   383→      .map((c) => ({ key: c.key, label: c.label, width: (st.columnWidths || {})[c.key] ?? 0, hidden: hiddenSet.has(c.key) }));
   384→
   385→    UI.orderPanel(document.getElementById(`list-cols-pipeline-${src}`), items, {
   386→      onMove: async (from, to) => {
   387→        const order = st.columnOrder || [...RM.config.LIST_DEFAULT_COLUMN_ORDER[src]];
   388→        const [moved] = order.splice(from, 1);
   389→        order.splice(to, 0, moved);
   390→        st.columnOrder = order;
   391→        await DB.setSetting(`list${cap(src)}ColumnOrder`, order);
   392→        ListsView.renderTable(src);
   393→      },
   394→      onWidth: async (key, w) => {
   395→        st.columnWidths = st.columnWidths || {};
   396→        st.columnWidths[key] = w;
   397→        await DB.setSetting(`list${cap(src)}ColumnWidths`, st.columnWidths);
   398→        ListsView.renderTable(src);
   399→      },
   400→      /* --- تغییر ۳: نمایش/مخفی‌کردن هر ستون --- */
   401→      onToggle: async (key, hidden) => {
   402→        const set = new Set(st.hiddenCols || []);
   403→        if (hidden) set.add(key);
   404→        else set.delete(key);
   405→        st.hiddenCols = [...set];
   406→        await DB.setSetting(`list${cap(src)}HiddenCols`, st.hiddenCols);

   554→  /** بازکردن مودال ویرایش یک رکورد */
   555→  ListsView.openEditModal = async function (src, recordId) {
   556→    const target = RM.config.IMPORT_TARGETS[src];
   557→    const record = await db[target.table].get(recordId);
   558→    if (!record) {
   559→      UI.toast('رکورد یافت نشد.', 'warning');
   560→      return;
   561→    }
   562→
   563→    /* مقدار مؤثر (فایل + ویرایش‌های ذخیره‌شده) برای پیش‌فرض فرم */
   564→    const effective = { ...record };
   565→    await RM.edits.applyToRecords(src, [effective]);
   566→    const editKey = effective._editKey ?? record.rollNumber;   // کلید ویرایش = شمارهٔ اصلی
   567→
   568→    const labels = await N.effectiveColumnLabels(src);
   569→    const fields = ListsView.editableFields(src);
   570→    const numeric = new Set(RM.config.LIST_NUMERIC_EDIT_FIELDS);
   571→    const int = new Set(RM.config.LIST_INT_EDIT_FIELDS);
   572→
   573→    const inputId = (f) => `lef-${f}`;
   574→
   575→    const fieldHtml = fields.map((f) => {
   576→      const isNum = numeric.has(f);
   577→      const label = U.escapeHtml(labels[f] || RM.config.FIELD_LABELS[f] || f);
   578→      const isEdited = !!(effective.__editedFields && effective.__editedFields.includes(f));
   579→
   580→      let value = effective[f];
   581→      if (value === null || value === undefined) value = '';
   582→      else if (value instanceof Date) value = String(value);
   583→      else value = String(value);
   584→
   585→      return `
   586→        <div class="field">
   587→          <label for="${inputId(f)}">${label}
   588→            ${isEdited ? '<span class="field-edit-flag" title="این فیلد ویرایش شده و مقدار فعلی جایگزین فایل است">ویرایش‌شده</span>' : ''}
   589→            ${isNum ? '<span class="unit">(عددی)</span>' : ''}
   590→          </label>
   591→          <input type="text" id="${inputId(f)}" ${isNum ? 'inputmode="decimal"' : ''} dir="${f === 'rollNumber' || f === 'grade' || f === 'palletNumber' ? 'ltr' : 'auto'}"
   592→                 autocomplete="off" value="${U.escapeHtml(value)}" data-lef="${f}"
   593→                 aria-label="مقدار ${label}">
   594→          <p class="field-error" id="lef-err-${f}" hidden></p>
   595→        </div>`;
   596→    }).join('');
   597→
   598→    document.getElementById('modal-edit-title').textContent =
   599→      `ویرایش رکورد — ${RM.config.DATASET_LABELS[src]} · ${U.escapeHtml(record.rollNumber || 'بدون شماره')}`;
   600→
   601→    document.getElementById('modal-edit-body').innerHTML = `
   602→      <div class="modal-edit-fields">${fieldHtml}</div>
   603→      <p class="modal-edit-note">
   604→        • مقادیر خالی = «بدون مقدار» (null) ذخیره می‌شود.<br>
   605→        • ویرایش ذخیره می‌شود و از این پس <b>جایگزین اطلاعات فایل ایمپورت‌شده</b> می‌شود (پس از هر Import هم دوباره اعمال می‌گردد).<br>
   606→        • هر ویرایش جدید روی ویرایش قبلی همان رول ذخیره می‌شود؛ اگر مقداری را به مقدار اصلی فایل برگردانید، ویرایش همان فیلد حذف می‌شود.
   607→      </p>`;
   608→
   609→    document.getElementById('modal-edit').hidden = false;
   610→
   611→    ListsView.editing = {
   612→      sourceKey: src,
   613→      recordId,
   614→      imported: record,        // رکورد اصلی فایل (مبنای تشخیص بازگشت به اصلی)
   615→      effective,
   616→      editKey,                 // شماره رول اصلی — کلید رکورد ویرایش
   617→    };
   618→
   619→    const first = document.querySelector('#modal-edit-body input');
   620→    if (first) first.focus();
   621→  };

   629→  /** ذخیرهٔ ویرایش (دکمهٔ «ذخیرهٔ تغییرات») */
   630→  ListsView.saveEditModal = async function () {
   631→    const ctx = ListsView.editing;
   632→    if (!ctx) return;
   633→
   634→    const numeric = new Set(RM.config.LIST_NUMERIC_EDIT_FIELDS);
   635→    const int = new Set(RM.config.LIST_INT_EDIT_FIELDS);
   636→    const fields = ListsView.editableFields(ctx.sourceKey);
   637→
   638→    const changes = {};
   639→    let hasInvalid = false;
   640→
   641→    for (const f of fields) {
   642→      const input = document.querySelector(`[data-lef="${f}"]`);
   643→      if (!input) continue;
   644→      const raw = input.value.trim();
   645→
   646→      if (raw === '') {
   647→        changes[f] = (f === 'rollNumber') ? '' : null;
   648→        continue;
   649→      }
   650→
   651→      if (numeric.has(f)) {
   652→        const v = U.parseNumber(raw);
   653→        if (v === null) {
   654→          UI.setFieldError(input, document.getElementById(`lef-err-${f}`), 'عدد معتبر وارد کنید.');
   655→          hasInvalid = true;
   656→          continue;
   657→        }
   658→        changes[f] = int.has(f) ? Math.trunc(v) : v;
   659→      } else {
   660→        changes[f] = raw;
   661→      }
   662→    }
   663→
   664→    if (hasInvalid) {
   665→      UI.toast('برخی فیلدها عدد معتبر نیستند — خطاها را برطرف کنید.', 'error');
   666→      return;
   667→    }
   668→
   669→    /* شماره رول خالی؟ (کلید ویرایش) */
   670→    if ('rollNumber' in changes && (changes.rollNumber === '' || changes.rollNumber === null)) {
   671→      const input = document.querySelector('[data-lef="rollNumber"]');
   672→      UI.setFieldError(input, document.getElementById('lef-err-rollNumber'), 'شماره رول نمی‌تواند خالی باشد.');
   673→      UI.toast('شماره رول نمی‌تواند خالی باشد.', 'error');
   674→      return;
   675→    }
   676→
   677→    const result = await RM.edits.saveEdit(ctx.sourceKey, ctx.editKey, changes, ctx.imported);
   678→
   679→    ListsView.closeEditModal();
   680→
   681→    const changed = result.applied.length + result.removed.length;
   682→    if (!changed && Object.keys(result.fields).length === 0) {
   683→      UI.toast('تغییری ثبت نشد — همهٔ مقادیر با اطلاعات فایل یکسان است.', 'info');
   684→    } else {
   685→      UI.toast(
   686→        `ویرایش رول «${U.escapeHtml(String(ctx.editKey))}» ذخیره شد — ${U.faNum(Object.keys(result.fields).length)} فیلد ویرایش‌شده. از این پس جایگزین اطلاعات فایل است.`,
   687→        'success',
   688→        6000
   689→      );
   690→    }
   691→
   692→    await RM.refreshAll();
   693→  };
