    40→  UI.toast = function (message, type = 'info', duration = 4200) {
    41→    const stack = document.getElementById('toasts');
    42→    if (!stack) return;
    43→
    44→    const el = document.createElement('div');
    45→    el.className = `toast toast-${type}`;
    46→    el.setAttribute('role', 'status');
    47→    el.innerHTML = `${UI.icons[type] || UI.icons.info}<div>${message}</div>`;
    48→    stack.appendChild(el);
    49→
    50→    const timer = setTimeout(dismiss, duration);
    51→    function dismiss() {
    52→      clearTimeout(timer);
    53→      el.classList.add('leaving');
    54→      el.addEventListener('animationend', () => el.remove(), { once: true });
    55→    }
    56→    el.addEventListener('click', dismiss);
    57→  };

    67→  UI.confirm = function (options) {
    68→    return new Promise((resolve) => {
    69→      const overlay = document.getElementById('modal');
    70→      const titleEl = document.getElementById('modal-title');
    71→      const msgEl = document.getElementById('modal-msg');
    72→      const btnOk = document.getElementById('modal-confirm');
    73→      const btnCancel = document.getElementById('modal-cancel');
    74→
    75→      titleEl.textContent = options.title || 'تأیید عملیات';
    76→      // html اختیاری برای آمار تأیید (محتوای داخلی قابل اعتماد — UI خودمان می‌سازدش)
    77→      msgEl.innerHTML = (options.html || '') + U.escapeHtml(options.message || '');
    78→      btnOk.textContent = options.confirmText || 'بله، انجام بده';
    79→      btnOk.classList.toggle('btn-danger', !!options.danger);
    80→      btnOk.classList.toggle('btn-primary', !options.danger);
    81→
    82→      overlay.hidden = false;
    83→      btnOk.focus();
    84→
    85→      function close(result) {
    86→        overlay.hidden = true;
    87→        btnOk.removeEventListener('click', onOk);
    88→        btnCancel.removeEventListener('click', onCancel);
    89→        overlay.removeEventListener('click', onOverlay);
    90→        document.removeEventListener('keydown', onKey);
    91→        modalResolve = null;
    92→        resolve(result);
    93→      }
    94→      const onOk = () => close(true);
    95→      const onCancel = () => close(false);
    96→      const onOverlay = (e) => { if (e.target === overlay) close(false); };
    97→      const onKey = (e) => { if (e.key === 'Escape') close(false); };
    98→
    99→      btnOk.addEventListener('click', onOk);
   100→      btnCancel.addEventListener('click', onCancel);
   101→      overlay.addEventListener('click', onOverlay);
   102→      document.addEventListener('keydown', onKey);
   103→    });
   104→  };

   159→  /* ---------------- جدول صفحه‌بندی‌شده (§56) ---------------- */
   160→
   161→  /**
   162→   * رندر جدول صفحه‌بندی‌شده:
   163→   * @param wrap       عنصر نگهدارندهٔ جدول
   164→   * @param columns    [{key, label, render?, className?, sortable?}]
   165→   * @param rows       داده‌ها
   166→   * @param opts       { page, pageSize, emptyText, footerInfo }
   167→   */
   168→  UI.renderTable = function (wrap, columns, rows, opts = {}) {
   169→    const pageSize = opts.pageSize || RM.config.PAGE_SIZE;
   170→    const page = opts.page || 1;
   171→    const emptyText = opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.';
   172→
   173→    if (!rows.length) {
   174→      wrap.innerHTML = `
   175→        <div class="empty-state">
   176→          ${UI.icons.alert}
   177→          <p>${emptyText}</p>
   178→        </div>`;
   179→      return { page: 1, pages: 0 };
   180→    }
   181→
   182→    const pages = Math.max(1, Math.ceil(rows.length / pageSize));
   183→    const safePage = Math.min(page, pages);
   184→    const slice = rows.slice((safePage - 1) * pageSize, safePage * pageSize);
   185→
   186→    const thead = '<thead><tr>' +
   187→      columns.map((c) => `<th>${c.label}</th>`).join('') +
   188→      '</tr></thead>';
   189→
   190→    const tbody = '<tbody>' + slice.map((row) => {
   191→      const cells = columns.map((c) => {
   192→        const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   193→        return `<td class="${c.className || ''}">${content}</td>`;
   194→      }).join('');
   195→      return `<tr>${cells}</tr>`;
   196→    }).join('') + '</tbody>';
   197→
   198→    // نوار صفحه‌بندی (اگر بیش از یک صفحه)
   199→    const pager = pages > 1
   200→      ? `<div class="pager" role="navigation" aria-label="صفحه‌بندی جدول">
   201→           <button type="button" class="btn btn-ghost btn-sm" data-page="1" ${safePage === 1 ? 'disabled' : ''}>ابتدا</button>
   202→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage - 1}" ${safePage === 1 ? 'disabled' : ''}>قبلی</button>
   203→           <span class="pager-info">صفحهٔ ${U.faNum(safePage)} از ${U.faNum(pages)} — ${U.faNum(rows.length)} ردیف</span>
   204→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage + 1}" ${safePage === pages ? 'disabled' : ''}>بعدی</button>
   205→           <button type="button" class="btn btn-ghost btn-sm" data-page="${pages}" ${safePage === pages ? 'disabled' : ''}>انتها</button>
   206→         </div>`
   207→      : (opts.footerInfo ? `<div class="pager"><span class="pager-info">${opts.footerInfo(rows.length)}</span></div>` : '');
   208→
   209→    wrap.innerHTML = `
   210→      <div class="table-wrap table-scroll">
   211→        <table class="data-table">${thead}${tbody}</table>
   212→      </div>
   213→      ${pager}`;
   214→
   215→    // رویداد صفحه‌بندی
   216→    wrap.querySelectorAll('[data-page]').forEach((btn) => {
   217→      btn.addEventListener('click', () => {
   218→        const nextPage = Number(btn.dataset.page);
   219→        if (opts.onPageChange) opts.onPageChange(nextPage);
   220→      });
   221→    });
   222→
   223→    return { page: safePage, pages };
   224→  };

   327→  /**
   328→   * رندر «پنل مرتب‌سازی» — چیپ‌های مراحل مرتب‌سازی با کشیدن‌ورها،
   329→   * تغییر جهت و حذف. مولفهٔ مشترک رول‌های خام و گزارش ستاپ‌ها.
   330→   * @param {HTMLElement} wrap       عنصر نگهدارندهٔ چیپ‌ها
   331→   * @param {Array}        pipeline  [{field, direction}]
   332→   * @param {object}       fields    رجیستری فیلدها ({label})
   333→   * @param {object}       handlers  { onDir(i), onRemove(i), onReorder(from, to), emptyText? }
   334→   */
   335→  UI.sortPanel = function (wrap, pipeline, fields, handlers) {
   336→    wrap.innerHTML = pipeline.map((step, i) => {
   337→      const meta = fields[step.field] || { label: step.field };
   338→      return `
   339→        <div class="sort-chip" draggable="true" data-index="${i}" title="برای تغییر اولویت بکشید">
   340→          <span class="grip">${UI.icons.grip}</span>
   341→          <span class="sort-priority">${U.faNum(i + 1)}</span>
   342→          <span class="sort-label">${U.escapeHtml(meta.label)}</span>
   343→          <button type="button" class="sort-dir ${step.direction}" data-dir="${i}"
   344→                  aria-label="تغییر جهت مرتب‌سازی" title="تغییر جهت">
   345→            ${step.direction === 'asc' ? '↑' : '↓'}
   346→          </button>
   347→          <button type="button" class="sort-remove" data-remove="${i}"
   348→                  aria-label="حذف از مرتب‌سازی" title="حذف">${UI.icons.cross}</button>
   349→        </div>`;
   350→    }).join('') ||
   351→      `<span class="muted" style="font-size:.78rem">${U.escapeHtml(handlers.emptyText || 'مرتب‌سازی پیش‌فرض')}</span>`;
   352→
   353→    // تغییر جهت
   354→    wrap.querySelectorAll('[data-dir]').forEach((btn) => {
   355→      btn.addEventListener('click', () => handlers.onDir(Number(btn.dataset.dir)));
   356→    });
   357→
   358→    // حذف مرحله
   359→    wrap.querySelectorAll('[data-remove]').forEach((btn) => {
   360→      btn.addEventListener('click', () => handlers.onRemove(Number(btn.dataset.remove)));
   361→    });
   362→
   363→    // ---------- کشیدن و رها کردن برای تغییر ترتیب (§37) ----------
   364→    let dragSrcIndex = null;
   365→
   366→    wrap.querySelectorAll('.sort-chip').forEach((chip) => {
   367→      chip.addEventListener('dragstart', (e) => {
   368→        dragSrcIndex = Number(chip.dataset.index);
   369→        chip.classList.add('dragging');
   370→        e.dataTransfer.effectAllowed = 'move';
   371→        e.dataTransfer.setData('text/plain', String(dragSrcIndex));
   372→      });
   373→
   374→      chip.addEventListener('dragend', () => {
   375→        chip.classList.remove('dragging');
   376→        wrap.querySelectorAll('.sort-chip').forEach((c) => c.classList.remove('drag-over'));
   377→        dragSrcIndex = null;
   378→      });
   379→
   380→      chip.addEventListener('dragover', (e) => {
   381→        e.preventDefault();
   382→        e.dataTransfer.dropEffect = 'move';
   383→        if (!chip.classList.contains('drag-over')) chip.classList.add('drag-over');
   384→      });
   385→
   386→      chip.addEventListener('dragleave', () => chip.classList.remove('drag-over'));
   387→
   388→      chip.addEventListener('drop', (e) => {
   389→        e.preventDefault();
   390→        const from = dragSrcIndex;
   391→        const to = Number(chip.dataset.index);
   392→        if (from === null || from === to) return;
   393→        handlers.onReorder(from, to);
   394→      });
   395→    });
   396→  };
   397→
   398→  /* ---------------- جدول بلند با فیلتر زیر سرستون‌ها (مشترک ۳ جدول) ---------------- */
   399→
   400→  /**
   401→   * نرمال‌سازی متن جست‌وجو — ارقام فارسی/عربی → لاتین + حذف جداکننده‌های
   402→   * هزارگان و فاصله‌ها. هم روی «مقدار جست‌وجو» و هم روی «متن سلول» اعمال
   403→   * می‌شود تا «۸» با 8 و «۲۳» با 23 هم‌ارزش شود.
   404→   */
   405→  UI.filterNormalize = function (text) {
   406→    return U.toLatinDigits(String(text ?? ''))
   407→      .replace(/[,\u066C\u2019\u066B\s]/g, '')
   408→      .trim()
   409→      .toLowerCase();
   410→  };
   411→
   412→  /**
   413→   * منطق تطبیق «شامل بودن» (نسخهٔ ۲٫۳):
   414→   *   - عدد و متن هر دو با «شامل بودن» فیلتر می‌شوند (نه تطابق دقیق)
   415→   *   - invert = true → برعکس: «شامل نشود»
   416→   * @param {string[]} tokens   متن‌های قابل‌جست‌وجوی سلول (نمایش + مقدار خام)
   417→   * @param {string}   value    مقدار واردشده در باکس فیلتر
   418→   * @param {boolean}  invert   منطق برعکس فعال است؟
   419→   */
   420→  UI.filterContains = function (tokens, value, invert) {
   421→    const needle = UI.filterNormalize(value);
   422→    if (needle === '') return true;      // فیلتر خالی → همه
   423→    const hay = (tokens || [])
   424→      .map((t) => UI.filterNormalize(t))
   425→      .filter(Boolean)
   426→      .join(' ');
   427→    const contains = hay.includes(needle);
   428→    return invert ? !contains : contains;
   429→  };
   430→
   431→  /** خواندن/ساخت وضعیت فیلتر یک ستون ({ value, invert }) از state */
   432→  UI.filterStateOf = function (state, key) {
   433→    if (!state.filters[key] || typeof state.filters[key] !== 'object') {
   434→      state.filters[key] = { value: '', invert: false };
   435→    }
   436→    return state.filters[key];
   437→  };
   438→
   439→  /** آیکون «≠» دکمهٔ برعکس‌کردن منطق فیلتر */
   440→  UI.icons.notContains = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="12" cy="12" r="9"/><line x1="5.8" y1="18.2" x2="18.2" y2="5.8"/></svg>';
   441→
   442→  /**
   443→   * جدول بلند (حداقل ۵۰ رکورد) + فیلتر پیشرفتهٔ زیر سرستون‌ها — مولفهٔ
   444→   * مشترک جدول‌های «رول‌های برش‌خورده»، «رول‌های خام» و «گزارش ستاپ‌ها».
   445→   *
   446→   * منطق نسخهٔ ۲٫۳:
   447→   *   - فیلتر همهٔ ستون‌ها (عدد و متن) با «شامل بودن» + ارقام فارسی/لاتین
   448→   *   - دکمهٔ کوچک «≠» کنار هر باکس → سوییچ «شامل نشود»
   449→   *   - فوکوس باکس فیلتر حفظ می‌شود (فقط tbody دوباره رندر می‌شود،
   450→   *     سرستون/باکس‌ها بازسازی نمی‌شوند و فوکوس جابه‌جا نمی‌شود)
   451→   *   - سرستون + ردیف فیلتر چسبان هنگام اسکرول عمودی
   452→   *
   453→   * منطق نسخهٔ ۲٫۴:
   454→   *   - باکس «ردیف‌های قابل نمایش» در نوار پایین → ارتفاع جدول داینامیک
   455→   *     (فقط استایل ارتفاع به‌روز می‌شود؛ جدول/فیلترها/فوکوس دست‌نخورده)
   456→   *   - پشتیبانی عرض ستون (c.width → colgroup با table-layout:fixed)
   457→   *
   458→   * منطق نسخهٔ ۲٫۵:
   459→   *   - Wrap Text هدرها: عنوان ستون‌ها با توجه به عرض ستون چندخط می‌شود؛
   460→   *     ارتفاع ردیف سرستون اندازه‌گیری و در متغیر --ft-head-h ثبت می‌شود تا
   461→   *     ردیف فیلتر چسبان دقیقاً زیر آن بنشیند (ResizeObserver → هم‌گام با تغییر عرض)
   462→   *   - چیدمان باکس فیلتر: آیکون «≠» کوچک در سمت راست + باکس فیلتر در ادامه
   463→   *     که عرضش با عرض ستون تنظیم می‌شود (min-width: 0 → هرگز ناقص نمی‌شود)
   464→   *
   465→   * منطق نسخهٔ ۲٫۶:
   466→   *   - باکس‌های جمع (opts.sums): زیر ستون‌های عددی/زمانی، ردیف پایانی چسبان
   467→   *     (tfoot) با جمع همان ستون — جمع همیشه با دیتای «فیلترشدهٔ» فعلی
   468→   *     محاسبه می‌شود و با هر فیلتر/تغییر، به‌روز می‌شود.
   469→   *
   470→   * منطق جدید (لیست‌های بزرگ — تغییر ۳):
   471→   *   - opts.maxDomRows: فیلترها روی «همهٔ» رکوردها اعمال می‌شوند اما فقط
   472→   *     حداکثر N ردیف اولِ فیلترشده در DOM رندر می‌شوند (پاسخ‌گویی با دیتای
   473→   *     چند هزار ردیفی) — وضعیت/جمع‌ها همیشه از کل دیتای فیلترشده محاسبه می‌شوند.
   474→   *
   475→   * @param {HTMLElement} wrap     نگهدارندهٔ جدول
   476→   * @param {Array}  columns       [{key, label, render(row)→html, tokens(row)→string[],
   477→   *                                className?, filterable? (پیش‌فرض true), width? (px — نسخهٔ ۲٫۴)}]
   478→   * @param {Array}  rows          همهٔ رکوردها (مرتب‌شده توسط فراخوان)
   479→   * @param {object} opts          { state: {filters:{}, visibleRows?} (ماندگار بین رندرها),
   480→   *                                visibleRows, rowHeight, tableClass, emptyText,
   481→   *                                filteredEmptyText, footerNote, rowsControl? (bool — باکس تعداد ردیف),
   482→   *                                onRowsChange?(n) (ذخیرهٔ ماندگار),
   483→   *                                fixedCols? (bool — table-layout:fixed برای عرض‌های ستون),
   484→   *                                sums? ({ colKey: {type:'number'|'time'|'pair',
   485→   *                                        weightOf?(row,key)→number } }) — نسخهٔ ۲٫۶,
   486→   *                                maxDomRows? (number — سقف رندر DOM، فیلتر روی کل)،
   487→   *                                onRendered(visibleRows) }
   488→   */
   489→  UI.filterTable = function (wrap, columns, rows, opts = {}) {
   490→    if (!wrap) return;
   491→    const state = opts.state || { filters: {} };
   492→    state.filters = state.filters || {};
   493→    // تعداد ردیف قابل نمایش (نسخهٔ ۲٫۴): اولویت با state کاربر (ماندگار)، بعد opts
   494→    if (!Number.isFinite(Number(state.visibleRows)) || Number(state.visibleRows) <= 0) {
   495→      state.visibleRows = opts.visibleRows || 50;
   496→    }
   497→    const rowHeight = opts.rowHeight || 42;
   498→    const tableClass = opts.tableClass || 'cut-table';
   499→    const filterableCols = columns.filter((c) => c.filterable !== false);
   500→
   501→    /* ---------- تطبیق یک رکورد با همهٔ فیلترهای فعال ---------- */
   502→    const rowMatches = (row) => filterableCols.every((c) => {
   503→      const f = UI.filterStateOf(state, c.key);
   504→      if (String(f.value ?? '').trim() === '') return true;
   505→      return UI.filterContains(c.tokens ? c.tokens(row) : [String(row[c.key] ?? '')], f.value, f.invert);
   506→    });
   507→
   508→    /* ---------- ساخت سرستون + ردیف فیلتر (یک‌بار در هر رندر کامل) ---------- */
   509→    const headRow = columns.map((c) => `<th${c.headerTitle ? ` title="${U.escapeHtml(c.headerTitle)}"` : ''}${c.width ? ` style="width:${Number(c.width)}px;min-width:${Number(c.width)}px"` : ''}>${c.label}</th>`).join('');
   510→    const filterRow = columns.map((c) => {
   511→      if (c.filterable === false) return '<th class="cut-filter-cell"></th>';
   512→      const f = UI.filterStateOf(state, c.key);
   513→      return `
   514→      <th class="cut-filter-cell" aria-label="فیلتر ${U.escapeHtml(String(c.label))}">
   515→        <div class="ft-filter-box">
   516→          <button type="button" class="ft-invert${f.invert ? ' on' : ''}" data-ft-invert="${U.escapeHtml(c.key)}"
   517→                  aria-pressed="${String(!!f.invert)}"
   518→                  title="${f.invert ? 'منطق فعلی: شامل نشود — برای بازگشت به «شامل بودن» کلیک کنید' : 'برعکس‌کردن منطق: شامل نشود'}"
   519→                  aria-label="برعکس‌کردن منطق فیلتر ستون ${U.escapeHtml(String(c.label))}">
   520→            ${UI.icons.notContains}
   521→          </button>
   522→          <input type="text" class="cut-filter-input" data-ft-filter="${U.escapeHtml(c.key)}"
   523→                 value="${U.escapeHtml(f.value ?? '')}"
   524→                 autocomplete="off" placeholder="فیلتر…" inputmode="text"
   525→                 aria-label="فیلتر ستون ${U.escapeHtml(String(c.label))}">
   526→        </div>
   527→      </th>`;
   528→    }).join('');
   529→
   530→    /* ---------- باکس‌های جمع زیر ستون‌ها (نسخهٔ ۲٫۶) ----------
   531→       opts.sums = { colKey: { type: 'number' | 'time' | 'pair',
   532→                               weightOf?: (row, key) => number|null } }
   533→       جمع همیشه از دیتای فیلترشدهٔ فعلی (visible) محاسبه می‌شود. */
   534→    const sumSpecs = opts.sums || null;
   535→    const hasSums = !!(sumSpecs && Object.keys(sumSpecs).length);
   536→
   537→    /** ساخت HTML یک باکس جمع بر اساس نوع ستون */
   538→    const sumBoxHtml = (spec, cnt, wt, skipped) => {
   539→      if (spec.type === 'pair') {
   540→        return `<span class="ft-sum-sigma" aria-hidden="true">Σ</span>
   541→          <span class="ft-sum-pair">
   542→            <b class="ft-sum-val">${U.faNum(cnt)}</b>
   543→            <span class="ft-sum-wgt">Σ ${U.faWeightDot(wt)}</span>
   544→          </span>`;
   545→      }
   546→      const val = spec.type === 'time' ? U.faHMM(cnt) : U.faNum(cnt);
   547→      return `<span class="ft-sum-sigma" aria-hidden="true">Σ</span>
   548→        <b class="ft-sum-val${spec.type === 'time' ? ' ft-sum-time' : ''}">${val}</b>`
   549→        + (skipped ? `<sup class="ft-sum-skip" title="${U.faNum(skipped)} رکورد بدون قانون در جمع لحاظ نشد">*</sup>` : '');
   550→    };
   551→
   552→    /** به‌روزرسانی همهٔ باکس‌های جمع از ردیف‌های فیلترشده */
   553→    const updateSums = (visible) => {
   554→      if (!hasSums) return;
   555→      for (const c of columns) {
   556→        const spec = sumSpecs[c.key];
   557→        if (!spec) continue;
   558→        const box = wrap.querySelector(`[data-ft-sum-box="${c.key}"]`);
   559→        if (!box) continue;
   560→
   561→        if (spec.type === 'pair') {
   562→          let cnt = 0, wt = 0;
   563→          for (const row of visible) {
   564→            const v = Number(row[c.key]);
   565→            if (Number.isFinite(v)) cnt += v;
   566→            const w = spec.weightOf ? Number(spec.weightOf(row, c.key)) : NaN;
   567→            if (Number.isFinite(w)) wt += w;
   568→          }
   569→          box.innerHTML = sumBoxHtml(spec, cnt, wt, 0);
   570→          box.parentElement.title =
   571→            `جمع ستون «${String(c.label)}» با دیتای فیلترشده — ردیف اول: جمع تعداد · ردیف دوم: جمع وزن`;
   572→          continue;
   573→        }
   574→
   575→        let total = 0, counted = 0;
   576→        for (const row of visible) {
   577→          const v = row[c.key];
   578→          if (v === null || v === undefined || v === '') continue;   // «بدون قانون» و تهی‌ها
   579→          const n = Number(v);
   580→          if (Number.isFinite(n)) { total += n; counted++; }
   581→        }
   582→        const skipped = spec.type === 'time' ? (visible.length - counted) : 0;
   583→        box.innerHTML = sumBoxHtml(spec, total, 0, skipped);
   584→        box.parentElement.title = `جمع ستون «${String(c.label)}» — با دیتای فیلترشدهٔ فعلی`
   585→          + (spec.type === 'time' ? ` (${U.faNum(counted)} رکورد دارای قانون)` : '');
   586→      }
   587→    };
   588→
   589→    /* ---------- رندر بدنه (ایزوله — سرستون دست‌نخورده می‌ماند) ---------- */
   590→    /* سقف رندر DOM (تغییر ۳ — لیست‌های بزرگ): فیلتر روی «همهٔ» رکوردها اعمال
   591→       می‌شود؛ فقط رندر tbody سقف دارد. وضعیت/جمع‌ها از کل فیلترشده می‌آیند. */
   592→    const maxDom = Number(opts.maxDomRows) > 0 ? Number(opts.maxDomRows) : Infinity;
   593→
   594→    const renderBody = () => {
   595→      const visible = rows.filter(rowMatches);
   596→      const domRows = visible.length > maxDom ? visible.slice(0, maxDom) : visible;
   597→      const tbody = wrap.querySelector('.ft-tbody');
   598→      const statusEl = wrap.querySelector('.ft-status');
   599→      if (!tbody) return visible;
   600→
   601→      const activeCount = filterableCols.filter((c) =>
   602→        String(UI.filterStateOf(state, c.key).value ?? '').trim() !== '').length;
   603→
   604→      tbody.innerHTML = visible.length
   605→        ? domRows.map((row) => {
   606→            const cells = columns.map((c) => {
   607→              const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   608→              return `<td class="${c.className || ''}">${content}</td>`;
   609→            }).join('');
   610→            return `<tr>${cells}</tr>`;
   611→          }).join('')
   612→        : `<tr><td colspan="${columns.length}" class="muted" style="text-align:center;padding:18px">${
   613→            rows.length
   614→              ? (opts.filteredEmptyText || 'هیچ رکوردی با فیلترهای فعلی مطابقت ندارد — فیلترها را پاک کنید.')
   615→              : (opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.')
   616→          }</td></tr>`;
   617→
   618→      if (statusEl) {
   619→        statusEl.innerHTML = activeCount
   620→          ? `<span class="chip chip-strong">${U.faNum(visible.length)} از ${U.faNum(rows.length)} رکورد (فیلترشده)</span>` +
   621→            (domRows.length < visible.length
   622→              ? ` <span class="chip" title="برای پاسخ‌گویی فقط ${U.faNum(domRows.length)} ردیف اولِ فیلترشده رندر شده — با فیلتر دقیق‌تر رکوردهای دلخواه را ببینید">نمایش ${U.faNum(domRows.length)} ردیف</span>`
   623→              : '')
   624→          : `<span class="chip">${U.faNum(rows.length)} رکورد</span>` +
   625→            (domRows.length < visible.length ? ` <span class="chip">نمایش ${U.faNum(domRows.length)} ردیف اول</span>` : '');
   626→      }
   627→
   628→      updateSums(visible);     // نسخهٔ ۲٫۶: جمع‌ها با دیتای فیلترشده به‌روز می‌شوند
   629→
   630→      if (typeof opts.onRendered === 'function') opts.onRendered(visible, rows);
   631→      return visible;
   632→    };
   633→
   634→    /* ---------- اسکلت جدول (یک‌بار) ---------- */
   635→    if (!rows.length && !Object.values(state.filters).some((f) => f && String(f.value ?? '').trim() !== '')) {
   636→      wrap.innerHTML = `
   637→        <div class="empty-state">
   638→          ${UI.icons.alert}
   639→          <h4>رکوردی برای نمایش وجود ندارد</h4>
   640→          <p>${opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.'}</p>
   641→        </div>`;
   642→      if (typeof opts.onRendered === 'function') opts.onRendered([], rows);
   643→      return;
   644→    }
   645→
   646→    const applyHeight = (scrollEl, n) => {
   647→      scrollEl.style.height = `calc(${n} * ${rowHeight}px)`;
   648→      scrollEl.style.maxHeight = `calc(${n} * ${rowHeight}px)`;
   649→    };
   650→
   651→    // colgroup عرض ستون‌ها (نسخهٔ ۲٫۴) + جدول fixed-cols در صورت وجود عرض
   652→    const hasWidths = columns.some((c) => c.width);
   653→    const colgroup = hasWidths
   654→      ? `<colgroup>${columns.map((c) => `<col${c.width ? ` style="width:${Number(c.width)}px"` : ''}>`).join('')}</colgroup>`
   655→      : '';
   656→
   657→    const heightStyle = `height: calc(${state.visibleRows} * ${rowHeight}px); max-height: calc(${state.visibleRows} * ${rowHeight}px);`;
   658→
   659→    /* ردیف جمع چسبان زیر جدول (نسخهٔ ۲٫۶) — sticky bottom، هم‌عرض ستون‌ها؛
   660→       مقدار هر باکس در renderBody/updateSums با دیتای فیلترشده پر می‌شود. */
   661→    const sumFoot = hasSums
   662→      ? `<tfoot class="ft-sum-row"><tr>${columns.map((c) => {
   663→          if (!sumSpecs[c.key]) return '<td class="ft-sum-cell"></td>';
   664→          return `<td class="ft-sum-cell">
   665→            <div class="ft-sum-box" data-ft-sum-box="${c.key}"><span class="muted">…</span></div>
   666→          </td>`;
   667→        }).join('')}</tr></tfoot>`
   668→      : '';
   669→
   670→    wrap.innerHTML = `
   671→      <div class="cut-table-scroll" style="${heightStyle}">
   672→        <table class="data-table ${tableClass}${hasWidths ? ' fixed-cols' : ''}">
   673→          ${colgroup}
   674→          <thead>
   675→            <tr class="cut-head-row">${headRow}</tr>
   676→            <tr class="cut-filter-row">${filterRow}</tr>
   677→          </thead>
   678→          <tbody class="ft-tbody"></tbody>
   679→          ${sumFoot}
   680→        </table>
   681→      </div>
   682→      <div class="cut-filter-bar">
   683→        <span class="muted" style="font-size:.7rem">${opts.footerNote || 'فیلتر هر ستون در زیر سرستون همان ستون · منطق: شامل بودن (عدد و متن)'}</span>
   684→        <span class="ft-status"></span>
   685→        ${opts.rowsControl === false ? '' : `
   686→        <label class="ft-rows" title="تعداد ردیف قابل نمایش قبل از اسکرول — ارتفاع جدول طبق این عدد تغییر می‌کند">
   687→          <span>ردیف‌های قابل نمایش:</span>
   688→          <input type="number" class="ft-rows-input" data-ft-rows min="${RM.config.VISIBLE_ROWS_MIN}" max="${RM.config.VISIBLE_ROWS_MAX}" step="1" value="${state.visibleRows}" inputmode="numeric" aria-label="تعداد ردیف قابل نمایش">
   689→        </label>`}
   690→        <button type="button" class="btn btn-ghost btn-sm" data-ft-clear>پاک‌کردن فیلترها</button>
   691→      </div>`;
   692→
   693→    /* ---------- ارتفاع داینامیک سرستون (Wrap Text — نسخهٔ ۲٫۵) ----------
   694→       عنوان ستون‌ها ممکن است با عرض کم ستون چندخط شود؛ ارتفاع واقعی ردیف
   695→       سرستون اندازه‌گیری و در متغیر CSS --ft-head-h ثبت می‌شود تا ردیف فیلتر
   696→       چسبان دقیقاً زیر آن بنشیند. ResizeObserver تغییر عرض (پنل عرض ستون‌ها /
   697→       تغییر اندازهٔ پنجره) را دنبال می‌کند و مقدار را هم‌گام نگه می‌دارد. */
   698→    const syncHeadHeight = () => {
   699→      const headTr = wrap.querySelector('.cut-head-row');
   700→      if (!headTr) return;
   701→      const h = headTr.offsetHeight;
   702→      if (h > 0) wrap.style.setProperty('--ft-head-h', `${h}px`);
   703→    };
   704→    syncHeadHeight();
   705→    if (typeof ResizeObserver !== 'undefined') {
   706→      const headTr = wrap.querySelector('.cut-head-row');
   707→      if (wrap._ftHeadRO) { try { wrap._ftHeadRO.disconnect(); } catch (e) { /* noop */ } }
   708→      if (headTr) {
   709→        wrap._ftHeadRO = new ResizeObserver(syncHeadHeight);
   710→        wrap._ftHeadRO.observe(headTr);
   711→      }
   712→    }
   713→
   714→    /* ---------- رویدادها ---------- */
   715→
   716→    // ورود متن فیلتر → فقط tbody دوباره رندر می‌شود (فوکوس و caret حفظ می‌شود)
   717→    let timer = null;
   718→    wrap.querySelectorAll('[data-ft-filter]').forEach((el) => {
   719→      el.addEventListener('input', () => {
   720→        const f = UI.filterStateOf(state, el.dataset.ftFilter);
   721→        f.value = el.value;
   722→        clearTimeout(timer);
   723→        timer = setTimeout(renderBody, 180);
   724→      });
   725→    });
   726→
   727→    // دکمهٔ کوچک «≠» → سوییچ منطق شامل‌بودن/شامل‌نشدن
   728→    wrap.querySelectorAll('[data-ft-invert]').forEach((btn) => {
   729→      btn.addEventListener('click', () => {
   730→        const f = UI.filterStateOf(state, btn.dataset.ftInvert);
   731→        f.invert = !f.invert;
   732→        btn.classList.toggle('on', f.invert);
   733→        btn.setAttribute('aria-pressed', String(f.invert));
   734→        btn.title = f.invert
   735→          ? 'منطق فعلی: شامل نشود — برای بازگشت به «شامل بودن» کلیک کنید'
   736→          : 'برعکس‌کردن منطق: شامل نشود';
   737→        renderBody();
   738→      });
   739→    });
   740→
   741→    // پاک‌کردن همهٔ فیلترها
   742→    wrap.querySelector('[data-ft-clear]').addEventListener('click', () => {
   743→      for (const c of filterableCols) {
   744→        const f = UI.filterStateOf(state, c.key);
   745→        f.value = '';
   746→        f.invert = false;
   747→      }
   748→      wrap.querySelectorAll('[data-ft-filter]').forEach((el) => { el.value = ''; });
   749→      wrap.querySelectorAll('[data-ft-invert]').forEach((b) => {
   750→        b.classList.remove('on');
   751→        b.setAttribute('aria-pressed', 'false');
   752→        b.title = 'برعکس‌کردن منطق: شامل نشود';
   753→      });
   754→      renderBody();
   755→      UI.toast('فیلترها پاک شد.', 'info');
   756→    });
   757→
   758→    // باکس «ردیف‌های قابل نمایش» (نسخهٔ ۲٫۴) — فقط ارتفاع اسکرول تغییر می‌کند؛
   759→    // جدول/فیلترها/فوکوس دست‌نخورده می‌مانند. ماندگاری با onRowsChange.
   760→    const rowsInput = wrap.querySelector('[data-ft-rows]');
   761→    if (rowsInput) {
   762→      let rowsTimer = null;
   763→      rowsInput.addEventListener('input', () => {
   764→        clearTimeout(rowsTimer);
   765→        rowsTimer = setTimeout(() => {
   766→          const n = Math.min(
   767→            Math.max(U.parseInt(rowsInput.value) || RM.config.VISIBLE_ROWS_MIN, RM.config.VISIBLE_ROWS_MIN),
   768→            RM.config.VISIBLE_ROWS_MAX
   769→          );
   770→          state.visibleRows = n;
   771→          rowsInput.value = String(n);
   772→          const scrollEl = wrap.querySelector('.cut-table-scroll');
   773→          if (scrollEl) applyHeight(scrollEl, n);
   774→          if (typeof opts.onRowsChange === 'function') opts.onRowsChange(n);
   775→        }, 260);
   776→      });
   777→    }
   778→
   779→    renderBody();
   780→  };
   781→
   782→  /* ---------------- پنل ترتیب ستون‌ها (مشترک — چیپ‌های کشیدنی + باکس عرض) ---------------- */
   783→
   784→  /**
   785→   * پنل «ترتیب ستون‌ها» — نسخهٔ ۲٫۴:
   786→   *   - ترتیب با کشیدن‌ورها (drag) تغییر می‌کند (فلش‌های ↑/↓ حذف شدند)
   787→   *   - بجای فلش‌ها، هر ستون یک باکس عددی «عرض (px)» دارد؛ کاربر عرض دلخواه
   788→   *     هر ستون را وارد می‌کند (صفر یا خالی → عرض خودکار)
   789→   * برای جدول «گزارش تولید ستاپ‌ها».
   790→   * تغییر ۳ (جدید): اگر handlers.onToggle داده شود، هر چیپ یک دکمهٔ «چشم»
   791→   *   هم دارد — نمایش/مخفی‌کردن همان ستون در جدول (ستون مخفی از جدول
   792→   *   حذف می‌شود اما در پنل می‌ماند و با کلیک دوباره برمی‌گردد).
   793→   * @param {HTMLElement} wrap      عنصر نگهدارندهٔ چیپ‌ها
   794→   * @param {Array}        items    [{key, label, width, hidden?}] به ترتیب فعلی
   795→   * @param {object}       handlers { onMove(from, to), onWidth(key, width),
   796→   *                                  onToggle?(key, hidden) — اختیاری، emptyText? }
   797→   */
   798→  UI.orderPanel = function (wrap, items, handlers) {
   799→    if (!wrap) return;
   800→    wrap.innerHTML = items.map((it, i) => `
   801→      <div class="sort-chip order-chip${it.hidden ? ' col-hidden' : ''}" draggable="true" data-index="${i}" title="برای تغییر ترتیب بکشید">
   802→        <span class="grip">${UI.icons.grip}</span>
   803→        <span class="sort-priority">${U.faNum(i + 1)}</span>
   804→        <span class="sort-label">${U.escapeHtml(it.label)}</span>
   805→        ${handlers.onToggle ? `
   806→        <button type="button" class="order-vis${it.hidden ? '' : ' on'}" data-order-vis="${U.escapeHtml(it.key)}"
   807→                aria-pressed="${String(!it.hidden)}"
   808→                title="${it.hidden ? 'این ستون در جدول مخفی است — برای نمایش دوباره کلیک کنید' : 'مخفی‌کردن این ستون در جدول (از پنل دوباره نمایش داده می‌شود)'}"
   809→                aria-label="نمایش/مخفی‌کردن ستون ${U.escapeHtml(it.label)}">
   810→          ${it.hidden ? UI.icons.eyeOff : UI.icons.eye}
   811→        </button>` : ''}
   812→        <label class="order-width" title="عرض این ستون در جدول (پیکسل) — صفر یا خالی = خودکار">
   813→          <span class="order-width-unit">px</span>
   814→          <input type="number" class="order-width-input" data-order-width="${U.escapeHtml(it.key)}"
   815→                 min="0" max="1200" step="1" inputmode="numeric" aria-label="عرض ستون ${U.escapeHtml(it.label)}"
   816→                 value="${it.width != null ? Number(it.width) : ''}" placeholder="خودکار">
   817→        </label>
   818→      </div>`).join('') ||
   819→      `<span class="muted" style="font-size:.78rem">${U.escapeHtml(handlers.emptyText || 'بدون ستون')}</span>`;
   820→
   821→    // باکس عرض هر ستون (نسخهٔ ۲٫۴) — debounce سبک؛ خالی/۰ = عرض خودکار
   822→    let widthTimer = null;
   823→    wrap.querySelectorAll('[data-order-width]').forEach((inp) => {
   824→      inp.addEventListener('input', () => {
   825→        clearTimeout(widthTimer);
   826→        const key = inp.dataset.orderWidth;
   827→        widthTimer = setTimeout(() => {
   828→          const v = U.parseInt(inp.value);
   829→          handlers.onWidth(key, v === null ? 0 : Math.min(Math.max(v, 0), 1200));
   830→        }, 320);
   831→      });
   832→    });
   833→
   834→    // دکمهٔ «چشم» — نمایش/مخفی‌کردن ستون در جدول (تغییر ۳)
   835→    if (handlers.onToggle) {
   836→      wrap.querySelectorAll('[data-order-vis]').forEach((btn) => {
   837→        btn.addEventListener('click', (e) => {
   838→          e.stopPropagation();
   839→          const chip = btn.closest('.order-chip');
   840→          const idx = chip ? Number(chip.dataset.index) : -1;
   841→          const it = idx >= 0 ? items[idx] : null;
   842→          if (it) handlers.onToggle(it.key, !it.hidden);
   843→        });
   844→      });
   845→    }
   846→
   847→    // کشیدن و رها کردن
   848→    let dragSrcIndex = null;
   849→    wrap.querySelectorAll('.order-chip').forEach((chip) => {
   850→      chip.addEventListener('dragstart', (e) => {
   851→        dragSrcIndex = Number(chip.dataset.index);
   852→        chip.classList.add('dragging');
   853→        e.dataTransfer.effectAllowed = 'move';
   854→        e.dataTransfer.setData('text/plain', String(dragSrcIndex));
   855→      });
   856→      chip.addEventListener('dragend', () => {
   857→        chip.classList.remove('dragging');
   858→        wrap.querySelectorAll('.order-chip').forEach((c) => c.classList.remove('drag-over'));
   859→        dragSrcIndex = null;
   860→      });
   861→      chip.addEventListener('dragover', (e) => {
   862→        e.preventDefault();
   863→        e.dataTransfer.dropEffect = 'move';
   864→        if (!chip.classList.contains('drag-over')) chip.classList.add('drag-over');
   865→      });
   866→      chip.addEventListener('dragleave', () => chip.classList.remove('drag-over'));
   867→      chip.addEventListener('drop', (e) => {
   868→        e.preventDefault();
   869→        const from = dragSrcIndex;
   870→        const to = Number(chip.dataset.index);
   871→        if (from === null || from === to) return;
   872→        handlers.onMove(from, to);
   873→      });
   874→    });
   875→  };
   876→
   877→  RM.ui = UI;
   878→})();
