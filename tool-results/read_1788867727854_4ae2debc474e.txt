    56→  /** ساخت/بازیابی نمونهٔ چارت با شناسهٔ المان */
    57→  Dashboard.chartOf = function (elId) {
    58→    const el = document.getElementById(elId);
    59→    if (!el || !Dashboard.hasECharts()) return null;
    60→    if (Dashboard._charts[elId]) return Dashboard._charts[elId];
    61→    if (!el.clientWidth) return null;   // مخفی است — هنگام نمایش ساخته می‌شود
    62→    const inst = window.echarts.init(el);
    63→    Dashboard._charts[elId] = inst;
    64→    return inst;
    65→  };
    66→
    67→  /** اعمال گزینه‌ها روی چارت (بدون Merge — رندر تازه) */
    68→  Dashboard.applyChart = function (elId, option) {
    69→    const inst = Dashboard.chartOf(elId);
    70→    if (!inst) { Dashboard._pendingCharts = true; return false; }
    71→    inst.setOption(option, { notMerge: true });
    72→    return true;
    73→  };
    74→
    75→  /** حذف امن نمونهٔ چارت (حالت خالی — جلوگیری از نمونهٔ سرگردان) */
    76→  Dashboard.disposeChart = function (elId) {
    77→    if (Dashboard._charts[elId]) {
    78→      try { Dashboard._charts[elId].dispose(); } catch (e) { /* noop */ }
    79→      delete Dashboard._charts[elId];
    80→    }
    81→  };

    83→  /** رندر مجدد همهٔ نمودارها (تعویض تم / بازگشت به تب) */
    84→  Dashboard.rerenderCharts = function () {
    85→    if (!Dashboard._state) return;
    86→    Dashboard.renderMachineTimeChart(Dashboard._state);
    87→    Dashboard.renderRawStatusChart(Dashboard._state);
    88→    Dashboard._pendingCharts = false;
    89→  };
    90→
    91→  /** هوک نمایش تب — نمودارهای معوق/تازه‌شده را رندر می‌کند */

    95→      Dashboard._charts[id].resize();
    96→    }

   112→    themeObserver.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });

   192→  /* ---------- چارت مانده زمان متالایز به تفکیک دستگاه (نسخهٔ ۲٫۳ — ECharts) ---------- */
   193→
   194→  /**
   195→   * نمودار میله‌ای افقی گروهی ECharts — هر دستگاه (۱/۲/۳) دو میله:
   196→   *   «مانده زمان ستاپ» (فیروزه‌ای) و «مانده زمان تولید شده» (کهربایی).
   197→   * Tooltip فارسی با مدت دقیق + برچسب ساعت روی محور. Fallback: میله‌های CSS.
   198→   */
   199→  Dashboard.renderMachineTimeChart = function (state) {
   200→    const s = state.summary;
   201→    const machines = RM.config.VALID_MACHINES;
   202→
   203→    // بیشینه برای مقیاس
   204→    let maxVal = 0;
   205→    for (const m of machines) {
   206→      const t = s.machineTime[m];
   207→      if (t) maxVal = Math.max(maxVal, t.setup, t.produced);
   208→    }
   209→
   210→    const chipsEl = document.getElementById('machine-time-chips');
   211→    chipsEl.innerHTML = `
   212→      <span class="chip">قوانین زمان: ${U.faNum(s.timeRulesCount)}</span>
   213→      <span class="chip chip-strong">جمع مانده زمان ستاپ: ${U.faDuration(s.totalRemainingSetupTime)}</span>
   214→      <span class="chip chip-ok">جمع مانده زمان تولید شده: ${U.faDuration(s.totalRemainingProducedTime)}</span>
   215→      ${s.timeRuleCoverage.uncovered > 0
   216→        ? `<span class="chip chip-warn">${U.faNum(s.timeRuleCoverage.uncovered)} رکورد بدون قانون</span>`
   217→        : ''}
   218→    `;
   219→
   220→    const totalSetupH = U.hoursOf(s.totalRemainingSetupTime);
   221→    const totalProducedH = U.hoursOf(s.totalRemainingProducedTime);
   222→
   223→    document.getElementById('machine-time-footer').innerHTML = `
   224→      <div class="mt-total">
   225→        جمع کل — ستاپ: <b>${U.faDuration(s.totalRemainingSetupTime)}</b>${totalSetupH !== null ? ` (~${U.faNum(totalSetupH)} ساعت)` : ''} ·
   226→        تولید شده: <b>${U.faDuration(s.totalRemainingProducedTime)}</b>${totalProducedH !== null ? ` (~${U.faNum(totalProducedH)} ساعت)` : ''}
   227→      </div>`;
   228→
   229→    if (!Dashboard.hasECharts()) {
   230→      Dashboard.renderMachineTimeChartFallback(state);   // کتابخانه در دسترس نیست
   231→      return;
   232→    }
   233→
   234→    const P = Dashboard.chartPalette();
   235→    const cats = machines.map((m) => `دستگاه ${U.faNumPlain(m)}`).reverse();   // دستگاه ۱ بالا
   236→    const setupData = machines.map((m) => {
   237→      const t = s.machineTime[m];
   238→      return t && t.setup ? t.setup : 0;
   239→    }).reverse();
   240→    const producedData = machines.map((m) => {
   241→      const t = s.machineTime[m];
   242→      return t && t.produced ? t.produced : 0;
   243→    }).reverse();
   244→    const recordsByMachine = machines.map((m) => {
   245→      const t = s.machineTime[m];
   246→      return t ? t.records : 0;
   247→    }).reverse();
   248→
   249→    const option = {
   250→      animationDuration: 550,
   251→      animationEasing: 'cubicOut',
   252→      textStyle: { fontFamily: P.fontFamily },
   253→      grid: { left: 10, right: 26, top: 42, bottom: 8, containLabel: true },
   254→      legend: {
   255→        top: 4,
   256→        right: 8,
   257→        itemWidth: 14,
   258→        itemHeight: 9,
   259→        itemGap: 18,
   260→        textStyle: { color: P.muted, fontFamily: P.fontFamily, fontSize: 11 },
   261→        data: ['مانده زمان ستاپ', 'مانده زمان تولید شده'],
   262→      },
   263→      tooltip: {
   264→        trigger: 'axis',
   265→        axisPointer: { type: 'shadow', shadowStyle: { color: P.splitLine } },
   266→        backgroundColor: P.tooltipBg,
   267→        borderColor: P.tooltipBorder,
   268→        borderWidth: 1,
   269→        padding: [8, 12],
   270→        textStyle: { color: P.text, fontFamily: P.fontFamily, fontSize: 12 },
   271→        formatter: (params) => {
   272→          const idx = params[0] ? params[0].dataIndex : 0;
   273→          const m = RM.config.VALID_MACHINES[machines.length - 1 - idx] || '—';
   274→          const lines = [`<b>دستگاه ${U.faNumPlain(m)}</b> · ${U.faNum(recordsByMachine[idx] || 0)} رکورد`];
   275→          for (const p of params) {
   276→            lines.push(`${p.marker} ${p.seriesName}: <b>${U.faDuration(p.value)}</b>${p.value ? ` (~${U.faNum(Math.round(p.value / 60))} ساعت)` : ''}`);
   277→          }
   278→          return lines.join('<br>');
   279→        },
   280→      },
   281→      xAxis: {
   282→        type: 'value',
   283→        max: (v) => Math.max(v.max * 1.08, 60),
   284→        axisLabel: {
   285→          color: P.muted,
   286→          fontFamily: P.fontFamily,
   287→          fontSize: 10.5,
   288→          formatter: (v) => `${U.faNumPlain(Math.round(v / 60))} ساعت`,
   289→        },
   290→        splitLine: { lineStyle: { color: P.splitLine, width: 1 } },
   291→        axisLine: { show: false },
   292→        axisTick: { show: false },
   293→      },
   294→      yAxis: {
   295→        type: 'category',
   296→        data: cats,
   297→        axisLabel: { color: P.text, fontFamily: P.fontFamily, fontSize: 12, fontWeight: 600 },
   298→        axisLine: { lineStyle: { color: P.axisLine } },
   299→        axisTick: { show: false },
   300→      },
   301→      series: [
   302→        {
   303→          name: 'مانده زمان ستاپ',
   304→          type: 'bar',
   305→          data: setupData,
   306→          barMaxWidth: 18,
   307→          barGap: '28%',
   308→          itemStyle: { color: P.setup, borderRadius: [4, 4, 4, 4] },
   309→          emphasis: { itemStyle: { color: P.setup, opacity: 0.82 } },
   310→          label: {
   311→            show: true,
   312→            position: 'right',
   313→            color: P.muted,
   314→            fontFamily: P.fontFamily,
   315→            fontSize: 10,
   316→            formatter: (p) => (p.value ? `${U.faNumPlain(Math.round(p.value / 60))} ساعت` : ''),
   317→          },
   318→        },
   319→        {
   320→          name: 'مانده زمان تولید شده',
   321→          type: 'bar',
   322→          data: producedData,
   323→          barMaxWidth: 18,
   324→          itemStyle: { color: P.metal, borderRadius: [4, 4, 4, 4] },
   325→          emphasis: { itemStyle: { color: P.metal, opacity: 0.82 } },
   326→          label: {
   327→            show: true,
   328→            position: 'right',
   329→            color: P.muted,
   330→            fontFamily: P.fontFamily,
   331→            fontSize: 10,
   332→            formatter: (p) => (p.value ? `${U.faNumPlain(Math.round(p.value / 60))} ساعت` : ''),
   333→          },
   334→        },
   335→      ],
   336→    };
   337→
   338→    Dashboard.applyChart('machine-time-chart', option);
   339→  };
   340→
   341→  /* ---------- نمودار دوناتی وضعیت گروه‌های رول خام (نسخهٔ ۲٫۳) ---------- */
   342→
   343→  /**
   344→   * دونات ECharts — سهم گروه‌های «دارای ستاپ» (زمردی) / «بدون ستاپ» (کهربایی)
   345→   * و «غیرقابل تطبیق» (خاکستری) از گروه‌بندی ۳فاکتوری رول‌های خام.
   346→   */
   347→  Dashboard.renderRawStatusChart = function (state) {
   348→    const groups = state.rawRollGroups || [];
   349→    const el = document.getElementById('raw-status-chart');
   350→    if (!el) return;
   351→
   352→    let matched = 0, unmatched = 0, noRule = 0;
   353→    for (const g of groups) {
   354→      if (!g.hasRule) noRule++;
   355→      else if (g.matched) matched++;
   356→      else unmatched++;
   357→    }
   358→    const total = groups.length;
   359→
   360→    if (!total) {
   361→      Dashboard.disposeChart('raw-status-chart');
   362→      el.innerHTML = `
   363→        <div class="empty-state" style="min-height:220px">
   364→          ${UI.icons.alert}
   365→          <p>هنوز گروهی وجود ندارد — فایل «رول‌های موجود» را در تب «ورود داده‌ها» بارگذاری کنید.</p>
   366→        </div>`;
   367→      return;
   368→    }
   369→
   370→    if (!Dashboard.hasECharts()) {
   371→      // Fallback ساده: چیپ‌های متنی
   372→      el.innerHTML = `
   373→        <div class="mt-rows" style="padding:12px">
   374→          <div class="mt-row"><div class="mt-label"><b style="color:var(--raw)">دارای ستاپ</b></div><b>${U.faNum(matched)}</b></div>
   375→          <div class="mt-row"><div class="mt-label"><b style="color:var(--metal)">بدون ستاپ</b></div><b>${U.faNum(unmatched)}</b></div>
   376→          <div class="mt-row"><div class="mt-label"><b class="muted">غیرقابل تطبیق</b></div><b>${U.faNum(noRule)}</b></div>
   377→        </div>`;
   378→      return;
   379→    }
   380→
   381→    const P = Dashboard.chartPalette();
   382→    const pct = (v) => `${U.faNumPlain(Math.round((v / total) * 100))}٪`;
   383→
   384→    const option = {
   385→      animationDuration: 550,
   386→      textStyle: { fontFamily: P.fontFamily },
   387→      tooltip: {
   388→        trigger: 'item',
   389→        backgroundColor: P.tooltipBg,
   390→        borderColor: P.tooltipBorder,
   391→        borderWidth: 1,
   392→        padding: [8, 12],
   393→        textStyle: { color: P.text, fontFamily: P.fontFamily, fontSize: 12 },
   394→        formatter: (p) => `<b>${p.name}</b><br>${U.faNum(p.value)} گروه · ${pct(p.value)}`,
   395→      },
   396→      legend: {
   397→        bottom: 6,
   398→        right: 10,
   399→        orient: 'vertical',
   400→        itemWidth: 12,
   401→        itemHeight: 12,
   402→        itemGap: 10,
   403→        textStyle: { color: P.muted, fontFamily: P.fontFamily, fontSize: 11 },
   404→      },
   405→      series: [
   406→        {

   513→  Dashboard.renderAlerts = function (warnings) {
   514→    const wrap = document.getElementById('alerts-list');
   515→    const panel = document.getElementById('alerts-panel');
   516→
   517→    if (!warnings.length) {
   518→      panel.hidden = true;
   519→      wrap.innerHTML = '';
   520→      return;
   521→    }
   522→
   523→    panel.hidden = false;
   524→    document.getElementById('alerts-count').textContent = U.faNum(warnings.length);
   525→
   526→    wrap.innerHTML = warnings.map((w) => {
   527→      const meta = SEVERITY_META[w.severity] || SEVERITY_META.warning;
   528→      const hasDrill = !!w.drill;
   529→      return `
   530→        <div class="alert-item ${meta.cls}" role="alert" data-alert="${U.escapeHtml(w.type)}">
   531→          <span class="alert-badge">${meta.label}</span>
   532→          <div class="alert-body">
   533→            <b>${U.escapeHtml(w.title)}</b>
   534→            <p>${U.escapeHtml(w.detail)}</p>
   535→          </div>
   536→          ${hasDrill
   537→            ? `<button type="button" class="btn btn-ghost btn-sm" data-alert-drill="${U.escapeHtml(w.type)}" title="نمایش رکوردهای عامل این هشدار">مشاهدهٔ رکوردها</button>`
   538→            : (w.action
   539→              ? `<button type="button" class="btn btn-ghost btn-sm" data-alert-action="${w.action}">مشاهده</button>`
   540→              : '')}
   541→        </div>`;
   542→    }).join('');
   543→
   544→    // دکمهٔ «مشاهدهٔ رکوردها» → Drill-down دقیق رکوردهای عامل همان هشدار

   552→    // هشدارهای بدون Drill → رفتن به بخش مرتبط (رفتار قبلی حفظ شد)
   553→    wrap.querySelectorAll('[data-alert-action]').forEach((btn) => {

   558→  /**
   559→   * بازکردن Drill-down یک هشدار — دقیقاً رکوردهای عامل همان Rule.
   560→   * موجودیت‌های Roll از مودال مشترک با ستون‌های Mapping استفاده می‌کنند؛
   561→   * هشدارهای مربوط به ستاپ/قانون جدول اختصاصی خودشان را می‌گیرند.
   562→   */
   563→  Dashboard.openDrill = async function (warning) {
   564→    const d = warning.drill;
   565→    if (!d) return;
   566→
   567→    try {
   568→      /* --- هشدارهای مربوط به ستاپ‌ها --- */
   569→      if (d.source === 'setups') {
   570→        const state = Dashboard._state;
   571→        const rows = (d.filter === 'capacity-conflict')
   572→          ? state.setupReport.filter((r) => r.conflict > 0)
   573→          : (d.filter === 'no-time-rule')
   574→            ? state.setupReport.filter((r) => r.remainingSetupTime === null)
   575→            : state.setupReport.filter((r) => r.thickness === null || r.standardLength === null);
   576→
   577→        const timeCols = (d.filter === 'no-time-rule')
   578→          ? '<th>مانده زمان</th>'
   579→          : '<th>خام</th><th>متالایز</th><th>مغایرت</th>';
   580→
   581→        UI.infoModal(warning.title, `
   582→          <div class="drill-subtitle"><b>${U.faNum(rows.length)}</b> رکورد ستاپ</div>
   583→          <div class="table-wrap table-scroll" style="max-height:420px">
   584→            <table class="data-table">
   585→              <thead><tr><th>ستاپ</th><th>دستگاه</th><th>عرض</th><th>ضخامت</th><th>متراژ استاندارد</th><th>تعداد</th>${timeCols}</tr></thead>
   586→              <tbody>${rows.map((r) => `
   587→                <tr>
   588→                  <td>${U.faNum(r.setupNumber)}</td>
   589→                  <td>${U.faNum(r.machineNumber)}</td>
   590→                  <td>${U.faWidth(r.width)}</td>
   591→                  <td>${r.thickness === null ? '<span class="muted">ناقص</span>' : U.faNum(r.thickness)}</td>
   592→                  <td>${r.standardLength === null ? '<span class="muted">ناقص</span>' : U.faNum(r.standardLength)}</td>
   593→                  <td class="count-cell">${U.faNum(r.rollCount)}</td>
   594→                  ${d.filter === 'no-time-rule'
   595→                    ? `<td><span class="muted">بدون قانون زمان</span></td>`
   596→                    : `<td>${U.faNum(r.rawExisting)}</td>
   597→                       <td>${U.faNum(r.metallized)}</td>
   598→                       <td>${r.conflict > 0 ? `<span class="pill-bad status-pill">⚠ ${U.faNum(r.conflict)}</span>` : '—'}</td>`}
   599→                </tr>`).join('')}
   600→              </tbody>
   601→            </table>
   602→          </div>
   603→        `);
   604→        return;
   605→      }
   606→
   607→      /* --- هشدار هم‌پوشانی قوانین --- */
   608→      if (d.source === 'rules') {
   609→        UI.infoModal(warning.title, `
   610→          <div class="drill-subtitle"><b>${U.faNum(d.pairs.length)}</b> جفت قانون هم‌پوشان</div>
   611→          <div class="table-wrap">
   612→            <table class="data-table">
   613→              <thead><tr><th>ضخامت</th><th>بازهٔ قانون ۱</th><th>بازهٔ قانون ۲</th></tr></thead>
   614→              <tbody>${d.pairs.map(([a, b]) => `
   615→                <tr>
   616→                  <td class="thickness-cell">${U.faNum(a.thickness)}</td>
   617→                  <td>${U.faNum(a.minLength)} تا ${U.faNum(a.maxLength)} → ${U.faNum(a.standardLength)}</td>
   618→                  <td>${U.faNum(b.minLength)} تا ${U.faNum(b.maxLength)} → ${U.faNum(b.standardLength)}</td>
   619→                </tr>`).join('')}
   620→              </tbody>
   621→            </table>
   622→          </div>
   623→        `);
   624→        return;
   625→      }
   626→
   627→      /* --- هشدارهای رکوردی (rolls / archived) → مودال مشترک --- */
   628→      const sourceKey = d.source;
   629→      const table = RM.config.IMPORT_TARGETS[sourceKey].table;
   630→      let records = [];
   631→
   632→      if (d.issue) {
   633→        // نقص دادهٔ مستقیم: بدون عرض / بدون متراژ / بدون ضخامت / شماره رول / نوع فیلم
   634→        // رزولور مشترک نقص‌ها («length» → actualLength در مدل داخلی)
   635→        const all = await db[table].toArray();
   636→        records = RM.normalize.recordsWithIssue(all, d.issue);
   637→      } else if (d.filter === 'all') {
   638→        // Mapping ناقص — همهٔ رکوردهای همان Dataset (فیلدهای مفقود «—» نمایش داده می‌شوند)
   639→        records = await db[table].toArray();
   640→      } else if (d.filter === 'no-rule') {
   641→        // رول‌های خام بدون قانون متراژ — از گروه‌های بدون قانون (کلید‌محور)
   642→        const state = Dashboard._state;
   643→        const groups = state.rawRollGroups.filter((g) => !g.hasRule);
   644→        const ids = groups.flatMap((g) => g.rollIds);
   645→        records = (await db.rolls.bulkGet(ids)).filter(Boolean);
   646→      } else if (d.filter === 'group') {
   647→        // گروه‌های خاص (بدون ستاپ / مازاد خام)
   648→        const state = Dashboard._state;
   649→        const keySet = new Set(d.groupKeys || []);
   650→        const groups = state.rawRollGroups.filter((g) => keySet.has(g.key));
   651→        const ids = groups.flatMap((g) => g.rollIds);
   652→        records = (await db.rolls.bulkGet(ids)).filter(Boolean);
   653→      } else if (d.filter === 'metallized-unallocated') {
   654→        // مازاد متالایز — یونیک‌های دارای کلیدِ تخصیص‌نیافته
   655→        const state = Dashboard._state;
   656→        const keySet = new Set(d.groupKeys || []);
   657→        records = state.metallizedUniqueRecords.filter((r) =>
   658→          keySet.has(U.formatKey(r.width, r.setupNumber))
   659→        );
   660→      }
   661→
   662→      await UI.recordsDrillModal(warning.title, records, { source: sourceKey });
   663→    } catch (err) {
   664→      console.error('[Drill-down]', err);
   665→      UI.toast('نمایش رکوردهای این هشدار با خطا مواجه شد.', 'error');
   666→    }
   667→  };
