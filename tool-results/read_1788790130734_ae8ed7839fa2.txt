    98→  SetupView.applyThicknessInheritance = async function () {
    99→    const { setupNumber, thickness, standardLength } = F();
   100→    const value = U.parseInt(setupNumber.value);
   101→    const hintEl = document.getElementById('hint-thickness');
   102→
   103→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   104→
   105→    if (value === null || value < 0) {
   106→      SetupView.thicknessLocked = false;
   107→      thickness.disabled = false;
   108→      hintEl.hidden = true;
   109→      return;
   110→    }
   111→
   112→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   113→    const withThickness = existing.find((s) => s.thickness !== null);
   114→
   115→    if (withThickness) {
   116→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   117→      SetupView.thicknessLocked = true;
   118→      thickness.value = String(withThickness.thickness);
   119→      thickness.disabled = true;
   120→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   121→      hintEl.hidden = false;
   122→    } else {
   123→      SetupView.thicknessLocked = false;
   124→      thickness.disabled = false;
   125→      hintEl.hidden = true;
   126→    }
   127→
   128→    SetupView.refreshStandardLengthOptions(standardLength.value);
   129→  };

   132→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   133→    const { thickness, standardLength } = F();
   134→    const thicknessValue = U.parseNumber(thickness.value);
   135→    const errEl = document.getElementById('err-standard');
   136→
   137→    const options = thicknessValue === null
   138→      ? []
   139→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   140→
   141→    let html = '<option value="">— انتخاب کنید —</option>';
   142→    for (const opt of options) {
   143→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   144→    }
   145→    standardLength.innerHTML = html;
   146→
   147→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   148→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   149→      standardLength.value = keepValue;
   150→    } else {
   151→      standardLength.value = '';
   152→    }
   153→
   154→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   155→    if (thicknessValue !== null && !options.length) {
   156→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   157→      errEl.hidden = false;
   158→    } else {
   159→      errEl.textContent = '';
   160→      errEl.hidden = true;
   161→    }

   248→      /* --- ثبت جدید --- */

   251→        SetupView.showValidationErrors(result);
   252→        return;

   402→  /**
   403→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   404→   * بلافاصله به‌روز می‌شود (روش محاسبهٔ جدید):
   405→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   406→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   407→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست همان)
   408→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   409→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه):
   410→   *   ۱۶ / ۳ / ۴۸ / ۵
   411→   */
   412→  SetupView.bulkSummaryStats = function (drafts) {
   413→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   414→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست) — «جمع عرض الگوی برش»
   415→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   416→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   417→
   418→    for (const d of drafts) {
   419→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   420→      motherRollsTotal += rollCount;
   421→
   422→      const w = U.parseNumber(d.motherToken);
   423→      if (w !== null) motherWidthSet.add(w);
   424→
   425→      const parsed = Setups.parsePattern(d.patternText);
   426→      if (parsed.ok && parsed.parts.length) {
   427→        patternWidthsTotal += parsed.parts.length * rollCount;
   428→        for (const part of parsed.parts) patternWidthSet.add(part);
   429→      }
   430→    }
   431→
   432→    return {
   433→      count: drafts.length,
   434→      motherRollsTotal,                            // ۱۶
   435→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   436→      patternWidthsTotal,                          // ۴۸
   437→      uniquePatternWidths: patternWidthSet.size,   // ۵
   438→    };
   439→  };

   630→    const b = SetupView._bulk;
   631→    if (!b) return;
   632→    const rowsIdx = rowIdx === null ? b.drafts.map((_, i) => i) : [rowIdx];

   639→      const parsed = Setups.parsePattern(d.patternText);

   644→      tr.querySelector('[data-bulk-cell="sum"]').innerHTML =
   645→        sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>';
   646→      tr.querySelector('[data-bulk-cell="trim"]').innerHTML =
   647→        trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>';
   648→    }

   692→  SetupView.applyBulkCheck = function (check) {
   693→    SetupView._bulkCheck = check;
   694→    const b = SetupView._bulk;
   695→    if (!b) return;
   696→
   697→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   698→    const globalMap = {
   699→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   700→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   701→    };
   702→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {
   703→      const errEl = document.getElementById(errId);
   704→      const inputEl = document.getElementById(inputId);
   705→      const msg = check.globalErrors[field] || '';
   706→      if (errEl) { errEl.textContent = msg; errEl.hidden = !msg; }
   707→      if (inputEl) inputEl.classList.toggle('invalid', !!msg);
   708→    }
   709→
   710→    // وضعیت هر ردیف
   711→    b.drafts.forEach((d, i) => {
   712→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   713→      if (!tr) return;
   714→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   715→      const rec = check.records[i];
   716→      const errs = rec ? rec.errors : [];
   717→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   718→      cell.innerHTML = errs.length
   719→        ? `<span class="status-pill pill-bad">✕ خطا</span>
   720→           <ul class="bulk-errors">${errs.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`
   721→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   722→    });
   723→
   724→    const commitBtn = document.getElementById('btn-bulk-commit');
   725→    const hasStructure = b.structureErrors.length > 0;
   726→    commitBtn.disabled = !(check.ok === true && !hasStructure);
   727→  };

   838→          SetupView.sortPipeline[i].direction =
   839→            SetupView.sortPipeline[i].direction === 'asc' ? 'desc' : 'asc';
   840→          SetupView.saveSort();
   841→          SetupView.renderSortBuilder();
   842→          SetupView.renderReportTable();
   843→        },
   844→        onRemove: (i) => {
   845→          SetupView.sortPipeline.splice(i, 1);
   846→          SetupView.saveSort();
   847→          SetupView.renderSortBuilder();
   848→          SetupView.renderReportTable();
   849→        },
   850→        onReorder: (from, to) => {
   851→          const [moved] = SetupView.sortPipeline.splice(from, 1);
   852→          SetupView.sortPipeline.splice(to, 0, moved);
   853→          SetupView.saveSort();
   854→          SetupView.renderSortBuilder();
   855→          SetupView.renderReportTable();
   856→        },
   857→      }
   858→    );
   859→  };

   893→  /** اعمال مرتب‌سازی چندمرحله‌ای روی ردیف‌های گزارش (از آخرین اولویت به اولین) */
   894→  SetupView.applySetupSort = function (rows) {
   895→    const sorted = [...rows];
   896→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
   897→      const { field, direction } = SetupView.sortPipeline[i];
   898→      sorted.sort((a, b) => {
   899→        if (field === 'cuttingPattern') {
   900→          const ka = Setups.normalizePatternKey(a.cuttingPattern) || '';
   901→          const kb = Setups.normalizePatternKey(b.cuttingPattern) || '';
   902→          const r = ka.localeCompare(kb, 'fa', { numeric: true });
   903→          return direction === 'desc' ? -r : r;
   904→        }
   905→        return U.compareNumeric(a[field], b[field], direction);
   906→      });
   907→    }
   908→    return sorted;
   909→  };

   928→  /** تطبیق یک ردیف با مقدار فیلتر یک ستون: عدد = برابری عددی · متن = شامل بودن */
   929→  SetupView.rowMatchesFilter = function (row, key, value) {
   930→    const raw = row[key];
   931→    const num = U.parseNumber(value);
   932→
   933→    if (num !== null && raw !== null && raw !== undefined && Number.isFinite(Number(raw))) {
   934→      return Number(raw) === num;
   935→    }
   936→    const hay = (key === 'cuttingPattern')
   937→      ? ((Setups.normalizePatternKey(raw) || '') + ' ' + String(raw ?? ''))
   938→      : String(raw ?? '');
   939→    return hay.includes(value);
   940→  };
