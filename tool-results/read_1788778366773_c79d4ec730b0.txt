    71→  SettingsView.initThemePicker = function () {
    72→    const buttons = document.querySelectorAll('[data-theme-pick]');
    73→    if (!buttons.length) return;
    74→
    75→    const apply = (theme, { persist = true, notify = false } = {}) => {
    76→      if (theme === 'dark') {
    77→        document.documentElement.setAttribute('data-theme', 'dark');
    78→      } else {
    79→        document.documentElement.removeAttribute('data-theme');
    80→      }
    81→      if (persist) {
    82→        try { localStorage.setItem(SettingsView.THEME_KEY, theme); } catch { /* حالت خصوصی */ }
    83→      }
    84→      // به‌روزرسانی وضعیت دکمه‌ها
    85→      buttons.forEach((btn) => {
    86→        const active = btn.dataset.themePick === theme;
    87→        btn.classList.toggle('active', active);
    88→        btn.setAttribute('aria-checked', String(active));
    89→      });
    90→      if (notify) {
    91→        UI.toast(theme === 'dark' ? 'تم تاریک فعال شد.' : 'تم روشن فعال شد.', 'success');
    92→      }
    93→    };

    96→    let current = 'light';
    97→    try { current = localStorage.getItem(SettingsView.THEME_KEY) || 'light'; } catch { /* — */ }
    98→    apply(current, { persist: false });

   118→  SettingsView.saveRule = async function () {
   119→    const form = document.getElementById('rule-form');
   120→    UI.clearFormErrors(form);
   121→
   122→    const input = ruleInput();
   123→    const errMap = {
   124→      thickness: document.getElementById('err-r-thickness'),
   125→      minLength: document.getElementById('err-r-min'),
   126→      maxLength: document.getElementById('err-r-max'),
   127→      standardLength: document.getElementById('err-r-standard'),
   128→    };
   129→
   130→    const result = editingRuleId === null
   131→      ? await Rules.create(input)
   132→      : await Rules.update(editingRuleId, input);
   133→
   134→    if (!result.ok) {
   135→      // خطاهای Field-Level (§69)
   136→      if (result.errors) {
   137→        let first = null;
   138→        for (const [field, errEl] of Object.entries(errMap)) {
   139→          const msg = result.errors[field];
   140→          const inputEl = document.getElementById(`r-${field === 'minLength' ? 'min' : field === 'maxLength' ? 'max' : field === 'standardLength' ? 'standard' : 'thickness'}`);
   141→          UI.setFieldError(inputEl, errEl, msg);
   142→          if (msg && !first) first = inputEl;
   143→        }
   144→        if (first) first.focus();
   145→      }
   146→      if (result.message) UI.toast(U.escapeHtml(result.message), 'error', 6500);
   147→      return;
   148→    }
   149→
   150→    UI.toast(
   151→      editingRuleId === null
   152→        ? 'قانون متراژ استاندارد جدید ذخیره شد.'
   153→        : 'قانون ویرایش شد.',
   154→      'success'

   274→  SettingsView.restoreBackup = async function (file) {
   275→    try {
   276→      // خواندن و اعتبارسنجی ساختار (§72: Backup Corrupted)
   277→      const text = await file.text();
   278→      let payload;
   279→      try {
   280→        payload = JSON.parse(text);
   281→      } catch {
   282→        UI.toast('فایل پشتیبان معتبر نیست (JSON نامعتبر).', 'error');
   283→        return;
   284→      }
   285→
   286→      const check = Backup.validate(payload);
   287→      if (!check.ok) {
   288→        UI.toast(`فایل پشتیبان معتبر نیست: ${U.escapeHtml(check.errors[0] || '')}`, 'error', 7000);
   289→        return;
   290→      }
   291→
   292→      const c = payload.counts;
   293→
   294→      // تأیید بازیابی (§70 — عملیات خطرناک)
   295→      const ok = await UI.confirm({
   296→        title: 'تأیید بازیابی Backup',
   297→        html: `<div class="confirm-stats">
   298→          <div>ستاپ‌ها: <b>${U.faNum(c.setupRolls)}</b> · قوانین: <b>${U.faNum(c.standardLengthRules)}</b></div>
   299→          <div>رکوردهای rolls: <b>${U.faNum(c.rolls)}</b> · آرشیو: <b>${U.faNum(c.archivedRolls)}</b></div>
   300→          <div>تاریخ پشتیبان: <b>${U.faDate(payload.exportedAt)}</b></div>
   301→        </div>`,
   302→        message: 'تمام داده‌های فعلی برنامه با محتوای این فایل جایگزین خواهد شد. ادامه می‌دهید؟',
   303→        confirmText: 'بله، بازیابی کن',
   304→        danger: true,
   305→      });
   306→      if (!ok) return;
   307→
   308→      const result = await Backup.restore(payload);
   309→      if (!result.ok) {
   310→        UI.toast(`بازیابی ناموفق: ${U.escapeHtml(result.errors[0] || '')}`, 'error');
   311→        return;
   312→      }
   313→
   314→      UI.toast('بازیابی با موفقیت انجام شد — همهٔ محاسبات به‌روزرسانی گردید.', 'success');
   315→      await RM.refreshAll();
   316→      await SettingsView.renderAll();
   317→    } catch (err) {
   318→      console.error('[Restore]', err);
   319→      UI.toast('خطا در خواندن فایل پشتیبان.', 'error');
   320→    }
   321→  };

   388→  SettingsView.renderImportHistory = async function () {

   390→    const entries = await db.importMetadata.orderBy('importedAt').reverse().toArray();
   391→
   392→    if (!entries.length) {

   397→    // محدودیت نمایش: پیش‌فرض آخرین موارد — جدول بلند نشود
   398→    const shown = SettingsView._historyExpanded ? entries : entries.slice(0, HISTORY_PAGE);
   399→    const total = entries.length;
   400→    const hiddenCount = total - shown.length;
   401→
   402→    const toggleBtn = hiddenCount > 0
   403→      ? `<button type="button" class="btn btn-ghost btn-sm" data-history-toggle="expand">
   404→           نمایش همه (${U.faNum(total)})
   405→         </button>`

   407→        ? `<button type="button" class="btn btn-ghost btn-sm" data-history-toggle="collapse">
   408→             نمایش ${U.faNum(HISTORY_PAGE)} مورد آخر
   409→           </button>`
   410→        : '');

   412→    // عنوان کاربرپسنده + نام واقعی فایل برای Audit/Traceability

   414→      <div class="history-meta">
   415→        <span class="chip">${U.faNum(total)} Import ثبت‌شده</span>

   417→        ${toggleBtn}
   418→      </div>
   419→      <div class="table-wrap">
   420→        <table class="data-table"><thead><tr>
   421→          <th>نسخه</th><th>Dataset</th><th>نام فایل</th><th>وضعیت</th><th>تاریخ</th><th>رکورد</th><th>خام/یونیک</th><th>هش</th>
   422→        </tr></thead><tbody>
   423→          ${shown.map((e) => `<tr>
   424→            <td>#${U.faNum(e.datasetVersion)}</td>
   425→            <td>${U.escapeHtml(RM.config.DATASET_LABELS[e.target] || e.target)}</td>
   426→            <td><code>${U.escapeHtml(U.truncate(e.fileName, 24))}</code></td>
   427→            <td>${e.status === 'ok' ? '<span class="status-pill pill-ok">موفق</span>' : '—'}</td>
   428→            <td>${U.faDate(e.importedAt)}</td>
   429→            <td>${U.faNum(e.totalRecords)}</td>
   430→            <td>${U.faNum(e.rawRolls)}</td>
   431→            <td class="ltr" style="font-size:.68rem;color:var(--text-faint)">${U.escapeHtml(e.fileHash || '—')}</td>
   432→          </tr>`).join('')}
   433→        </tbody></table>
   434→      </div>`;
   435→
   436→    // تغییر حالت نمایش (باز/بسته)
   437→    wrap.querySelectorAll('[data-history-toggle]').forEach((btn) => {
   438→      btn.addEventListener('click', async () => {
   439→        SettingsView._historyExpanded = btn.dataset.historyToggle === 'expand';
   440→        await SettingsView.renderImportHistory();
   441→      });

   460→      const overrides = userMapping[sourceKey] || {};
   461→      const headers = await N.getDatasetHeaders(sourceKey);
   462→      const mapping = N.buildFieldMapping(sourceKey, headers, overrides);
   463→
   464→      const fieldsHtml = Object.entries(RM.config.COLUMN_MAP[sourceKey]).map(([field, spec]) => {

   468→          ? overrides[field] : '';

   496→          <h4 class="mapping-section-title">

   522→  /** جمع‌آوری Overrideهای کاربر از فرم یک Dataset */
   523→  SettingsView.collectMapping = function (sourceKey) {
   524→    const section = document.querySelector(`.mapping-section [data-map-save="${sourceKey}"]`).closest('.mapping-section');
   525→    const overrides = {};
   526→    section.querySelectorAll('[data-map-select]').forEach((sel) => {
   527→      if (sel.value.trim() !== '') overrides[sel.dataset.mapSelect] = sel.value;
   528→    });
   529→    return { overrides, section };
   530→  };

   532→  /**
   533→   * ذخیرهٔ Mapping یک Dataset:
   534→   *  ۱) اعتبارسنجی: هر ستون انتخاب‌شده باید در Dataset فعلی وجود داشته باشد
   535→   *  ۲) ذخیره در appSettings (columnMapping — مستقل برای هر Dataset)
   536→   *  ۳) اگر Dataset موجود باشد: بازسازی تراکنشی رکوردها با Mapping جدید
   537→   */
   538→  SettingsView.saveMapping = async function (sourceKey) {
   539→    const N = RM.normalize;
   540→    const { overrides, section } = SettingsView.collectMapping(sourceKey);
   541→    const headers = await N.getDatasetHeaders(sourceKey);
   542→    const normalizedHeaders = new Set(headers.map((h) => U.normalizeHeader(h)));
   543→
   544→    // پاک‌کردن خطاهای قبلی
   545→    section.querySelectorAll('[data-map-error]').forEach((el) => {
   546→      el.textContent = '';
   547→      el.hidden = true;
   548→    });
   549→    section.querySelectorAll('select.invalid').forEach((el) => el.classList.remove('invalid'));
   550→
   551→    // --- اعتبارسنجی ستون‌های انتخاب‌شده ---
   552→    let hasError = false;
   553→    let firstInvalid = null;
   554→    for (const [field, header] of Object.entries(overrides)) {
   555→      if (!normalizedHeaders.has(U.normalizeHeader(header))) {
   556→        if (headers.length) {
   557→          // Dataset موجود است اما ستون انتخاب‌شده در آن نیست
   558→          const errEl = section.querySelector(`[data-map-error="${field}"]`);
   559→          const sel = section.querySelector(`[data-map-select="${field}"]`);
   560→          const labelTxt = RM.config.FIELD_LABELS[field] || field;
   561→          if (errEl) {
   562→            errEl.textContent = `ستون «${header}» در Dataset فعلی «${RM.config.DATASET_LABELS[sourceKey]}» وجود ندارد.`;
   563→            errEl.hidden = false;
   564→          }
   565→          if (sel) sel.classList.add('invalid');
   566→          if (!firstInvalid) firstInvalid = sel;
   567→          hasError = true;
   568→        }
   569→        // اگر Dataset وارد نشده باشد، اعتبارسنجی ممکن نیست — بعد از Import بررسی می‌شود
   570→      }
   571→    }
   572→
   573→    if (hasError) {
   574→      UI.toast('ذخیرهٔ Mapping انجام نشد — ستون‌های انتخابی را بررسی کنید.', 'error', 6000);
   575→      if (firstInvalid) firstInvalid.focus();
   576→      return;
   577→    }
   578→
   579→    // --- ذخیره + بازسازی ---
   580→    try {
   581→      const userMapping = await N.getUserMapping();
   582→      const nextMapping = { ...userMapping, [sourceKey]: overrides };
   583→      const target = RM.config.IMPORT_TARGETS[sourceKey];
   584→      const table = RM.db.db[target.table];
   585→      const mapping = N.buildFieldMapping(sourceKey, headers, overrides);
   586→
   587→      let rebuiltCount = 0;
   588→      let recomputedIssues = null;
   589→
   590→      await RM.db.db.transaction('rw', table, RM.db.db.appSettings, RM.db.db.importMetadata, async () => {
   591→        // ذخیرهٔ تنظیمات
   592→        await RM.db.db.appSettings.put({ key: 'columnMapping', value: nextMapping });
   593→
   594→        // بازسازی تراکنشی رکوردهای Dataset با Mapping جدید
   595→        const existing = await table.toArray();
   596→        if (existing.length && existing.some((r) => r.raw)) {
   597→          const fresh = existing.map((r) => {
   598→            if (!r.raw) return r;   // رکورد بدون سطر خام — دست‌نخورده
   599→            const { record } = N.normalizeRow(r.raw, sourceKey, mapping);
   600→            return { ...record, id: r.id };
   601→          });
   602→          await table.clear();
   603→          await table.bulkPut(fresh);
   604→          rebuiltCount = fresh.length;
   605→
   606→          // به‌روزرسانی Issues در متادیتای آخرین Import
   607→          const issues = { width: 0, length: 0, thickness: 0, setupNumber: 0, rollNumber: 0, filmType: 0 };
   608→          for (const r of fresh) {
   609→            if (r.width === null) issues.width++;
   610→            if (r.actualLength === null) issues.length++;
   611→            if (r.thickness === null) issues.thickness++;
   612→            if (r.setupNumber === null) issues.setupNumber++;
   613→            if (r.rollNumber === '') issues.rollNumber++;
   614→            if (r.filmType === '') issues.filmType++;
   615→          }
   616→          recomputedIssues = issues;
   617→        }
   618→
   619→        // به‌روزرسانی متادیتا (Issues + Mapping Gaps تازه)
   620→        const meta = await RM.db.getLatestImportMeta(sourceKey);
   621→        if (meta) {
   622→          const patch = { issues: recomputedIssues || meta.issues, mappingGaps: mapping.__invalid || [] };
   623→          await RM.db.db.importMetadata.update(meta.id, patch);
   624→        }
   625→      });
   626→
   627→      UI.toast(
   628→        rebuiltCount
   629→          ? `Mapping ذخیره شد و ${U.faNum(rebuiltCount)} رکورد «${RM.config.DATASET_LABELS[sourceKey]}» با نگاشت جدید بازسازی شد.`
   630→          : 'Mapping ذخیره شد.' + (headers.length ? '' : ' (Dataset وارد نشده — پس از Import اعمال می‌شود)'),
   631→        'success',
   632→        6000
   633→      );
   634→
   635→      await RM.refreshAll();
   636→      await SettingsView.renderMappingPanel();
   637→    } catch (err) {
   638→      console.error('[SaveMapping]', err);
   639→      UI.toast('خطا در ذخیرهٔ Mapping — تغییری اعمال نشد.', 'error');
   640→    }
   641→  };

   643→  /** بازگردانی Mapping یک Dataset به پیش‌فرض (حذف Overrideهای کاربر) */
   644→  SettingsView.resetMapping = async function (sourceKey) {
   645→    const ok = await UI.confirm({
   646→      title: 'بازگردانی Mapping به پیش‌فرض',
   647→      message: `انتخاب‌های سفارشی شما برای «${RM.config.DATASET_LABELS[sourceKey]}» حذف و نگاشت پیش‌فرض اعمال می‌شود. رکوردهای Dataset با نگاشت پیش‌فرض بازسازی خواهند شد. ادامه می‌دهید؟`,
   648→      confirmText: 'بله، بازگردانی کن',
   649→    });
   650→    if (!ok) return;
   651→
   652→    const userMapping = await RM.normalize.getUserMapping();
   653→    const nextMapping = { ...userMapping };
   654→    delete nextMapping[sourceKey];
   655→
   656→    try {
   657→      const target = RM.config.IMPORT_TARGETS[sourceKey];
   658→      const table = RM.db.db[target.table];
   659→      const headers = await RM.normalize.getDatasetHeaders(sourceKey);
   660→      const mapping = RM.normalize.buildFieldMapping(sourceKey, headers, {});
   661→
   662→      await RM.db.db.transaction('rw', table, RM.db.db.appSettings, RM.db.db.importMetadata, async () => {
   663→        await RM.db.db.appSettings.put({ key: 'columnMapping', value: nextMapping });
   664→
   665→        const existing = await table.toArray();
   666→        if (existing.length && existing.some((r) => r.raw)) {
   667→          const fresh = existing.map((r) => {
   668→            if (!r.raw) return r;
   669→            const { record } = RM.normalize.normalizeRow(r.raw, sourceKey, mapping);
   670→            return { ...record, id: r.id };
   671→          });
   672→          await table.clear();
   673→          await table.bulkPut(fresh);
   674→        }
   675→
   676→        const meta = await RM.db.getLatestImportMeta(sourceKey);
   677→        if (meta) {
   678→          await RM.db.db.importMetadata.update(meta.id, { mappingGaps: [] });
   679→        }
   680→      });
   681→
   682→      UI.toast('Mapping به پیش‌فرض بازگردانی شد.', 'success');
   683→      await RM.refreshAll();
   684→      await SettingsView.renderMappingPanel();
   685→    } catch (err) {
   686→      console.error('[ResetMapping]', err);
   687→      UI.toast('خطا در بازگردانی Mapping.', 'error');
   688→    }
   689→  };
