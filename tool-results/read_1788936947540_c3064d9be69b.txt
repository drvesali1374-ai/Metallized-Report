    56→  /** ساخت/بازیابی نمونهٔ چارت با شناسهٔ المان */
    57→  Dashboard.chartOf = function (elId) {
    58→    const el = document.getElementById(elId);
    59→    if (!el || !Dashboard.hasECharts()) return null;
    60→    if (Dashboard._charts[elId]) return Dashboard._charts[elId];
    61→    if (!el.clientWidth) return null;   // مخفی است — هنگام نمایش ساخته می‌شود
    62→    const inst = window.echarts.init(el);
    63→    Dashboard._charts[elId] = inst;
    64→    return inst;
    65→  };

    67→  /** اعمال گزینه‌ها روی چارت (بدون Merge — رندر تازه) */
    68→  Dashboard.applyChart = function (elId, option) {
    69→    const inst = Dashboard.chartOf(elId);
    70→    if (!inst) { Dashboard._pendingCharts = true; return false; }
    71→    inst.setOption(option, { notMerge: true });
    72→    return true;
    73→  };

   211→  /* ---------- چارت مانده زمان متالایز به تفکیک دستگاه (نسخهٔ ۲٫۳ — ECharts) ---------- */
   212→
   213→  /**
   214→   * نمودار میله‌ای افقی گروهی ECharts — هر دستگاه (۱/۲/۳) دو میله:
   215→   *   «مانده زمان ستاپ» (فیروزه‌ای) و «مانده زمان موجودی» (کهربایی).
   216→   * نسخهٔ ۲٫۵ — راست‌به‌چپ: نام دستگاه‌ها در سمت راست (yAxis position:right)
   217→   * و میله‌ها از راست (کنار نام‌ها) به چپ رشد می‌کنند (xAxis inverse).
   218→   * Tooltip فارسی با مدت دقیق + برچسب ساعت روی محور. Fallback: میله‌های CSS.
   219→   * زیر نمودار: آمار دوخطی (کمترین دستگاه + میانگین) — نسخهٔ ۲٫۴.
   220→   */
   221→  Dashboard.renderMachineTimeChart = function (state) {
   222→    const s = state.summary;
   223→    const machines = RM.config.VALID_MACHINES;
   224→
   225→    // بیشینه برای مقیاس
   226→    let maxVal = 0;
   227→    for (const m of machines) {
   228→      const t = s.machineTime[m];
   229→      if (t) maxVal = Math.max(maxVal, t.setup, t.produced);
   230→    }
   231→
   232→    const chipsEl = document.getElementById('machine-time-chips');
   233→    chipsEl.innerHTML = `
   234→      <span class="chip">قوانین زمان: ${U.faNum(s.timeRulesCount)}</span>
   235→      <span class="chip chip-strong">جمع مانده زمان ستاپ: ${U.faHMM(s.totalRemainingSetupTime)}</span>
   236→      <span class="chip chip-ok">جمع مانده زمان موجودی: ${U.faHMM(s.totalRemainingProducedTime)}</span>
   237→      ${s.timeRuleCoverage.uncovered > 0
   238→        ? `<span class="chip chip-warn">${U.faNum(s.timeRuleCoverage.uncovered)} رکورد بدون قانون</span>`
   239→        : ''}
   240→    `;
   241→
   242→    /* ---------- آمار دوخطی زیر نمودار (نسخهٔ ۲٫۴) ----------
   243→       خط ۱: دستگاهی که کمترین مانده زمان (موجودی) را دارد + هر دو زمان آن
   244→       خط ۲: میانگین دستگاه‌ها (جمع ÷ تعداد دستگاه‌ها) + هر دو زمان
   245→       همهٔ زمان‌ها با فرمت [H]:mm + معادل روزانه (روز ۲۴ ساعته) */
   246→    const producedOf = (m) => (s.machineTime[m] ? s.machineTime[m].produced : 0);
   247→    const setupOf = (m) => (s.machineTime[m] ? s.machineTime[m].setup : 0);
   248→    let minMachine = machines[0];
   249→    let minProduced = Infinity;
   250→    for (const m of machines) {
   251→      if (producedOf(m) < minProduced) { minProduced = producedOf(m); minMachine = m; }
   252→    }
   253→    const machineCount = machines.length;
   254→    const avgProduced = s.totalRemainingProducedTime / machineCount;
   255→    const avgSetup = s.totalRemainingSetupTime / machineCount;
   256→
   257→    const statPair = (producedMin, setupMin) => `
   258→      <span class="mt-stat-pair">مانده زمان موجودی: <b>${U.faHMM(producedMin)}</b>
   259→        <span class="mt-days">· معادل روزانه: ${U.faDaysOf(producedMin)}</span></span>
   260→      <span class="mt-stat-arrow" aria-hidden="true">←</span>
   261→      <span class="mt-stat-pair">مانده زمان ستاپ: <b>${U.faHMM(setupMin)}</b>
   262→        <span class="mt-days">· معادل روزانه: ${U.faDaysOf(setupMin)}</span></span>`;
   263→
   264→    document.getElementById('machine-time-footer').innerHTML = `
   265→      <div class="mt-stats">
   266→        <div class="mt-stat-line">
   267→          <span class="mt-stat-who">دستگاه دارای کمترین مانده زمان (دستگاه ${U.faNum(minMachine)})</span>
   268→          <span class="mt-stat-arrow" aria-hidden="true">←</span>
   269→          ${statPair(producedOf(minMachine), setupOf(minMachine))}
   270→        </div>
   271→        <div class="mt-stat-line">
   272→          <span class="mt-stat-who">میانگین دستگاه‌ها (${U.faNum(machineCount)} دستگاه)</span>
   273→          <span class="mt-stat-arrow" aria-hidden="true">←</span>
   274→          ${statPair(avgProduced, avgSetup)}
   275→        </div>
   276→      </div>`;
   277→
   278→    if (!Dashboard.hasECharts()) {
   279→      Dashboard.renderMachineTimeChartFallback(state);   // کتابخانه در دسترس نیست
   280→      return;
   281→    }
   282→
   283→    const P = Dashboard.chartPalette();
   284→    const cats = machines.map((m) => `دستگاه ${U.faNumPlain(m)}`).reverse();   // دستگاه ۱ بالا
   285→    const setupData = machines.map((m) => {
   286→      const t = s.machineTime[m];
   287→      return t && t.setup ? t.setup : 0;
   288→    }).reverse();
   289→    const producedData = machines.map((m) => {
   290→      const t = s.machineTime[m];
   291→      return t && t.produced ? t.produced : 0;
   292→    }).reverse();
   293→    const recordsByMachine = machines.map((m) => {
   294→      const t = s.machineTime[m];
   295→      return t ? t.records : 0;
   296→    }).reverse();
   297→
   298→    const option = {
   299→      animationDuration: 550,
   300→      animationEasing: 'cubicOut',
   301→      textStyle: { fontFamily: P.fontFamily },
   302→      // نسخهٔ ۲٫۵: راست‌به‌چپ — میله‌ها از راست (کنار نام دستگاه‌ها) به چپ رشد می‌کنند
   303→      grid: { left: 56, right: 8, top: 42, bottom: 8, containLabel: true },
   304→      legend: {
   305→        top: 4,
   306→        right: 8,
   307→        itemWidth: 14,
   308→        itemHeight: 9,
   309→        itemGap: 18,
   310→        textStyle: { color: P.muted, fontFamily: P.fontFamily, fontSize: 11 },
   311→        data: ['مانده زمان ستاپ', 'مانده زمان موجودی'],
   312→      },
   313→      tooltip: {
   314→        trigger: 'axis',
   315→        axisPointer: { type: 'shadow', shadowStyle: { color: P.splitLine } },
   316→        backgroundColor: P.tooltipBg,
   317→        borderColor: P.tooltipBorder,
   318→        borderWidth: 1,
   319→        padding: [8, 12],
   320→        textStyle: { color: P.text, fontFamily: P.fontFamily, fontSize: 12 },
   321→        formatter: (params) => {
   322→          const idx = params[0] ? params[0].dataIndex : 0;
   323→          const m = RM.config.VALID_MACHINES[machines.length - 1 - idx] || '—';
   324→          const lines = [`<b>دستگاه ${U.faNumPlain(m)}</b> · ${U.faNum(recordsByMachine[idx] || 0)} رکورد`];
   325→          for (const p of params) {
   326→            lines.push(`${p.marker} ${p.seriesName}: <b>${U.faDuration(p.value)}</b>${p.value ? ` (~${U.faNum(Math.round(p.value / 60))} ساعت)` : ''}`);
   327→          }
   328→          return lines.join('<br>');
   329→        },
   330→      },
   331→      xAxis: {
   332→        type: 'value',
   333→        inverse: true,                       // نسخهٔ ۲٫۵: صفر در راست → رشد به چپ
   334→        max: (v) => Math.max(v.max * 1.08, 60),
   335→        axisLabel: {
   336→          color: P.muted,
   337→          fontFamily: P.fontFamily,
   338→          fontSize: 10.5,
   339→          formatter: (v) => `${U.faNumPlain(Math.round(v / 60))} ساعت`,
   340→        },
   341→        splitLine: { lineStyle: { color: P.splitLine, width: 1 } },
   342→        axisLine: { show: false },
   343→        axisTick: { show: false },
   344→      },
   345→      yAxis: {
   346→        type: 'category',
   347→        position: 'right',                  // نسخهٔ ۲٫۵: نام دستگاه‌ها در سمت راست
   348→        data: cats,
   349→        axisLabel: { color: P.text, fontFamily: P.fontFamily, fontSize: 12, fontWeight: 600 },
   350→        axisLine: { lineStyle: { color: P.axisLine } },
   351→        axisTick: { show: false },
   352→      },
   353→      series: [
   354→        {
   355→          name: 'مانده زمان ستاپ',
   356→          type: 'bar',
   357→          data: setupData,
   358→          barMaxWidth: 18,
   359→          barGap: '28%',
   360→          itemStyle: { color: P.setup, borderRadius: [4, 0, 0, 4] },
   361→          emphasis: { itemStyle: { color: P.setup, opacity: 0.82 } },
   362→          label: {
   363→            show: true,
   364→            position: 'left',              // نسخهٔ ۲٫۵: نوک میله (سمت چپ — رشد راست‌به‌چپ)
   365→            color: P.muted,
   366→            fontFamily: P.fontFamily,

   783→          <div class="table-wrap table-scroll" style="max-height:420px">
   784→            <table class="data-table">
   785→              <thead><tr><th>ستاپ</th><th>دستگاه</th><th>عرض</th><th>ضخامت</th><th>متراژ استاندارد</th><th>تعداد</th>${timeCols}</tr></thead>
   786→              <tbody>${rows.map((r) => `
   787→                <tr>
   788→                  <td>${U.faNum(r.setupNumber)}</td>
   789→                  <td>${U.faNum(r.machineNumber)}</td>
   790→                  <td>${U.faWidth(r.width)}</td>
   791→                  <td>${r.thickness === null ? '<span class="muted">ناقص</span>' : U.faNum(r.thickness)}</td>
   792→                  <td>${r.standardLength === null ? '<span class="muted">ناقص</span>' : U.faNum(r.standardLength)}</td>
   793→                  <td class="count-cell">${U.faNum(r.rollCount)}</td>
   794→                  ${d.filter === 'no-time-rule'
   795→                    ? `<td><span class="muted">بدون قانون زمان</span></td>`
   796→                    : `<td>${U.faNum(r.rawExisting)}</td>
   797→                       <td>${U.faNum(r.metallized)}</td>
   798→                       <td>${r.conflict > 0 ? `<span class="pill-bad status-pill">⚠ ${U.faNum(r.conflict)}</span>` : '—'}</td>`}
   799→                </tr>`).join('')}
   800→              </tbody>
   801→            </table>
   802→          </div>
   803→        `);
   804→        return;
   805→      }
   806→
   807→      /* --- هشدار هم‌پوشانی قوانین --- */
   808→      if (d.source === 'rules') {
   809→        UI.infoModal(warning.title, `
   810→          <div class="drill-subtitle"><b>${U.faNum(d.pairs.length)}</b> جفت قانون هم‌پوشان</div>
   811→          <div class="table-wrap">
   812→            <table class="data-table">
   813→              <thead><tr><th>ضخامت</th><th>بازهٔ قانون ۱</th><th>بازهٔ قانون ۲</th></tr></thead>
   814→              <tbody>${d.pairs.map(([a, b]) => `
   815→                <tr>
   816→                  <td class="thickness-cell">${U.faNum(a.thickness)}</td>
   817→                  <td>${U.faNum(a.minLength)} تا ${U.faNum(a.maxLength)} → ${U.faNum(a.standardLength)}</td>
   818→                  <td>${U.faNum(b.minLength)} تا ${U.faNum(b.maxLength)} → ${U.faNum(b.standardLength)}</td>
   819→                </tr>`).join('')}
   820→              </tbody>
   821→            </table>
   822→          </div>
   823→        `);
   824→        return;
   825→      }
   826→
   827→      /* --- هشدارهای رکوردی (rolls / archived) → مودال مشترک --- */
   828→      const sourceKey = d.source;
   829→      const table = RM.config.IMPORT_TARGETS[sourceKey].table;
   830→      let records = [];
   831→
   832→      if (d.issue) {
   833→        // نقص دادهٔ مستقیم: بدون عرض / بدون متراژ / بدون ضخامت / شماره رول / نوع فیلم
   834→        // رزولور مشترک نقص‌ها («length» → actualLength در مدل داخلی)
   835→        const all = await db[table].toArray();
   836→        records = RM.normalize.recordsWithIssue(all, d.issue);
   837→      } else if (d.filter === 'all') {
   838→        // Mapping ناقص — همهٔ رکوردهای همان Dataset (فیلدهای مفقود «—» نمایش داده می‌شوند)
   839→        records = await db[table].toArray();
   840→      } else if (d.filter === 'no-rule') {
   841→        // رول‌های خام بدون قانون متراژ — از گروه‌های بدون قانون (کلید‌محور)
   842→        const state = Dashboard._state;
   843→        const groups = state.rawRollGroups.filter((g) => !g.hasRule);
   844→        const ids = groups.flatMap((g) => g.rollIds);
   845→        records = (await db.rolls.bulkGet(ids)).filter(Boolean);
   846→      } else if (d.filter === 'group') {
   847→        // گروه‌های خاص (بدون ستاپ / مازاد خام)
   848→        const state = Dashboard._state;
   849→        const keySet = new Set(d.groupKeys || []);
   850→        const groups = state.rawRollGroups.filter((g) => keySet.has(g.key));
   851→        const ids = groups.flatMap((g) => g.rollIds);
   852→        records = (await db.rolls.bulkGet(ids)).filter(Boolean);
   853→      } else if (d.filter === 'metallized-unallocated') {
   854→        // مازاد متالایز — یونیک‌های دارای کلیدِ تخصیص‌نیافته
   855→        const state = Dashboard._state;
   856→        const keySet = new Set(d.groupKeys || []);
   857→        records = state.metallizedUniqueRecords.filter((r) =>
   858→          keySet.has(U.formatKey(r.width, r.setupNumber))
   859→        );
   860→      }
   861→
   862→      await UI.recordsDrillModal(warning.title, records, { source: sourceKey });
   863→    } catch (err) {
   864→      console.error('[Drill-down]', err);
   865→      UI.toast('نمایش رکوردهای این هشدار با خطا مواجه شد.', 'error');
   866→    }
   867→  };

   884→        : '<span class="muted">فایلی بارگذاری نشده</span>';
