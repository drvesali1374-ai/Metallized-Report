    98→  SetupView.applyThicknessInheritance = async function () {
    99→    const { setupNumber, thickness, standardLength } = F();
   100→    const value = U.parseInt(setupNumber.value);
   101→    const hintEl = document.getElementById('hint-thickness');
   102→
   103→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   104→
   105→    if (value === null || value < 0) {
   106→      SetupView.thicknessLocked = false;
   107→      thickness.disabled = false;
   108→      hintEl.hidden = true;
   109→      return;
   110→    }
   111→
   112→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   113→    const withThickness = existing.find((s) => s.thickness !== null);
   114→
   115→    if (withThickness) {
   116→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   117→      SetupView.thicknessLocked = true;
   118→      thickness.value = String(withThickness.thickness);
   119→      thickness.disabled = true;
   120→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   121→      hintEl.hidden = false;
   122→    } else {
   123→      SetupView.thicknessLocked = false;
   124→      thickness.disabled = false;
   125→      hintEl.hidden = true;
   126→    }
   127→
   128→    SetupView.refreshStandardLengthOptions(standardLength.value);
   129→  };

   131→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   132→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   133→    const { thickness, standardLength } = F();
   134→    const thicknessValue = U.parseNumber(thickness.value);
   135→    const errEl = document.getElementById('err-standard');
   136→
   137→    const options = thicknessValue === null
   138→      ? []
   139→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   140→
   141→    let html = '<option value="">— انتخاب کنید —</option>';
   142→    for (const opt of options) {
   143→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   144→    }
   145→    standardLength.innerHTML = html;
   146→
   147→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   148→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   149→      standardLength.value = keepValue;
   150→    } else {
   151→      standardLength.value = '';
   152→    }
   153→
   154→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   155→    if (thicknessValue !== null && !options.length) {
   156→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   157→      errEl.hidden = false;
   158→    } else {
   159→      errEl.textContent = '';
   160→      errEl.hidden = true;
   161→    }

   228→          confirmText: 'بله، ضخامت همه را تغییر بده',

   231→          UI.toast('عملیات ویرایش لغو شد.', 'info');

   237→      if (!result.ok) {
   238→        SetupView.showValidationErrors(result);
   239→        return;
   240→      }
   241→      UI.toast(
   242→        (info.affected > 0
   243→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   244→          : 'رکورد با موفقیت ویرایش شد.'),
   245→        'success'
   246→      );
   247→    } else {
   248→      /* --- ثبت جدید --- */
   249→      const result = await Setups.create(input);
   250→      if (!result.ok) {
   251→        SetupView.showValidationErrors(result);
   252→        return;
   253→      }
   254→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   255→    }
   256→
   257→    SetupView.resetForm();
   258→    await RM.refreshAll();
   259→    await SetupView.render();
   260→  };

   281→    }
   282→
   283→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   284→    if (firstInvalid) firstInvalid.focus();

   402→  /**
   403→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   404→   * بلافاصله به‌روز می‌شود:
   405→   *   تعداد رکوردها، کل عرض‌های مادر، کل رول‌های الگوی برش،
   406→   *   جمع عدد ست‌ها، رول‌های یونیک مادر، رول‌های یونیک الگو.
   407→   */
   408→  SetupView.bulkSummaryStats = function (drafts) {
   409→    let mothers = 0, pieces = 0, sets = 0;
   410→    const motherSet = new Set();
   411→    const patternSet = new Set();
   412→
   413→    for (const d of drafts) {
   414→      const occ = d.occurrences || 1;
   415→      mothers += occ;                                        // هر occurrence یک مادر است
   416→      const parsed = Setups.parsePattern(d.patternText);
   417→      if (parsed.ok) {
   418→        pieces += parsed.parts.length * occ;                 // هر قطعهٔ الگو یک رول برش است
   419→        if (parsed.parts.length) patternSet.add(Setups.normalizePatternKey(d.patternText));
   420→      }
   421→      if (d.rollCount !== null) sets += d.rollCount;         // عدد ست‌ها (انتهای ردیف‌های فرد)
   422→      const w = U.parseNumber(d.motherToken);
   423→      if (w !== null) motherSet.add(w);
   424→    }
   425→
   426→    return {
   427→      count: drafts.length, mothers, pieces, sets,
   428→      uniqueMothers: motherSet.size, uniquePatterns: patternSet.size,
   429→    };
   430→  };

   600→  SetupView.onBulkRowInput = function (el) {
   601→    const b = SetupView._bulk;
   602→    if (!b) return;
   603→    const i = Number(el.dataset.bulkRow);
   604→    const draft = b.drafts[i];
   605→    if (!draft) return;
   606→
   607→    const field = el.dataset.bulkField;
   608→    if (field === 'rollCount') {
   609→      const v = U.parseInt(el.value);
   610→      draft.rollCount = v === null ? null : v;
   611→    } else {
   612→      draft[field] = el.value;
   613→    }
   614→
   615→    SetupView.renderBulkSummary();          // زنده — بلافاصله
   616→    SetupView.updateBulkRowComputed(i);
   617→    SetupView.scheduleBulkRevalidate();
   618→  };

   620→  /** سلول‌های محاسباتی ردیف‌ها (مجموع الگو / پرت عرض) */
   621→  SetupView.updateBulkRowComputed = function (rowIdx = null) {
   622→    const b = SetupView._bulk;
   623→    if (!b) return;
   624→    const rowsIdx = rowIdx === null ? b.drafts.map((_, i) => i) : [rowIdx];
   625→
   626→    for (const i of rowsIdx) {
   627→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   628→      if (!tr) continue;
   629→      const d = b.drafts[i];
   630→
   631→      const parsed = Setups.parsePattern(d.patternText);
   632→      const mother = U.parseNumber(d.motherToken);
   633→      const sum = parsed.ok ? parsed.sum : null;
   634→      const trim = parsed.ok && mother !== null ? mother - sum : null;
   635→
   636→      tr.querySelector('[data-bulk-cell="sum"]').innerHTML =
   637→        sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>';
   638→      tr.querySelector('[data-bulk-cell="trim"]').innerHTML =
   639→        trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>';
   640→    }
   641→  };

   643→  /**
   644→   * بازسازی گزینه‌های «متراژ استاندارد» با تغییر ضخامت سراسری —
   645→   * مقدار رکوردها اگر در قوانین ضخامت جدید معتبر باشد حفظ می‌شود.
   646→   */
   647→  SetupView.rebuildBulkStandardOptions = function () {
   648→    const b = SetupView._bulk;
   649→    if (!b) return;
   650→    const options = SetupView.bulkStandardOptions();
   651→    const optionsStr = options.map(String);
   652→
   653→    for (let i = 0; i < b.drafts.length; i++) {
   654→      const sel = document.querySelector(`[data-bulk-field="standardLength"][data-bulk-row="${i}"]`);
   655→      if (!sel) continue;
   656→      const keep = optionsStr.includes(String(b.drafts[i].standardLength)) ? b.drafts[i].standardLength : '';
   657→      b.drafts[i].standardLength = keep;
   658→      sel.innerHTML = '<option value="">—</option>' +
   659→        options.map((o) => `<option value="${o}" ${String(o) === String(keep) ? 'selected' : ''}>${U.faNum(o)}</option>`).join('');
   660→    }
   661→
   662→    const g = document.getElementById('bulk-g-standard');
   663→    if (g) {
   664→      g.innerHTML = '<option value="">— بدون تغییر —</option>' +
   665→        options.map((o) => `<option value="${o}">${U.faNum(o)}</option>`).join('');
   666→    }
   667→  };

   669→  /** اعتبارسنجی زنده با debounce (پس از توقف تایپ) */
   670→  SetupView.scheduleBulkRevalidate = function () {

   675→  /** اعتبارسنجی زندهٔ پیش‌نمایش + به‌روزرسانی وضعیت ردیف‌ها و دکمهٔ Commit */
   676→  SetupView.revalidateBulk = async function () {
   677→    const b = SetupView._bulk;
   678→    if (!b) return;
   679→    const check = await Setups.validateBulkDraft(b.common, b.drafts);
   680→    SetupView.applyBulkCheck(check);
   681→  };

   683→  /** اعمال نتیجهٔ اعتبارسنجی روی UI پیش‌نمایش (خطاهای سراسری + وضعیت ردیف‌ها) */
   684→  SetupView.applyBulkCheck = function (check) {
   685→    SetupView._bulkCheck = check;
   686→    const b = SetupView._bulk;
   687→    if (!b) return;
   688→
   689→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   690→    const globalMap = {
   691→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   692→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   693→    };
   694→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {
   695→      const errEl = document.getElementById(errId);
   696→      const inputEl = document.getElementById(inputId);
   697→      const msg = check.globalErrors[field] || '';
   698→      if (errEl) { errEl.textContent = msg; errEl.hidden = !msg; }
   699→      if (inputEl) inputEl.classList.toggle('invalid', !!msg);
   700→    }
   701→
   702→    // وضعیت هر ردیف
   703→    b.drafts.forEach((d, i) => {
   704→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   705→      if (!tr) return;
   706→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   707→      const rec = check.records[i];
   708→      const errs = rec ? rec.errors : [];
   709→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   710→      cell.innerHTML = errs.length
   711→        ? `<span class="status-pill pill-bad">✕ خطا</span>
   712→           <ul class="bulk-errors">${errs.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`
   713→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   714→    });
   715→
   716→    const commitBtn = document.getElementById('btn-bulk-commit');
   717→    const hasStructure = b.structureErrors.length > 0;
   718→    commitBtn.disabled = !(check.ok === true && !hasStructure);
   719→  };

   721→  /** Commit نهایی — یک تراکنش واحد از وضعیت ویرایش‌شده؛ در صورت خطا هیچ رکوردی ثبت نمی‌شود */
   722→  SetupView.commitBulk = async function () {
   723→    const b = SetupView._bulk;

   731→      return;
   732→    }
   733→
   734→    UI.toast(
   735→      `${U.faNum(result.count)} رکورد ستاپ در یک تراکنش واحد ثبت شد (Bulk Paste).`,

   737→      5000

   798→  SetupView.renderPatternCards = function (setup) {

   885→  /** اعمال مرتب‌سازی چندمرحله‌ای روی ردیف‌های گزارش (از آخرین اولویت به اولین) */
   886→  SetupView.applySetupSort = function (rows) {
   887→    const sorted = [...rows];
   888→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
   889→      const { field, direction } = SetupView.sortPipeline[i];
   890→      sorted.sort((a, b) => {
   891→        if (field === 'cuttingPattern') {
   892→          const ka = Setups.normalizePatternKey(a.cuttingPattern) || '';
   893→          const kb = Setups.normalizePatternKey(b.cuttingPattern) || '';
   894→          const r = ka.localeCompare(kb, 'fa', { numeric: true });
   895→          return direction === 'desc' ? -r : r;
   896→        }
   897→        return U.compareNumeric(a[field], b[field], direction);
   898→      });
   899→    }
   900→    return sorted;
   901→  };

   919→  /** تطبیق یک ردیف با مقدار فیلتر یک ستون: عدد = برابری عددی · متن = شامل بودن */
   920→  SetupView.rowMatchesFilter = function (row, key, value) {
   921→    const raw = row[key];
   922→    const num = U.parseNumber(value);
   923→
   924→    if (num !== null && raw !== null && raw !== undefined && Number.isFinite(Number(raw))) {
   925→      return Number(raw) === num;
   926→    }
   927→    const hay = (key === 'cuttingPattern')
   928→      ? ((Setups.normalizePatternKey(raw) || '') + ' ' + String(raw ?? ''))
   929→      : String(raw ?? '');
   930→    return hay.includes(value);
   931→  };
