   160→  /** ارث‌بری ضخامت (§22): اگر ستاپ موجود باشد → قفل خودکار */
   161→  SetupView.applyThicknessInheritance = async function () {
   162→    const { setupNumber, thickness, standardLength } = F();
   163→    const value = U.parseInt(setupNumber.value);
   164→    const hintEl = document.getElementById('hint-thickness');
   165→
   166→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   167→
   168→    if (value === null || value < 0) {
   169→      SetupView.thicknessLocked = false;
   170→      thickness.disabled = false;
   171→      hintEl.hidden = true;
   172→      return;
   173→    }
   174→
   175→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   176→    const withThickness = existing.find((s) => s.thickness !== null);
   177→
   178→    if (withThickness) {
   179→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   180→      SetupView.thicknessLocked = true;
   181→      thickness.value = String(withThickness.thickness);
   182→      thickness.disabled = true;
   183→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   184→      hintEl.hidden = false;
   185→    } else {
   186→      SetupView.thicknessLocked = false;
   187→      thickness.disabled = false;
   188→      hintEl.hidden = true;
   189→    }
   190→
   191→    SetupView.refreshStandardLengthOptions(standardLength.value);
   192→  };

   194→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   195→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   196→    const { thickness, standardLength } = F();
   197→    const thicknessValue = U.parseNumber(thickness.value);
   198→    const errEl = document.getElementById('err-standard');
   199→
   200→    const options = thicknessValue === null
   201→      ? []
   202→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   203→
   204→    let html = '<option value="">— انتخاب کنید —</option>';
   205→    for (const opt of options) {
   206→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   207→    }
   208→    standardLength.innerHTML = html;
   209→
   210→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   211→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   212→      standardLength.value = keepValue;
   213→    } else {
   214→      standardLength.value = '';
   215→    }
   216→
   217→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   218→    if (thicknessValue !== null && !options.length) {
   219→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   220→      errEl.hidden = false;
   221→    } else {
   222→      errEl.textContent = '';
   223→      errEl.hidden = true;
   224→    }
   225→  };

   228→  SetupView.updatePatternFeedback = function () {
   229→    const { width, cuttingPattern } = F();
   230→    const box = document.getElementById('pattern-feedback');
   231→
   232→    const result = Setups.validatePattern(cuttingPattern.value, U.parseNumber(width.value));
   233→
   234→    if (!cuttingPattern.value.trim()) {
   235→      box.className = 'pattern-feedback';
   236→      box.innerHTML = '<span class="muted">اختیاری — مثال: 1100-1170</span>';
   237→      return;
   238→    }
   239→
   240→    if (!result.ok) {
   241→      box.className = 'pattern-feedback invalid';
   242→      box.innerHTML = `<span class="pf-status">✕</span> ${result.message}`;
   243→      return;
   244→    }
   245→
   246→    const motherWidth = U.parseNumber(width.value);
   247→    const remaining = motherWidth !== null ? motherWidth - result.sum : null;
   248→
   249→    box.className = 'pattern-feedback valid';
   250→    box.innerHTML =
   251→      `<span class="pf-status">✓</span> مجموع: <b>${U.faWidth(result.sum)}</b>` +
   252→      (remaining !== null && remaining >= 0
   253→        ? ` · باقی‌مانده (پرت عرض): <b>${U.faWidth(remaining)}</b>`
   254→        : '') +
   255→      ` · بخش‌ها: ${U.faNum(result.parts.length)}`;
   256→  };

   262→  SetupView.submit = async function () {
   263→    const form = document.getElementById('setup-form');
   264→    const f = F();
   265→    UI.clearFormErrors(form);
   266→
   267→    const input = {
   268→      setupNumber: f.setupNumber.value,
   269→      machineNumber: f.machineNumber.value,
   270→      width: f.width.value,
   271→      thickness: f.thickness.value,
   272→      standardLength: f.standardLength.value,
   273→      rollCount: f.rollCount.value,
   274→      cuttingPattern: f.cuttingPattern.value,
   275→    };
   276→
   277→    /* --- ویرایش: بررسی آبشار ضخامت (§23) --- */
   278→    if (SetupView.editingId !== null) {
   279→      const newThickness = U.parseNumber(input.thickness);
   280→      const info = await Setups.cascadeInfo(SetupView.editingId, newThickness);
   281→
   282→      if (info.affected > 0) {
   283→        const ok = await UI.confirm({
   284→          title: 'تأیید تغییر ضخامت',
   285→          html: `
   286→            <div class="confirm-stats">
   287→              <div>تغییر ضخامت ستاپ <b>${U.faNum(info.setupNumber)}</b> باعث تغییر ضخامت
   288→              <b>${U.faNum(info.affected)}</b> رکورد دیگر نیز خواهد شد.</div>
   289→            </div>`,
   290→          message: 'این عملیات تراکنشی است و در صورت خطا به حالت قبل برمی‌گردد. ادامه می‌دهید؟',
   291→          confirmText: 'بله، ضخامت همه را تغییر بده',
   292→        });
   293→        if (!ok) {
   294→          UI.toast('عملیات ویرایش لغو شد.', 'info');
   295→          return;
   296→        }
   297→      }
   298→
   299→      const result = await Setups.update(SetupView.editingId, input, { cascadedCount: info.affected });
   300→      if (!result.ok) {
   301→        SetupView.showValidationErrors(result);
   302→        return;
   303→      }
   304→      UI.toast(
   305→        (info.affected > 0
   306→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   307→          : 'رکورد با موفقیت ویرایش شد.'),
   308→        'success'
   309→      );
   310→    } else {
   311→      /* --- ثبت جدید --- */
   312→      const result = await Setups.create(input);
   313→      if (!result.ok) {
   314→        SetupView.showValidationErrors(result);
   315→        return;
   316→      }
   317→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   318→    }
   319→
   320→    SetupView.resetForm();
   321→    await RM.refreshAll();
   322→    await SetupView.render();
   323→  };

   325→  /** نمایش خطاهای Field-Level (§69): حاشیهٔ قرمز + پیام + فوکوس */
   326→  SetupView.showValidationErrors = function (result) {
   327→    const form = document.getElementById('setup-form');
   328→    const f = F();
   329→    const fieldMap = {
   330→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   331→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   332→      width: [f.width, document.getElementById('err-width')],
   333→      thickness: [f.thickness, document.getElementById('err-thickness')],
   334→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   335→      rollCount: [f.rollCount, document.getElementById('err-count')],
   336→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   337→    };
   338→
   339→    let firstInvalid = null;
   340→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   341→      const msg = result.errors && result.errors[field];
   342→      UI.setFieldError(input, errEl, msg);
   343→      if (msg && !firstInvalid) firstInvalid = input;
   344→    }
   345→
   346→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   347→    if (firstInvalid) firstInvalid.focus();
   348→  };

   469→  /**
   470→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   471→   * بلافاصله به‌روز می‌شود (روش محاسبه + تغییر ۱ — تعداد ست):
   472→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   473→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   474→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست × تعداد)
   475→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   476→   * تعداد ست هر رکورد = متراژ استاندارد ÷ کوچکترین متراژ استاندارد ضخامت
   477→   * سراسری (رول ۱ ستی) — اگر متراژ انتخاب نشده باشد، ۱ ست فرض می‌شود.
   478→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه ۱-ستی):
   479→   *   ۱۶ / ۳ / ۴۸ / ۵
   480→   */
   481→  SetupView.bulkSummaryStats = function (drafts, rules = [], thickness = null) {
   482→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   483→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست × تعداد) — «جمع عرض الگوی برش»
   484→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   485→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   486→
   487→    for (const d of drafts) {
   488→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   489→      motherRollsTotal += rollCount;
   490→
   491→      const w = U.parseNumber(d.motherToken);
   492→      if (w !== null) motherWidthSet.add(w);
   493→
   494→      const parsed = Setups.parsePattern(d.patternText);
   495→      if (parsed.ok && parsed.parts.length) {
   496→        /* --- تغییر ۱: تعداد فرزند = قطعه‌های الگو × تعداد ست × تعداد مادر --- */
   497→        const std = U.parseNumber(d.standardLength);
   498→        const sc = RM.rulesEngine.setCount(rules, thickness, std);
   499→        const setCount = sc !== null && sc > 0 ? sc : 1;
   500→        patternWidthsTotal += parsed.parts.length * setCount * rollCount;
   501→        for (const part of parsed.parts) patternWidthSet.add(part);
   502→      }
   503→    }
   504→
   505→    return {
   506→      count: drafts.length,
   507→      motherRollsTotal,                            // ۱۶
   508→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   509→      patternWidthsTotal,                          // ۴۸ (با تعداد ست ضرب می‌شود)
   510→      uniquePatternWidths: patternWidthSet.size,   // ۵
   511→    };
   512→  };

   734→  /** ادغام ردیف تکراری با همتای معتبرش (تغییر ۲) — تعداد رول ردیف تکراری
   735→      به تعداد رول همتای معتبر اضافه می‌شود، ردیف تکراری حذف می‌شود و
   736→      خلاصه/اعتبارسنجی/دکمهٔ Commit بلافاصله به‌روز می‌شوند.
   737→      مثال: تکراری با تعداد ۲ + معتبر با تعداد ۵ → همتا ۷ می‌شود. */
   738→  SetupView.mergeBulkDraft = function (i) {
   739→    const b = SetupView._bulk;
   740→    if (!b || i < 0 || i >= b.drafts.length) return;
   741→
   742→    const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   743→    const btn = tr ? tr.querySelector('[data-bulk-merge]') : null;
   744→    const target = btn ? Number(btn.dataset.mergeTarget) : NaN;
   745→    if (!Number.isInteger(target) || target < 0 || target >= b.drafts.length || target === i) {
   746→      UI.toast('همتای معتبری برای ادغام این ردیف یافت نشد.', 'warning');
   747→      return;
   748→    }
   749→
   750→    const dup = b.drafts[i];
   751→    const dest = b.drafts[target];
   752→    const dupCount = dup.rollCount === null ? 0 : (dup.rollCount || 0);
   753→    const destCount = dest.rollCount === null ? 0 : (dest.rollCount || 0);
   754→
   755→    dest.rollCount = destCount + dupCount;
   756→
   757→    const dupWidth = U.escapeHtml(String(dup.motherToken ?? '?'));
   758→    b.drafts.splice(i, 1);
   759→
   760→    SetupView.renderBulkPreview();
   761→    SetupView.renderBulkSummary();
   762→    SetupView.revalidateBulk();
   763→
   764→    UI.toast(
   765→      `ردیف تکراری (عرض ${dupWidth}، تعداد ${U.faNum(dupCount)}) در همتای معتبرش ادغام شد — تعداد رول همتا: ${U.faNum(destCount)} + ${U.faNum(dupCount)} = ${U.faNum(dest.rollCount)}.`,
   766→      'success',
   767→      6000
   768→    );
   769→  };

   771→  /** تغییر یک فیلد رکورد — خلاصه/سلول‌های همان ردیف بلافاصله؛ اعتبارسنجی با debounce */

   777→    if (!draft) return;

   779→    const field = el.dataset.bulkField;
   780→    if (field === 'rollCount') {
   781→      const v = U.parseInt(el.value);
   782→      draft.rollCount = v === null ? null : v;
   783→    } else {
   784→      draft[field] = el.value;
   785→    }
   786→
   787→    SetupView.renderBulkSummary();          // زنده — بلافاصله
   788→    SetupView.updateBulkRowComputed(i);
   789→    SetupView.scheduleBulkRevalidate();
   790→  };

   903→  /** اعمال نتیجهٔ اعتبارسنجی روی UI پیش‌نمایش (خطاهای سراسری + وضعیت ردیف‌ها) */
   904→  SetupView.applyBulkCheck = function (check) {
   905→    SetupView._bulkCheck = check;
   906→    const b = SetupView._bulk;
   907→    if (!b) return;
   908→
   909→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   910→    const globalMap = {
   911→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   912→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   913→    };
   914→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {
   915→      const errEl = document.getElementById(errId);
   916→      const inputEl = document.getElementById(inputId);
   917→      const msg = check.globalErrors[field] || '';
   918→      if (errEl) { errEl.textContent = msg; errEl.hidden = !msg; }
   919→      if (inputEl) inputEl.classList.toggle('invalid', !!msg);
   920→    }
   921→
   922→    // وضعیت هر ردیف
   923→    b.drafts.forEach((d, i) => {
   924→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   925→      if (!tr) return;
   926→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   927→      const rec = check.records[i];
   928→      const errs = rec ? rec.errors : [];
   929→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   930→      cell.innerHTML = errs.length
   931→        ? `<span class="status-pill pill-bad">✕ خطا</span>
   932→           <ul class="bulk-errors">${errs.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`
   933→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   934→    });
   935→
   936→    /* --- تغییر ۲ (نسخهٔ قبل): آیکون ادغام ردیف‌های تکراری با همتای معتبر ---
   937→       ردیفی که خطای «تکراری» دارد و ردیف معتبری با همان کلید یکتا در همین
   938→       Batch وجود دارد → دکمهٔ ادغام کنار آیکون حذف نمایش داده می‌شود. */
   939→    const validKeyIndex = new Map();   // uniqueKey → اندیس اولین ردیف معتبر
   940→    b.drafts.forEach((d, i) => {
   941→      const rec = check.records[i];
   942→      if (rec && !rec.errors.length && rec.record) {
   943→        const k = SetupView.bulkRowKey(d, b.common);
   944→        if (!validKeyIndex.has(k)) validKeyIndex.set(k, i);
   945→      }
   946→    });
   947→    b.drafts.forEach((d, i) => {
   948→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   949→      if (!tr) return;
   950→      const btn = tr.querySelector('[data-bulk-merge]');
   951→      if (!btn) return;
   952→
   953→      const rec = check.records[i];
   954→      const errs = rec ? rec.errors : [];
   955→      const isDuplicate = errs.some((e) => /تکراری/.test(String(e)));
   956→      const k = SetupView.bulkRowKey(d, b.common);
   957→      const target = isDuplicate ? (validKeyIndex.get(k) ?? null) : null;
   958→
   959→      if (target !== null && target !== i) {
   960→        btn.hidden = false;
   961→        btn.dataset.mergeTarget = String(target);
   962→        btn.title = `ادغام این ردیف تکراری (تعداد ${U.faNum(d.rollCount ?? 0)}) با ردیف ${U.faNum(target + 1)} معتبر — تعداد رول جمع می‌شود`;
   963→      } else {
   964→        btn.hidden = true;
   965→        delete btn.dataset.mergeTarget;
   966→      }
   967→    });
   968→
   969→    const commitBtn = document.getElementById('btn-bulk-commit');
   970→    const hasStructure = b.structureErrors.length > 0;
   971→    // نسخهٔ ۲٫۶: با حذف همهٔ رکوردها، دکمهٔ Commit هم غیرفعال می‌ماند
   972→    commitBtn.disabled = !(b.drafts.length > 0 && check.ok === true && !hasStructure);
   973→
   974→    /* --- پنل «کنترل تعداد مجاز» (تغییر ۱) + هشدار سراسری --- */
   975→    SetupView.renderBulkAllowedPanel(check);
   976→  };

  1054→  /** کارت‌های الگوی برش (§30 + تغییر ۲ — نمایش کارت به ازای هر قطعه)
  1055→      هر قطعهٔ الگو یک کارت مستقل دارد؛ عرض‌های تکراری کارت تکراری می‌سازند
  1056→      (مثلاً الگوی 1100-1100 برای مادر 2200 → دو کارت 1100) و نشان «×تعداد»
  1057→      زیر عرض‌ها حذف شد. زیر کارت‌ها:
  1058→        - بدون پرت: خط «Σ مجموع قطعه‌ها · فرزند هر مادر: N رول»
  1059→          (N = تعداد قطعه‌ها × تعداد ست — جزئیات در Tooltip)
  1060→        - دارای پرت (مادر > Σ الگو): فقط «پرت عرض: X» نمایش داده می‌شود. */
  1061→  SetupView.renderPatternCards = function (setup) {
  1062→    const parsed = Setups.parsePattern(setup.cuttingPattern);
  1063→    if (!setup.cuttingPattern) return '<span class="muted">—</span>';
  1064→    if (!parsed.ok) return `<span class="film-bad">${U.escapeHtml(setup.cuttingPattern)}</span>`;
  1065→
  1066→    /* تعداد ست از موتور محاسبه (تغییر ۱) — رکورد بی‌قانون = ۱ ست */
  1067→    const setCount = Number.isFinite(setup.setCount) && setup.setCount > 0 ? setup.setCount : 1;
  1068→
  1069→    /* تغییر ۲: یک کارت برای هر قطعهٔ الگو (بدون تجمیع عرض‌های یکسان) */
  1070→    const cards = parsed.parts
  1071→      .map((w) => `<span class="pattern-card">${U.faWidth(w)}</span>`)
  1072→      .join('');
  1073→
  1074→    const trim = setup.width - parsed.sum;
  1075→    const childPerMother = parsed.parts.length * setCount;
  1076→
  1077→    /* زیر کارت‌ها — فقط مقدار پرت وقتی الگو نسبت به مادر پرت دارد */
  1078→    const caption = trim > 0
  1079→      ? `<span class="pattern-trim" title="عرض رول مادر ${U.faWidth(setup.width)} − Σ الگو ${U.faWidth(parsed.sum)} = ${U.faWidth(trim)}">پرت عرض: ${U.faWidth(trim)}</span>`
  1080→      : `Σ ${U.faWidth(parsed.sum)} · فرزند هر مادر: ${U.faNum(childPerMother)} رول`;
  1081→
  1082→    return `
  1083→      <div class="pattern-cards">${cards}</div>
  1084→      <div class="pattern-sum" title="${U.faNum(parsed.parts.length)} قطعه × ${U.faNum(setCount)} ست = ${U.faNum(childPerMother)} رول فرزند از هر مادر · Σ الگو = ${U.faWidth(parsed.sum)}${setCount !== 1 ? ` (تعداد ست: ${U.faNum(setCount)})` : ''}">${caption}</div>`;
  1085→  };

  1158→    const sorted = [...rows];
  1159→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {

  1161→      sorted.sort((a, b) => {

  1166→          return direction === 'desc' ? -r : r;
  1167→        }
  1168→        return U.compareNumeric(a[field], b[field], direction);
  1169→      });
  1170→    }
  1171→    return sorted;
