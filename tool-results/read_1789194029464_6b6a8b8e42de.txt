   160→  /** ارث‌بری ضخامت (§22): اگر ستاپ موجود باشد → قفل خودکار */
   161→  SetupView.applyThicknessInheritance = async function () {
   162→    const { setupNumber, thickness, standardLength } = F();
   163→    const value = U.parseInt(setupNumber.value);
   164→    const hintEl = document.getElementById('hint-thickness');
   165→
   166→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   167→
   168→    if (value === null || value < 0) {
   169→      SetupView.thicknessLocked = false;
   170→      thickness.disabled = false;
   171→      hintEl.hidden = true;
   172→      return;
   173→    }
   174→
   175→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   176→    const withThickness = existing.find((s) => s.thickness !== null);
   177→
   178→    if (withThickness) {
   179→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   180→      SetupView.thicknessLocked = true;
   181→      thickness.value = String(withThickness.thickness);
   182→      thickness.disabled = true;
   183→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   184→      hintEl.hidden = false;
   185→    } else {
   186→      SetupView.thicknessLocked = false;
   187→      thickness.disabled = false;
   188→      hintEl.hidden = true;
   189→    }
   190→
   191→    SetupView.refreshStandardLengthOptions(standardLength.value);
   192→  };

   194→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   195→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   196→    const { thickness, standardLength } = F();
   197→    const thicknessValue = U.parseNumber(thickness.value);
   198→    const errEl = document.getElementById('err-standard');
   199→
   200→    const options = thicknessValue === null
   201→      ? []
   202→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   203→
   204→    let html = '<option value="">— انتخاب کنید —</option>';
   205→    for (const opt of options) {
   206→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   207→    }
   208→    standardLength.innerHTML = html;
   209→
   210→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   211→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   212→      standardLength.value = keepValue;
   213→    } else {
   214→      standardLength.value = '';
   215→    }
   216→
   217→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   218→    if (thicknessValue !== null && !options.length) {
   219→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   220→      errEl.hidden = false;
   221→    } else {
   222→      errEl.textContent = '';
   223→      errEl.hidden = true;
   224→    }
   225→  };

   228→  SetupView.updatePatternFeedback = function () {
   229→    const { width, cuttingPattern } = F();
   230→    const box = document.getElementById('pattern-feedback');
   231→
   232→    const result = Setups.validatePattern(cuttingPattern.value, U.parseNumber(width.value));
   233→
   234→    if (!cuttingPattern.value.trim()) {
   235→      box.className = 'pattern-feedback';
   236→      box.innerHTML = '<span class="muted">اختیاری — مثال: 1100-1170</span>';
   237→      return;
   238→    }
   239→
   240→    if (!result.ok) {
   241→      box.className = 'pattern-feedback invalid';
   242→      box.innerHTML = `<span class="pf-status">✕</span> ${result.message}`;
   243→      return;
   244→    }
   245→
   246→    const motherWidth = U.parseNumber(width.value);
   247→    const remaining = motherWidth !== null ? motherWidth - result.sum : null;
   248→
   249→    box.className = 'pattern-feedback valid';
   250→    box.innerHTML =
   251→      `<span class="pf-status">✓</span> مجموع: <b>${U.faWidth(result.sum)}</b>` +
   252→      (remaining !== null && remaining >= 0
   253→        ? ` · باقی‌مانده (پرت عرض): <b>${U.faWidth(remaining)}</b>`
   254→        : '') +
   255→      ` · بخش‌ها: ${U.faNum(result.parts.length)}`;
   256→  };

   262→  SetupView.submit = async function () {
   263→    const form = document.getElementById('setup-form');
   264→    const f = F();
   265→    UI.clearFormErrors(form);
   266→
   267→    const input = {
   268→      setupNumber: f.setupNumber.value,
   269→      machineNumber: f.machineNumber.value,
   270→      width: f.width.value,
   271→      thickness: f.thickness.value,
   272→      standardLength: f.standardLength.value,
   273→      rollCount: f.rollCount.value,
   274→      cuttingPattern: f.cuttingPattern.value,
   275→    };
   276→
   277→    /* --- ویرایش: بررسی آبشار ضخامت (§23) --- */
   278→    if (SetupView.editingId !== null) {
   279→      const newThickness = U.parseNumber(input.thickness);
   280→      const info = await Setups.cascadeInfo(SetupView.editingId, newThickness);
   281→
   282→      if (info.affected > 0) {
   283→        const ok = await UI.confirm({
   284→          title: 'تأیید تغییر ضخامت',
   285→          html: `
   286→            <div class="confirm-stats">
   287→              <div>تغییر ضخامت ستاپ <b>${U.faNum(info.setupNumber)}</b> باعث تغییر ضخامت
   288→              <b>${U.faNum(info.affected)}</b> رکورد دیگر نیز خواهد شد.</div>
   289→            </div>`,
   290→          message: 'این عملیات تراکنشی است و در صورت خطا به حالت قبل برمی‌گردد. ادامه می‌دهید؟',
   291→          confirmText: 'بله، ضخامت همه را تغییر بده',
   292→        });
   293→        if (!ok) {
   294→          UI.toast('عملیات ویرایش لغو شد.', 'info');
   295→          return;
   296→        }
   297→      }
   298→
   299→      const result = await Setups.update(SetupView.editingId, input, { cascadedCount: info.affected });
   300→      if (!result.ok) {
   301→        SetupView.showValidationErrors(result);
   302→        return;
   303→      }
   304→      UI.toast(
   305→        (info.affected > 0
   306→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   307→          : 'رکورد با موفقیت ویرایش شد.'),
   308→        'success'
   309→      );
   310→    } else {
   311→      /* --- ثبت جدید --- */
   312→      const result = await Setups.create(input);
   313→      if (!result.ok) {
   314→        SetupView.showValidationErrors(result);
   315→        return;
   316→      }
   317→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   318→    }
   319→
   320→    SetupView.resetForm();
   321→    await RM.refreshAll();
   322→    await SetupView.render();
   323→  };

   325→  /** نمایش خطاهای Field-Level (§69): حاشیهٔ قرمز + پیام + فوکوس */
   326→  SetupView.showValidationErrors = function (result) {
   327→    const form = document.getElementById('setup-form');
   328→    const f = F();
   329→    const fieldMap = {
   330→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   331→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   332→      width: [f.width, document.getElementById('err-width')],
   333→      thickness: [f.thickness, document.getElementById('err-thickness')],
   334→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   335→      rollCount: [f.rollCount, document.getElementById('err-count')],
   336→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   337→    };
   338→
   339→    let firstInvalid = null;
   340→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   341→      const msg = result.errors && result.errors[field];
   342→      UI.setFieldError(input, errEl, msg);
   343→      if (msg && !firstInvalid) firstInvalid = input;
   344→    }
   345→
   346→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   347→    if (firstInvalid) firstInvalid.focus();
   348→  };

   465→  /**
   466→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   467→   * بلافاصله به‌روز می‌شود (روش محاسبه + تغییر ۱ — تعداد ست):
   468→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   469→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   470→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست × تعداد)
   471→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   472→   * تعداد ست هر رکورد = متراژ استاندارد ÷ کوچکترین متراژ استاندارد ضخامت
   473→   * سراسری (رول ۱ ستی) — اگر متراژ انتخاب نشده باشد، ۱ ست فرض می‌شود.
   474→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه ۱-ستی):
   475→   *   ۱۶ / ۳ / ۴۸ / ۵
   476→   */
   477→  SetupView.bulkSummaryStats = function (drafts, rules = [], thickness = null) {
   478→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   479→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست × تعداد) — «جمع عرض الگوی برش»
   480→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   481→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   482→
   483→    for (const d of drafts) {
   484→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   485→      motherRollsTotal += rollCount;
   486→
   487→      const w = U.parseNumber(d.motherToken);
   488→      if (w !== null) motherWidthSet.add(w);
   489→
   490→      const parsed = Setups.parsePattern(d.patternText);
   491→      if (parsed.ok && parsed.parts.length) {
   492→        /* --- تغییر ۱: تعداد فرزند = قطعه‌های الگو × تعداد ست × تعداد مادر --- */
   493→        const std = U.parseNumber(d.standardLength);
   494→        const sc = RM.rulesEngine.setCount(rules, thickness, std);
   495→        const setCount = sc !== null && sc > 0 ? sc : 1;
   496→        patternWidthsTotal += parsed.parts.length * setCount * rollCount;
   497→        for (const part of parsed.parts) patternWidthSet.add(part);
   498→      }
   499→    }
   500→
   501→    return {
   502→      count: drafts.length,
   503→      motherRollsTotal,                            // ۱۶
   504→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   505→      patternWidthsTotal,                          // ۴۸ (با تعداد ست ضرب می‌شود)
   506→      uniquePatternWidths: patternWidthSet.size,   // ۵
   507→    };
   508→  };

   729→  /** ادغام ردیف تکراری با همتای معتبرش (تغییر ۲) — تعداد رول ردیف تکراری
   730→      به تعداد رول همتای معتبر اضافه می‌شود، ردیف تکراری حذف می‌شود و
   731→      خلاصه/اعتبارسنجی/دکمهٔ Commit بلافاصله به‌روز می‌شوند.
   732→      مثال: تکراری با تعداد ۲ + معتبر با تعداد ۵ → همتا ۷ می‌شود. */
   733→  SetupView.mergeBulkDraft = function (i) {
   734→    const b = SetupView._bulk;
   735→    if (!b || i < 0 || i >= b.drafts.length) return;
   736→
   737→    const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   738→    const btn = tr ? tr.querySelector('[data-bulk-merge]') : null;
   739→    const target = btn ? Number(btn.dataset.mergeTarget) : NaN;
   740→    if (!Number.isInteger(target) || target < 0 || target >= b.drafts.length || target === i) {
   741→      UI.toast('همتای معتبری برای ادغام این ردیف یافت نشد.', 'warning');
   742→      return;
   743→    }
   744→
   745→    const dup = b.drafts[i];
   746→    const dest = b.drafts[target];
   747→    const dupCount = dup.rollCount === null ? 0 : (dup.rollCount || 0);
   748→    const destCount = dest.rollCount === null ? 0 : (dest.rollCount || 0);
   749→
   750→    dest.rollCount = destCount + dupCount;
   751→
   752→    const dupWidth = U.escapeHtml(String(dup.motherToken ?? '?'));
   753→    b.drafts.splice(i, 1);
   754→
   755→    SetupView.renderBulkPreview();
   756→    SetupView.renderBulkSummary();
   757→    SetupView.revalidateBulk();
   758→
   759→    UI.toast(
   760→      `ردیف تکراری (عرض ${dupWidth}، تعداد ${U.faNum(dupCount)}) در همتای معتبرش ادغام شد — تعداد رول همتا: ${U.faNum(destCount)} + ${U.faNum(dupCount)} = ${U.faNum(dest.rollCount)}.`,
   761→      'success',
   762→      6000
   763→    );
   764→  };

   766→  /** تغییر یک فیلد رکورد — خلاصه/سلول‌های همان ردیف بلافاصله؛ اعتبارسنجی با debounce */
   767→  SetupView.onBulkRowInput = function (el) {
   768→    const b = SetupView._bulk;
   769→    if (!b) return;
   770→    const i = Number(el.dataset.bulkRow);
   771→    const draft = b.drafts[i];
   772→    if (!draft) return;
   773→
   774→    const field = el.dataset.bulkField;
   775→    if (field === 'rollCount') {
   776→      const v = U.parseInt(el.value);
   777→      draft.rollCount = v === null ? null : v;
   778→    } else {
   779→      draft[field] = el.value;
   780→    }
   781→
   782→    SetupView.renderBulkSummary();          // زنده — بلافاصله
   783→    SetupView.updateBulkRowComputed(i);
   784→    SetupView.scheduleBulkRevalidate();

   851→  SetupView.applyBulkCheck = function (check) {
   852→    SetupView._bulkCheck = check;
   853→    const b = SetupView._bulk;
   854→    if (!b) return;
   855→
   856→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   857→    const globalMap = {
   858→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   859→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   860→    };
   861→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {

   863→      const inputEl = document.getElementById(inputId);
   864→      const msg = check.globalErrors[field] || '';

   870→    b.drafts.forEach((d, i) => {

   872→      if (!tr) return;
   873→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   874→      const rec = check.records[i];
   875→      const errs = rec ? rec.errors : [];
   876→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   877→      cell.innerHTML = errs.length
   878→        ? `<span class="status-pill pill-bad">✕ خطا</span>

   880→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   881→    });

   883→    /* --- تغییر ۲: آیکون ادغام ردیف‌های تکراری با همتای معتبر ---
   884→       ردیفی که خطای «تکراری» دارد و ردیف معتبری با همان کلید یکتا در همین
   885→       Batch وجود دارد → دکمهٔ ادغام کنار آیکون حذف نمایش داده می‌شود. */
   886→    const validKeyIndex = new Map();   // uniqueKey → اندیس اولین ردیف معتبر
   887→    b.drafts.forEach((d, i) => {
   888→      const rec = check.records[i];
   889→      if (rec && !rec.errors.length && rec.record) {
   890→        const k = SetupView.bulkRowKey(d, b.common);
   891→        if (!validKeyIndex.has(k)) validKeyIndex.set(k, i);
   892→      }
   893→    });
   894→    b.drafts.forEach((d, i) => {
   895→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   896→      if (!tr) return;
   897→      const btn = tr.querySelector('[data-bulk-merge]');
   898→      if (!btn) return;
   899→
   900→      const rec = check.records[i];
   901→      const errs = rec ? rec.errors : [];
   902→      const isDuplicate = errs.some((e) => /تکراری/.test(String(e)));
   903→      const k = SetupView.bulkRowKey(d, b.common);
   904→      const target = isDuplicate ? (validKeyIndex.get(k) ?? null) : null;
   905→
   906→      if (target !== null && target !== i) {
   907→        btn.hidden = false;
   908→        btn.dataset.mergeTarget = String(target);
   909→        btn.title = `ادغام این ردیف تکراری (تعداد ${U.faNum(d.rollCount ?? 0)}) با ردیف ${U.faNum(target + 1)} معتبر — تعداد رول جمع می‌شود`;
   910→      } else {
   911→        btn.hidden = true;
   912→        delete btn.dataset.mergeTarget;
   913→      }
   914→    });

   916→    const commitBtn = document.getElementById('btn-bulk-commit');
   917→    const hasStructure = b.structureErrors.length > 0;
   918→    // نسخهٔ ۲٫۶: با حذف همهٔ رکوردها، دکمهٔ Commit هم غیرفعال می‌ماند
   919→    commitBtn.disabled = !(b.drafts.length > 0 && check.ok === true && !hasStructure);
   920→  };

  1001→  SetupView.renderPatternCards = function (setup) {
  1002→    const parsed = Setups.parsePattern(setup.cuttingPattern);
  1003→    if (!setup.cuttingPattern) return '<span class="muted">—</span>';
  1004→    if (!parsed.ok) return `<span class="film-bad">${U.escapeHtml(setup.cuttingPattern)}</span>`;
  1005→
  1006→    /* تعداد ست از موتور محاسبه (تغییر ۱) — رکورد بی‌قانون = ۱ ست */
  1007→    const setCount = Number.isFinite(setup.setCount) && setup.setCount > 0 ? setup.setCount : 1;
  1008→
  1009→    /* تکرار هر عرض در الگو (Multiset) */
  1010→    const mult = new Map();
  1011→    for (const p of parsed.parts) mult.set(p, (mult.get(p) || 0) + 1);
  1012→
  1013→    const cards = [...mult.entries()]
  1014→      .map(([w, m]) => `<span class="pattern-card">${U.faWidth(w)}<b class="pc-mult">×${U.faNum(m * setCount)}</b></span>`)
  1015→      .join('');
  1016→
  1017→    const trim = setup.width - parsed.sum;
  1018→    const childPerMother = parsed.parts.length * setCount;
  1019→    return `
  1020→      <div class="pattern-cards">${cards}</div>
  1021→      <div class="pattern-sum">
  1022→        Σ ${U.faWidth(parsed.sum)} · فرزند هر مادر: ${U.faNum(childPerMother)} رول${setCount !== 1 ? ` (${U.faNum(setCount)} ست)` : ''}
  1023→        ${trim > 0 ? ` · پرت عرض: ${U.faWidth(trim)}` : ''}
  1024→      </div>`;
  1025→  };

  1097→  SetupView.applySetupSort = function (rows) {
  1098→    const sorted = [...rows];
  1099→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
  1100→      const { field, direction } = SetupView.sortPipeline[i];
  1101→      sorted.sort((a, b) => {

  1103→          const ka = Setups.normalizePatternKey(a.cuttingPattern) || '';
  1104→          const kb = Setups.normalizePatternKey(b.cuttingPattern) || '';

  1106→          return direction === 'desc' ? -r : r;
  1107→        }
  1108→        return U.compareNumeric(a[field], b[field], direction);
  1109→      });
  1110→    }
  1111→    return sorted;
