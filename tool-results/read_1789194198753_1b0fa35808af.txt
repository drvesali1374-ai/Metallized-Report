    73→      // html اختیاری برای آمار تأیید (محتوای داخلی قابل اعتماد — UI خودمان می‌سازدش)
    74→      msgEl.innerHTML = (options.html || '') + U.escapeHtml(options.message || '');
    75→      btnOk.textContent = options.confirmText || 'بله، انجام بده';
    76→      btnOk.classList.toggle('btn-danger', !!options.danger);
    77→      btnOk.classList.toggle('btn-primary', !options.danger);
    78→
    79→      overlay.hidden = false;
    80→      btnOk.focus();
    81→
    82→      function close(result) {
    83→        overlay.hidden = true;
    84→        btnOk.removeEventListener('click', onOk);
    85→        btnCancel.removeEventListener('click', onCancel);
    86→        overlay.removeEventListener('click', onOverlay);
    87→        document.removeEventListener('keydown', onKey);
    88→        modalResolve = null;
    89→        resolve(result);
    90→      }
    91→      const onOk = () => close(true);
    92→      const onCancel = () => close(false);
    93→      const onOverlay = (e) => { if (e.target === overlay) close(false); };
    94→      const onKey = (e) => { if (e.key === 'Escape') close(false); };
    95→
    96→      btnOk.addEventListener('click', onOk);
    97→      btnCancel.addEventListener('click', onCancel);
    98→      overlay.addEventListener('click', onOverlay);
    99→      document.addEventListener('keydown', onKey);
   100→    });

   156→  /* ---------------- جدول صفحه‌بندی‌شده (§56) ---------------- */

   158→  /**
   159→   * رندر جدول صفحه‌بندی‌شده:
   160→   * @param wrap       عنصر نگهدارندهٔ جدول
   161→   * @param columns    [{key, label, render?, className?, sortable?}]
   162→   * @param rows       داده‌ها
   163→   * @param opts       { page, pageSize, emptyText, footerInfo }
   164→   */
   165→  UI.renderTable = function (wrap, columns, rows, opts = {}) {
   166→    const pageSize = opts.pageSize || RM.config.PAGE_SIZE;
   167→    const page = opts.page || 1;
   168→    const emptyText = opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.';
   169→
   170→    if (!rows.length) {
   171→      wrap.innerHTML = `
   172→        <div class="empty-state">
   173→          ${UI.icons.alert}
   174→          <p>${emptyText}</p>
   175→        </div>`;
   176→      return { page: 1, pages: 0 };
   177→    }
   178→
   179→    const pages = Math.max(1, Math.ceil(rows.length / pageSize));
   180→    const safePage = Math.min(page, pages);
   181→    const slice = rows.slice((safePage - 1) * pageSize, safePage * pageSize);
   182→
   183→    const thead = '<thead><tr>' +
   184→      columns.map((c) => `<th>${c.label}</th>`).join('') +
   185→      '</tr></thead>';
   186→
   187→    const tbody = '<tbody>' + slice.map((row) => {
   188→      const cells = columns.map((c) => {
   189→        const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   190→        return `<td class="${c.className || ''}">${content}</td>`;
   191→      }).join('');
   192→      return `<tr>${cells}</tr>`;
   193→    }).join('') + '</tbody>';
   194→
   195→    // نوار صفحه‌بندی (اگر بیش از یک صفحه)
   196→    const pager = pages > 1
   197→      ? `<div class="pager" role="navigation" aria-label="صفحه‌بندی جدول">
   198→           <button type="button" class="btn btn-ghost btn-sm" data-page="1" ${safePage === 1 ? 'disabled' : ''}>ابتدا</button>
   199→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage - 1}" ${safePage === 1 ? 'disabled' : ''}>قبلی</button>
   200→           <span class="pager-info">صفحهٔ ${U.faNum(safePage)} از ${U.faNum(pages)} — ${U.faNum(rows.length)} ردیف</span>
   201→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage + 1}" ${safePage === pages ? 'disabled' : ''}>بعدی</button>
   202→           <button type="button" class="btn btn-ghost btn-sm" data-page="${pages}" ${safePage === pages ? 'disabled' : ''}>انتها</button>
   203→         </div>`
   204→      : (opts.footerInfo ? `<div class="pager"><span class="pager-info">${opts.footerInfo(rows.length)}</span></div>` : '');
   205→
   206→    wrap.innerHTML = `
   207→      <div class="table-wrap table-scroll">
   208→        <table class="data-table">${thead}${tbody}</table>
   209→      </div>
   210→      ${pager}`;
   211→
   212→    // رویداد صفحه‌بندی
   213→    wrap.querySelectorAll('[data-page]').forEach((btn) => {
   214→      btn.addEventListener('click', () => {
   215→        const nextPage = Number(btn.dataset.page);
   216→        if (opts.onPageChange) opts.onPageChange(nextPage);
   217→      });
   218→    });
   219→
   220→    return { page: safePage, pages };
   221→  };

   322→  /* ---------------- پنل مرتب‌سازی مشترک (چیپ‌های کشیدنی — §36-§38) ---------------- */

   324→  /**
   325→   * رندر «پنل مرتب‌سازی» — چیپ‌های مراحل مرتب‌سازی با کشیدن‌ورها،
   326→   * تغییر جهت و حذف. مولفهٔ مشترک رول‌های خام و گزارش ستاپ‌ها.
   327→   * @param {HTMLElement} wrap       عنصر نگهدارندهٔ چیپ‌ها
   328→   * @param {Array}        pipeline  [{field, direction}]
   329→   * @param {object}       fields    رجیستری فیلدها ({label})
   330→   * @param {object}       handlers  { onDir(i), onRemove(i), onReorder(from, to), emptyText? }
   331→   */
   332→  UI.sortPanel = function (wrap, pipeline, fields, handlers) {
   333→    wrap.innerHTML = pipeline.map((step, i) => {
   334→      const meta = fields[step.field] || { label: step.field };
   335→      return `
   336→        <div class="sort-chip" draggable="true" data-index="${i}" title="برای تغییر اولویت بکشید">
   337→          <span class="grip">${UI.icons.grip}</span>
   338→          <span class="sort-priority">${U.faNum(i + 1)}</span>
   339→          <span class="sort-label">${U.escapeHtml(meta.label)}</span>
   340→          <button type="button" class="sort-dir ${step.direction}" data-dir="${i}"
   341→                  aria-label="تغییر جهت مرتب‌سازی" title="تغییر جهت">
   342→            ${step.direction === 'asc' ? '↑' : '↓'}
   343→          </button>
   344→          <button type="button" class="sort-remove" data-remove="${i}"
   345→                  aria-label="حذف از مرتب‌سازی" title="حذف">${UI.icons.cross}</button>
   346→        </div>`;
   347→    }).join('') ||
   348→      `<span class="muted" style="font-size:.78rem">${U.escapeHtml(handlers.emptyText || 'مرتب‌سازی پیش‌فرض')}</span>`;
   349→
   350→    // تغییر جهت
   351→    wrap.querySelectorAll('[data-dir]').forEach((btn) => {
   352→      btn.addEventListener('click', () => handlers.onDir(Number(btn.dataset.dir)));
   353→    });
   354→
   355→    // حذف مرحله
   356→    wrap.querySelectorAll('[data-remove]').forEach((btn) => {
   357→      btn.addEventListener('click', () => handlers.onRemove(Number(btn.dataset.remove)));
   358→    });
   359→
   360→    // ---------- کشیدن و رها کردن برای تغییر ترتیب (§37) ----------
   361→    let dragSrcIndex = null;
   362→
   363→    wrap.querySelectorAll('.sort-chip').forEach((chip) => {
   364→      chip.addEventListener('dragstart', (e) => {
   365→        dragSrcIndex = Number(chip.dataset.index);
   366→        chip.classList.add('dragging');
   367→        e.dataTransfer.effectAllowed = 'move';
   368→        e.dataTransfer.setData('text/plain', String(dragSrcIndex));
   369→      });
   370→
   371→      chip.addEventListener('dragend', () => {
   372→        chip.classList.remove('dragging');
   373→        wrap.querySelectorAll('.sort-chip').forEach((c) => c.classList.remove('drag-over'));
   374→        dragSrcIndex = null;
   375→      });
   376→
   377→      chip.addEventListener('dragover', (e) => {
   378→        e.preventDefault();
   379→        e.dataTransfer.dropEffect = 'move';
   380→        if (!chip.classList.contains('drag-over')) chip.classList.add('drag-over');
   381→      });
   382→
   383→      chip.addEventListener('dragleave', () => chip.classList.remove('drag-over'));
   384→
   385→      chip.addEventListener('drop', (e) => {
   386→        e.preventDefault();
   387→        const from = dragSrcIndex;
   388→        const to = Number(chip.dataset.index);
   389→        if (from === null || from === to) return;
   390→        handlers.onReorder(from, to);
   391→      });
   392→    });
   393→  };
   394→
   395→  /* ---------------- جدول بلند با فیلتر زیر سرستون‌ها (مشترک ۳ جدول) ---------------- */
   396→
   397→  /**
   398→   * نرمال‌سازی متن جست‌وجو — ارقام فارسی/عربی → لاتین + حذف جداکننده‌های
   399→   * هزارگان و فاصله‌ها. هم روی «مقدار جست‌وجو» و هم روی «متن سلول» اعمال
   400→   * می‌شود تا «۸» با 8 و «۲۳» با 23 هم‌ارزش شود.
   401→   */
   402→  UI.filterNormalize = function (text) {
   403→    return U.toLatinDigits(String(text ?? ''))
   404→      .replace(/[,\u066C\u2019\u066B\s]/g, '')
   405→      .trim()
   406→      .toLowerCase();
   407→  };
   408→
   409→  /**
   410→   * منطق تطبیق «شامل بودن» (نسخهٔ ۲٫۳):
   411→   *   - عدد و متن هر دو با «شامل بودن» فیلتر می‌شوند (نه تطابق دقیق)
   412→   *   - invert = true → برعکس: «شامل نشود»
   413→   * @param {string[]} tokens   متن‌های قابل‌جست‌وجوی سلول (نمایش + مقدار خام)
   414→   * @param {string}   value    مقدار واردشده در باکس فیلتر
   415→   * @param {boolean}  invert   منطق برعکس فعال است؟
   416→   */
   417→  UI.filterContains = function (tokens, value, invert) {
   418→    const needle = UI.filterNormalize(value);
   419→    if (needle === '') return true;      // فیلتر خالی → همه
   420→    const hay = (tokens || [])
   421→      .map((t) => UI.filterNormalize(t))
   422→      .filter(Boolean)
   423→      .join(' ');
   424→    const contains = hay.includes(needle);
   425→    return invert ? !contains : contains;
   426→  };
   427→
   428→  /** خواندن/ساخت وضعیت فیلتر یک ستون ({ value, invert }) از state */
   429→  UI.filterStateOf = function (state, key) {
   430→    if (!state.filters[key] || typeof state.filters[key] !== 'object') {
   431→      state.filters[key] = { value: '', invert: false };
   432→    }
   433→    return state.filters[key];
   434→  };
   435→
   436→  /** آیکون «≠» دکمهٔ برعکس‌کردن منطق فیلتر */
   437→  UI.icons.notContains = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="12" cy="12" r="9"/><line x1="5.8" y1="18.2" x2="18.2" y2="5.8"/></svg>';
   438→
   439→  /**
   440→   * جدول بلند (حداقل ۵۰ رکورد) + فیلتر پیشرفتهٔ زیر سرستون‌ها — مولفهٔ
   441→   * مشترک جدول‌های «رول‌های برش‌خورده»، «رول‌های خام» و «گزارش ستاپ‌ها».
   442→   *
   443→   * منطق نسخهٔ ۲٫۳:
   444→   *   - فیلتر همهٔ ستون‌ها (عدد و متن) با «شامل بودن» + ارقام فارسی/لاتین
   445→   *   - دکمهٔ کوچک «≠» کنار هر باکس → سوییچ «شامل نشود»
   446→   *   - فوکوس باکس فیلتر حفظ می‌شود (فقط tbody دوباره رندر می‌شود،
   447→   *     سرستون/باکس‌ها بازسازی نمی‌شوند و فوکوس جابه‌جا نمی‌شود)
   448→   *   - سرستون + ردیف فیلتر چسبان هنگام اسکرول عمودی
   449→   *
   450→   * منطق نسخهٔ ۲٫۴:
   451→   *   - باکس «ردیف‌های قابل نمایش» در نوار پایین → ارتفاع جدول داینامیک
   452→   *     (فقط استایل ارتفاع به‌روز می‌شود؛ جدول/فیلترها/فوکوس دست‌نخورده)
   453→   *   - پشتیبانی عرض ستون (c.width → colgroup با table-layout:fixed)
   454→   *
   455→   * منطق نسخهٔ ۲٫۵:
   456→   *   - Wrap Text هدرها: عنوان ستون‌ها با توجه به عرض ستون چندخط می‌شود؛
   457→   *     ارتفاع ردیف سرستون اندازه‌گیری و در متغیر --ft-head-h ثبت می‌شود تا
   458→   *     ردیف فیلتر چسبان دقیقاً زیر آن بنشیند (ResizeObserver → هم‌گام با تغییر عرض)
   459→   *   - چیدمان باکس فیلتر: آیکون «≠» کوچک در سمت راست + باکس فیلتر در ادامه
   460→   *     که عرضش با عرض ستون تنظیم می‌شود (min-width: 0 → هرگز ناقص نمی‌شود)
   461→   *
   462→   * منطق نسخهٔ ۲٫۶:
   463→   *   - باکس‌های جمع (opts.sums): زیر ستون‌های عددی/زمانی، ردیف پایانی چسبان
   464→   *     (tfoot) با جمع همان ستون — جمع همیشه با دیتای «فیلترشدهٔ» فعلی
   465→   *     محاسبه می‌شود و با هر فیلتر/تغییر، به‌روز می‌شود.
   466→   *
   467→   * @param {HTMLElement} wrap     نگهدارندهٔ جدول
   468→   * @param {Array}  columns       [{key, label, render(row)→html, tokens(row)→string[],
   469→   *                                className?, filterable? (پیش‌فرض true), width? (px — نسخهٔ ۲٫۴)}]
   470→   * @param {Array}  rows          همهٔ رکوردها (مرتب‌شده توسط فراخوان)
   471→   * @param {object} opts          { state: {filters:{}, visibleRows?} (ماندگار بین رندرها),
   472→   *                                visibleRows, rowHeight, tableClass, emptyText,
   473→   *                                filteredEmptyText, footerNote, rowsControl? (bool — باکس تعداد ردیف),
   474→   *                                onRowsChange?(n) (ذخیرهٔ ماندگار),
   475→   *                                fixedCols? (bool — table-layout:fixed برای عرض‌های ستون),
   476→   *                                sums? ({ colKey: {type:'number'|'time'|'pair',
   477→   *                                        weightOf?(row,key)→number } }) — نسخهٔ ۲٫۶,
   478→   *                                onRendered(visibleRows) }
   479→   */
   480→  UI.filterTable = function (wrap, columns, rows, opts = {}) {
   481→    if (!wrap) return;
   482→    const state = opts.state || { filters: {} };
   483→    state.filters = state.filters || {};
   484→    // تعداد ردیف قابل نمایش (نسخهٔ ۲٫۴): اولویت با state کاربر (ماندگار)، بعد opts
   485→    if (!Number.isFinite(Number(state.visibleRows)) || Number(state.visibleRows) <= 0) {
   486→      state.visibleRows = opts.visibleRows || 50;
   487→    }
   488→    const rowHeight = opts.rowHeight || 42;
   489→    const tableClass = opts.tableClass || 'cut-table';
   490→    const filterableCols = columns.filter((c) => c.filterable !== false);
   491→
   492→    /* ---------- تطبیق یک رکورد با همهٔ فیلترهای فعال ---------- */
   493→    const rowMatches = (row) => filterableCols.every((c) => {
   494→      const f = UI.filterStateOf(state, c.key);
   495→      if (String(f.value ?? '').trim() === '') return true;
   496→      return UI.filterContains(c.tokens ? c.tokens(row) : [String(row[c.key] ?? '')], f.value, f.invert);
   497→    });
   498→
   499→    /* ---------- ساخت سرستون + ردیف فیلتر (یک‌بار در هر رندر کامل) ---------- */
   500→    const headRow = columns.map((c) => `<th${c.headerTitle ? ` title="${U.escapeHtml(c.headerTitle)}"` : ''}${c.width ? ` style="width:${Number(c.width)}px;min-width:${Number(c.width)}px"` : ''}>${c.label}</th>`).join('');
   501→    const filterRow = columns.map((c) => {
   502→      if (c.filterable === false) return '<th class="cut-filter-cell"></th>';
   503→      const f = UI.filterStateOf(state, c.key);
   504→      return `
   505→      <th class="cut-filter-cell" aria-label="فیلتر ${U.escapeHtml(String(c.label))}">
   506→        <div class="ft-filter-box">
   507→          <button type="button" class="ft-invert${f.invert ? ' on' : ''}" data-ft-invert="${U.escapeHtml(c.key)}"
   508→                  aria-pressed="${String(!!f.invert)}"
   509→                  title="${f.invert ? 'منطق فعلی: شامل نشود — برای بازگشت به «شامل بودن» کلیک کنید' : 'برعکس‌کردن منطق: شامل نشود'}"
   510→                  aria-label="برعکس‌کردن منطق فیلتر ستون ${U.escapeHtml(String(c.label))}">
   511→            ${UI.icons.notContains}
   512→          </button>
   513→          <input type="text" class="cut-filter-input" data-ft-filter="${U.escapeHtml(c.key)}"
   514→                 value="${U.escapeHtml(f.value ?? '')}"
   515→                 autocomplete="off" placeholder="فیلتر…" inputmode="text"
   516→                 aria-label="فیلتر ستون ${U.escapeHtml(String(c.label))}">
   517→        </div>
   518→      </th>`;
   519→    }).join('');
   520→
   521→    /* ---------- باکس‌های جمع زیر ستون‌ها (نسخهٔ ۲٫۶) ----------
   522→       opts.sums = { colKey: { type: 'number' | 'time' | 'pair',
   523→                               weightOf?: (row, key) => number|null } }
   524→       جمع همیشه از دیتای فیلترشدهٔ فعلی (visible) محاسبه می‌شود. */
   525→    const sumSpecs = opts.sums || null;
   526→    const hasSums = !!(sumSpecs && Object.keys(sumSpecs).length);
   527→
   528→    /** ساخت HTML یک باکس جمع بر اساس نوع ستون */
   529→    const sumBoxHtml = (spec, cnt, wt, skipped) => {
   530→      if (spec.type === 'pair') {
   531→        return `<span class="ft-sum-sigma" aria-hidden="true">Σ</span>
   532→          <span class="ft-sum-pair">
   533→            <b class="ft-sum-val">${U.faNum(cnt)}</b>
   534→            <span class="ft-sum-wgt">Σ ${U.faWeightDot(wt)}</span>
   535→          </span>`;
   536→      }
   537→      const val = spec.type === 'time' ? U.faHMM(cnt) : U.faNum(cnt);
   538→      return `<span class="ft-sum-sigma" aria-hidden="true">Σ</span>
   539→        <b class="ft-sum-val${spec.type === 'time' ? ' ft-sum-time' : ''}">${val}</b>`
   540→        + (skipped ? `<sup class="ft-sum-skip" title="${U.faNum(skipped)} رکورد بدون قانون در جمع لحاظ نشد">*</sup>` : '');
   541→    };
   542→
   543→    /** به‌روزرسانی همهٔ باکس‌های جمع از ردیف‌های فیلترشده */
   544→    const updateSums = (visible) => {
   545→      if (!hasSums) return;
   546→      for (const c of columns) {
   547→        const spec = sumSpecs[c.key];
   548→        if (!spec) continue;
   549→        const box = wrap.querySelector(`[data-ft-sum-box="${c.key}"]`);
   550→        if (!box) continue;
   551→
   552→        if (spec.type === 'pair') {
   553→          let cnt = 0, wt = 0;
   554→          for (const row of visible) {
   555→            const v = Number(row[c.key]);
   556→            if (Number.isFinite(v)) cnt += v;
   557→            const w = spec.weightOf ? Number(spec.weightOf(row, c.key)) : NaN;
   558→            if (Number.isFinite(w)) wt += w;
   559→          }
   560→          box.innerHTML = sumBoxHtml(spec, cnt, wt, 0);
   561→          box.parentElement.title =
   562→            `جمع ستون «${String(c.label)}» با دیتای فیلترشده — ردیف اول: جمع تعداد · ردیف دوم: جمع وزن`;
   563→          continue;
   564→        }
   565→
   566→        let total = 0, counted = 0;
   567→        for (const row of visible) {
   568→          const v = row[c.key];
   569→          if (v === null || v === undefined || v === '') continue;   // «بدون قانون» و تهی‌ها
   570→          const n = Number(v);
   571→          if (Number.isFinite(n)) { total += n; counted++; }
   572→        }
   573→        const skipped = spec.type === 'time' ? (visible.length - counted) : 0;
   574→        box.innerHTML = sumBoxHtml(spec, total, 0, skipped);
   575→        box.parentElement.title = `جمع ستون «${String(c.label)}» — با دیتای فیلترشدهٔ فعلی`
   576→          + (spec.type === 'time' ? ` (${U.faNum(counted)} رکورد دارای قانون)` : '');
   577→      }
   578→    };
   579→
   580→    /* ---------- رندر بدنه (ایزوله — سرستون دست‌نخورده می‌ماند) ---------- */
   581→    const renderBody = () => {
   582→      const visible = rows.filter(rowMatches);
   583→      const tbody = wrap.querySelector('.ft-tbody');
   584→      const statusEl = wrap.querySelector('.ft-status');
   585→      if (!tbody) return visible;
   586→
   587→      const activeCount = filterableCols.filter((c) =>
   588→        String(UI.filterStateOf(state, c.key).value ?? '').trim() !== '').length;
   589→
   590→      tbody.innerHTML = visible.length
   591→        ? visible.map((row) => {
   592→            const cells = columns.map((c) => {
   593→              const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   594→              return `<td class="${c.className || ''}">${content}</td>`;
   595→            }).join('');
   596→            return `<tr>${cells}</tr>`;
   597→          }).join('')
   598→        : `<tr><td colspan="${columns.length}" class="muted" style="text-align:center;padding:18px">${
   599→            rows.length
   600→              ? (opts.filteredEmptyText || 'هیچ رکوردی با فیلترهای فعلی مطابقت ندارد — فیلترها را پاک کنید.')
   601→              : (opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.')
   602→          }</td></tr>`;
   603→
   604→      if (statusEl) {
   605→        statusEl.innerHTML = activeCount
   606→          ? `<span class="chip chip-strong">${U.faNum(visible.length)} از ${U.faNum(rows.length)} رکورد (فیلترشده)</span>`
   607→          : `<span class="chip">${U.faNum(rows.length)} رکورد</span>`;
   608→      }
   609→
   610→      updateSums(visible);     // نسخهٔ ۲٫۶: جمع‌ها با دیتای فیلترشده به‌روز می‌شوند
   611→
   612→      if (typeof opts.onRendered === 'function') opts.onRendered(visible, rows);
   613→      return visible;
   614→    };
   615→
   616→    /* ---------- اسکلت جدول (یک‌بار) ---------- */
   617→    if (!rows.length && !Object.values(state.filters).some((f) => f && String(f.value ?? '').trim() !== '')) {
   618→      wrap.innerHTML = `
   619→        <div class="empty-state">
   620→          ${UI.icons.alert}
   621→          <h4>رکوردی برای نمایش وجود ندارد</h4>
   622→          <p>${opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.'}</p>
   623→        </div>`;
   624→      if (typeof opts.onRendered === 'function') opts.onRendered([], rows);
   625→      return;
   626→    }
   627→
   628→    const applyHeight = (scrollEl, n) => {
   629→      scrollEl.style.height = `calc(${n} * ${rowHeight}px)`;
   630→      scrollEl.style.maxHeight = `calc(${n} * ${rowHeight}px)`;
   631→    };
   632→
   633→    // colgroup عرض ستون‌ها (نسخهٔ ۲٫۴) + جدول fixed-cols در صورت وجود عرض
   634→    const hasWidths = columns.some((c) => c.width);
   635→    const colgroup = hasWidths
   636→      ? `<colgroup>${columns.map((c) => `<col${c.width ? ` style="width:${Number(c.width)}px"` : ''}>`).join('')}</colgroup>`
   637→      : '';
   638→
   639→    const heightStyle = `height: calc(${state.visibleRows} * ${rowHeight}px); max-height: calc(${state.visibleRows} * ${rowHeight}px);`;
   640→
   641→    /* ردیف جمع چسبان زیر جدول (نسخهٔ ۲٫۶) — sticky bottom، هم‌عرض ستون‌ها؛
   642→       مقدار هر باکس در renderBody/updateSums با دیتای فیلترشده پر می‌شود. */
   643→    const sumFoot = hasSums
   644→      ? `<tfoot class="ft-sum-row"><tr>${columns.map((c) => {
   645→          if (!sumSpecs[c.key]) return '<td class="ft-sum-cell"></td>';
   646→          return `<td class="ft-sum-cell">
   647→            <div class="ft-sum-box" data-ft-sum-box="${c.key}"><span class="muted">…</span></div>
   648→          </td>`;
   649→        }).join('')}</tr></tfoot>`
   650→      : '';
   651→
   652→    wrap.innerHTML = `
   653→      <div class="cut-table-scroll" style="${heightStyle}">
   654→        <table class="data-table ${tableClass}${hasWidths ? ' fixed-cols' : ''}">
   655→          ${colgroup}
   656→          <thead>
   657→            <tr class="cut-head-row">${headRow}</tr>
   658→            <tr class="cut-filter-row">${filterRow}</tr>
   659→          </thead>
   660→          <tbody class="ft-tbody"></tbody>
   661→          ${sumFoot}
   662→        </table>
   663→      </div>
   664→      <div class="cut-filter-bar">
   665→        <span class="muted" style="font-size:.7rem">${opts.footerNote || 'فیلتر هر ستون در زیر سرستون همان ستون · منطق: شامل بودن (عدد و متن)'}</span>
   666→        <span class="ft-status"></span>
   667→        ${opts.rowsControl === false ? '' : `
   668→        <label class="ft-rows" title="تعداد ردیف قابل نمایش قبل از اسکرول — ارتفاع جدول طبق این عدد تغییر می‌کند">
   669→          <span>ردیف‌های قابل نمایش:</span>
   670→          <input type="number" class="ft-rows-input" data-ft-rows min="${RM.config.VISIBLE_ROWS_MIN}" max="${RM.config.VISIBLE_ROWS_MAX}" step="1" value="${state.visibleRows}" inputmode="numeric" aria-label="تعداد ردیف قابل نمایش">
   671→        </label>`}
   672→        <button type="button" class="btn btn-ghost btn-sm" data-ft-clear>پاک‌کردن فیلترها</button>
   673→      </div>`;
   674→
   675→    /* ---------- ارتفاع داینامیک سرستون (Wrap Text — نسخهٔ ۲٫۵) ----------
   676→       عنوان ستون‌ها ممکن است با عرض کم ستون چندخط شود؛ ارتفاع واقعی ردیف
   677→       سرستون اندازه‌گیری و در متغیر CSS --ft-head-h ثبت می‌شود تا ردیف فیلتر
   678→       چسبان دقیقاً زیر آن بنشیند. ResizeObserver تغییر عرض (پنل عرض ستون‌ها /
   679→       تغییر اندازهٔ پنجره) را دنبال می‌کند و مقدار را هم‌گام نگه می‌دارد. */
   680→    const syncHeadHeight = () => {
   681→      const headTr = wrap.querySelector('.cut-head-row');
   682→      if (!headTr) return;
   683→      const h = headTr.offsetHeight;
   684→      if (h > 0) wrap.style.setProperty('--ft-head-h', `${h}px`);
   685→    };
   686→    syncHeadHeight();
   687→    if (typeof ResizeObserver !== 'undefined') {
   688→      const headTr = wrap.querySelector('.cut-head-row');
   689→      if (wrap._ftHeadRO) { try { wrap._ftHeadRO.disconnect(); } catch (e) { /* noop */ } }
   690→      if (headTr) {
   691→        wrap._ftHeadRO = new ResizeObserver(syncHeadHeight);
   692→        wrap._ftHeadRO.observe(headTr);
   693→      }
   694→    }
   695→
   696→    /* ---------- رویدادها ---------- */
   697→
   698→    // ورود متن فیلتر → فقط tbody دوباره رندر می‌شود (فوکوس و caret حفظ می‌شود)
   699→    let timer = null;
   700→    wrap.querySelectorAll('[data-ft-filter]').forEach((el) => {
   701→      el.addEventListener('input', () => {
   702→        const f = UI.filterStateOf(state, el.dataset.ftFilter);
   703→        f.value = el.value;
   704→        clearTimeout(timer);
   705→        timer = setTimeout(renderBody, 180);
   706→      });
   707→    });
   708→
   709→    // دکمهٔ کوچک «≠» → سوییچ منطق شامل‌بودن/شامل‌نشدن
   710→    wrap.querySelectorAll('[data-ft-invert]').forEach((btn) => {
   711→      btn.addEventListener('click', () => {
   712→        const f = UI.filterStateOf(state, btn.dataset.ftInvert);
   713→        f.invert = !f.invert;
   714→        btn.classList.toggle('on', f.invert);
   715→        btn.setAttribute('aria-pressed', String(f.invert));
   716→        btn.title = f.invert
   717→          ? 'منطق فعلی: شامل نشود — برای بازگشت به «شامل بودن» کلیک کنید'
   718→          : 'برعکس‌کردن منطق: شامل نشود';
   719→        renderBody();
   720→      });
   721→    });
   722→

   733→        b.setAttribute('aria-pressed', 'false');
   734→        b.title = 'برعکس‌کردن منطق: شامل نشود';

   764→  /* ---------------- پنل ترتیب ستون‌ها (مشترک — چیپ‌های کشیدنی + باکس عرض) ---------------- */
   765→
   766→  /**
   767→   * پنل «ترتیب ستون‌ها» — نسخهٔ ۲٫۴:
   768→   *   - ترتیب با کشیدن‌ورها (drag) تغییر می‌کند (فلش‌های ↑/↓ حذف شدند)
   769→   *   - بجای فلش‌ها، هر ستون یک باکس عددی «عرض (px)» دارد؛ کاربر عرض دلخواه
   770→   *     هر ستون را وارد می‌کند (صفر یا خالی → عرض خودکار)
   771→   * برای جدول «گزارش تولید ستاپ‌ها».
   772→   * @param {HTMLElement} wrap      عنصر نگهدارندهٔ چیپ‌ها
   773→   * @param {Array}        items    [{key, label, width}] به ترتیب فعلی
   774→   * @param {object}       handlers { onMove(from, to), onWidth(key, width), emptyText? }
   775→   */
   776→  UI.orderPanel = function (wrap, items, handlers) {
   777→    if (!wrap) return;
   778→    wrap.innerHTML = items.map((it, i) => `
   779→      <div class="sort-chip order-chip" draggable="true" data-index="${i}" title="برای تغییر ترتیب بکشید">
   780→        <span class="grip">${UI.icons.grip}</span>
   781→        <span class="sort-priority">${U.faNum(i + 1)}</span>
   782→        <span class="sort-label">${U.escapeHtml(it.label)}</span>
   783→        <label class="order-width" title="عرض این ستون در جدول (پیکسل) — صفر یا خالی = خودکار">
   784→          <span class="order-width-unit">px</span>
   785→          <input type="number" class="order-width-input" data-order-width="${U.escapeHtml(it.key)}"
   786→                 min="0" max="1200" step="1" inputmode="numeric" aria-label="عرض ستون ${U.escapeHtml(it.label)}"
   787→                 value="${it.width != null ? Number(it.width) : ''}" placeholder="خودکار">
   788→        </label>
   789→      </div>`).join('') ||
   790→      `<span class="muted" style="font-size:.78rem">${U.escapeHtml(handlers.emptyText || 'بدون ستون')}</span>`;
   791→
   792→    // باکس عرض هر ستون (نسخهٔ ۲٫۴) — debounce سبک؛ خالی/۰ = عرض خودکار
   793→    let widthTimer = null;
   794→    wrap.querySelectorAll('[data-order-width]').forEach((inp) => {
   795→      inp.addEventListener('input', () => {
   796→        clearTimeout(widthTimer);
   797→        const key = inp.dataset.orderWidth;
   798→        widthTimer = setTimeout(() => {
   799→          const v = U.parseInt(inp.value);
   800→          handlers.onWidth(key, v === null ? 0 : Math.min(Math.max(v, 0), 1200));
   801→        }, 320);
   802→      });
   803→    });
   804→
   805→    // کشیدن و رها کردن
   806→    let dragSrcIndex = null;
   807→    wrap.querySelectorAll('.order-chip').forEach((chip) => {
   808→      chip.addEventListener('dragstart', (e) => {
   809→        dragSrcIndex = Number(chip.dataset.index);
   810→        chip.classList.add('dragging');
   811→        e.dataTransfer.effectAllowed = 'move';
   812→        e.dataTransfer.setData('text/plain', String(dragSrcIndex));
   813→      });
   814→      chip.addEventListener('dragend', () => {
   815→        chip.classList.remove('dragging');
   816→        wrap.querySelectorAll('.order-chip').forEach((c) => c.classList.remove('drag-over'));
   817→        dragSrcIndex = null;
   818→      });
   819→      chip.addEventListener('dragover', (e) => {
   820→        e.preventDefault();
   821→        e.dataTransfer.dropEffect = 'move';
   822→        if (!chip.classList.contains('drag-over')) chip.classList.add('drag-over');
   823→      });
   824→      chip.addEventListener('dragleave', () => chip.classList.remove('drag-over'));
   825→      chip.addEventListener('drop', (e) => {
   826→        e.preventDefault();
   827→        const from = dragSrcIndex;
   828→        const to = Number(chip.dataset.index);
   829→        if (from === null || from === to) return;
   830→        handlers.onMove(from, to);
   831→      });
   832→    });
   833→  };
   834→
   835→  RM.ui = UI;
   836→})();
