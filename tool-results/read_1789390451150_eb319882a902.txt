     1→/* =========================================================================
     2→   setups.js — موتور مدیریت ستاپ‌ها (§20-§30، §89-§91)
     3→   -------------------------------------------------------------------------
     4→   مدل ستاپ:
     5→     { id, setupNumber, machineNumber(1|2|3), width, thickness,
     6→       standardLength, rollCount, cuttingPattern, createdAt, updatedAt }
     7→
     8→   قوانین حاکم:
     9→     - کلید یکتا: عرض + شماره ستاپ + متراژ استاندارد + شماره دستگاه (§25)
    10→     - ارث‌بری ضخامت: رکورد جدیدِ ستاپ موجود → ضخامت قفل (§22)
    11→     - تغییر ضخامت در ویرایش → آبشار تراکنشی با تأیید کاربر (§23)
    12→     - متراژ استاندارد فقط از قوانینِ همان ضخامت (§24)
    13→     - دستگاه فقط ۱/۲/۳ — دستگاه متالایز (§21)
    14→     - الگوی برش: مجموع ≤ عرض مادر (§27-§28)
    15→   ========================================================================= */
    16→
    17→'use strict';
    18→
    19→(function () {
    20→  const U = RM.utils;
    21→  const R = RM.rulesEngine;
    22→  const db = RM.db.db;
    23→
    24→  const Setups = {};
    25→
    26→  /* ---------------- الگوی برش (§27-§29) ---------------- */
    27→
    28→  /**
    29→   * تجزیهٔ الگوی برش: "1100-1170" → [1100, 1170]
    30→   * ارقام فارسی/لاتین پذیرفته می‌شوند (§95).
    31→   * خروجی: { ok, parts, sum, message }
    32→   */
    33→  Setups.parsePattern = function (patternText) {
    34→    const raw = String(patternText ?? '').trim();
    35→    if (raw === '') return { ok: true, parts: [], sum: 0, message: '' };   // الگو اختیاری است
    36→
    37→    const tokens = raw.split('-');
    38→    const parts = [];
    39→
    40→    for (const token of tokens) {
    41→      const trimmed = token.trim();
    42→      const num = U.parseNumber(trimmed);
    43→      if (trimmed === '' || num === null || num <= 0) {
    44→        return {
    45→          ok: false, parts: [], sum: null,
    46→          message: `بخش «${U.escapeHtml(U.truncate(trimmed || 'خالی', 12))}» در الگوی برش عدد معتبر و مثبت نیست.`,
    47→        };
    48→      }
    49→      parts.push(Math.round(num * 100) / 100);   // حذف خطای اعشار شناور
    50→    }
    51→
    52→    const sum = parts.reduce((a, b) => a + b, 0);
    53→    return { ok: true, parts, sum, message: '' };
    54→  };
    55→
    56→  /**
    57→   * اعتبارسنجی الگو نسبت به عرض مادر (§28).
    58→   * قانون: PatternSum <= MotherWidth (کمتر مجاز — پرت عرض).
    59→   */
    60→  Setups.validatePattern = function (patternText, motherWidth) {
    61→    const parsed = Setups.parsePattern(patternText);
    62→    if (!parsed.ok) return parsed;
    63→    if (!parsed.parts.length) return { ok: true, parts: [], sum: 0, message: '' };
    64→
    65→    if (motherWidth === null || motherWidth === undefined || !(motherWidth > 0)) {
    66→      return { ok: false, parts: parsed.parts, sum: parsed.sum, message: 'ابتدا عرض رول مادر را وارد کنید.' };
    67→    }
    68→
    69→    if (parsed.sum > motherWidth) {
    70→      return {
    71→        ok: false, parts: parsed.parts, sum: parsed.sum,
    72→        message: `مجموع عرض الگوی برش (${U.faWidth(parsed.sum)}) بیشتر از عرض رول مادر (${U.faWidth(motherWidth)}) است.`,
    73→      };
    74→    }
    75→
    76→    return { ok: true, parts: parsed.parts, sum: parsed.sum, message: '' };
    77→  };
    78→
    79→  /* ---------------- نرمال‌سازی الگو برای کلید یکتا (ترتیب‌آگنوستیک) ---------------- */
    80→
    81→  /**
    82→   * کلید نرمال‌شدهٔ الگوی برش — ترتیب قطعات در یکسان‌بودن الگوها بی‌اثر است
    83→   * (مقایسه به‌صورت Multiset):
    84→   *   "1200-1050" و "1050-1200" → "1050-1200"
    85→   *   "800-930-930" ، "930-800-930" و "930-930-800" → "800-930-930"
    86→   * الگوهای واقعاً متفاوت کلیدهای متفاوت می‌سازند.
    87→   * الگوی خالی → "" (بدون الگو).
    88→   */
    89→  Setups.normalizePatternKey = function (patternText) {
    90→    const parsed = Setups.parsePattern(patternText);
    91→    if (!parsed.ok || !parsed.parts.length) return '';
    92→    return [...parsed.parts].sort((a, b) => a - b).join('-');
    93→  };
    94→
    95→  /* ---------------- ساخت کلید یکتا (§25 + الگوی برش) ---------------- */
    96→
    97→  /**
    98→   * کلید یکتای ستاپ: عرض + شماره ستاپ + متراژ استاندارد + شماره دستگاه
    99→   * + الگوی نرمال‌شدهٔ برش (ترتیب قطعات مهم نیست).
   100→   */
   101→  Setups.uniqueKey = function (setup) {
   102→    const patternKey = Setups.normalizePatternKey(setup.cuttingPattern);
   103→    return `${setup.width}-${setup.setupNumber}-${setup.standardLength}-${setup.machineNumber}-${patternKey}`;
   104→  };
   105→
   106→  /* ---------------- اعتبارسنجی کامل فرم (§26) ---------------- */
   107→
   108→  /**
   109→   * اعتبارسنجی کامل ورودی ستاپ (ایجاد و ویرایش).
   110→   * @param {object} input        مقادیر خام فرم
   111→   * @param {object} rules        قوانین متراژ موجود
   112→   * @param {Array}  allSetups    تمام ستاپ‌های موجود
   113→   * @param {number|null} editId  شناسهٔ رکورد در حال ویرایش (در بررسی یکتایی مستثنی)
   114→   * @param {object}  options     { allowThicknessChange } — در ویرایش، تغییر ضخامت
   115→   *                               با آبشار مجاز است (§23) و قفل ارث‌بری (§22) اعمال نمی‌شود
   116→   * خروجی: { ok } یا { ok:false, errors:{field:msg}, message }
   117→   */
   118→  Setups.validate = function (input, rules, allSetups, editId = null, options = {}) {
   119→    const errors = {};
   120→
   121→    /* --- شماره ستاپ: عدد صحیح نامنفی (§96) --- */
   122→    const setupNumber = U.parseInt(input.setupNumber);
   123→    if (setupNumber === null || setupNumber < 0) {
   124→      errors.setupNumber = 'شماره ستاپ باید عدد صحیح و نامنفی باشد.';
   125→    }
   126→
   127→    /* --- شماره دستگاه: فقط ۱/۲/۳ (§21) --- */
   128→    const machineNumber = U.parseInt(input.machineNumber);
   129→    if (!RM.config.VALID_MACHINES.includes(machineNumber)) {
   130→      errors.machineNumber = 'شماره دستگاه باید ۱، ۲ یا ۳ باشد (دستگاه‌های متالایز).';
   131→    }
   132→
   133→    /* --- عرض رول مادر: عدد مثبت --- */
   134→    const width = U.parseNumber(input.width);
   135→    if (width === null || width <= 0) {
   136→      errors.width = 'عرض رول مادر باید عددی بزرگ‌تر از صفر باشد.';
   137→    }
   138→
   139→    /* --- ضخامت + ارث‌بری (§22) --- */
   140→    const thickness = U.parseNumber(input.thickness);
   141→    if (thickness === null || thickness <= 0) {
   142→      errors.thickness = 'ضخامت باید عددی مثبت (میکرون) باشد.';
   143→    } else if (!options.allowThicknessChange) {
   144→      // در ثبت جدید: ضخامت ستاپ موجود به ارث می‌رسد و قفل است (§22)
   145→      const sameSetup = allSetups.filter(
   146→        (s) => s.setupNumber === setupNumber && (editId === null || s.id !== editId)
   147→      );
   148→      if (sameSetup.length && sameSetup.some((s) => s.thickness !== null && s.thickness !== thickness)) {
   149→        errors.thickness = `ضخامت ستاپ ${U.faNum(setupNumber)} قبلاً ${U.faNum(sameSetup[0].thickness)} تعیین شده و باید یکسان باشد (ارث‌بری ضخامت).`;
   150→      }
   151→    }
   152→
   153→    /* --- متراژ استاندارد: فقط از قوانین همان ضخامت (§24) --- */
   154→    const standardLength = U.parseNumber(input.standardLength);
   155→    if (standardLength === null) {
   156→      errors.standardLength = 'متراژ استاندارد را انتخاب کنید.';
   157→    } else if (thickness !== null && !R.standardLengthsFor(rules, thickness).includes(standardLength)) {
   158→      const available = R.standardLengthsFor(rules, thickness);
   159→      errors.standardLength = available.length
   160→        ? 'متراژ استاندارد انتخابی در قوانین این ضخامت تعریف نشده است.'
   161→        : 'برای این ضخامت هنوز قانون متراژی تعریف نشده — ابتدا در تنظیمات قانون تعریف کنید.';
   162→    }
   163→
   164→    /* --- تعداد رول: عدد صحیح ≥ ۱ (§49) --- */
   165→    const rollCount = U.parseInt(input.rollCount);
   166→    if (rollCount === null || rollCount < 1) {
   167→      errors.rollCount = 'تعداد رول باید عدد صحیح و حداقل ۱ باشد.';
   168→    }
   169→
   170→    /* --- الگوی برش (§27-§28) --- */
   171→    const patternResult = Setups.validatePattern(input.cuttingPattern, width);
   172→    if (!patternResult.ok) {
   173→      errors.cuttingPattern = patternResult.message;
   174→    }
   175→
   176→    /* --- تاریخ ستاپ (شمسی — نسخهٔ ۲٫۸): الزامی + ارث‌بری مانند ضخامت --- */
   177→    const setupDateParsed = U.parseJalaliDate(input.setupDate);
   178→    if (!setupDateParsed) {
   179→      errors.setupDate = 'تاریخ ستاپ را با فرمت شمسی معتبر وارد کنید — مثال: 1405/06/18 (روزهای هر ماه و سال کبیسه بررسی می‌شود).';
   180→    } else if (!options.allowDateChange) {
   181→      // در ثبت جدید: تاریخ ستاپ موجود به ارث می‌رسد و قفل است (مثل ضخامت §22)
   182→      const sameSetupD = allSetups.filter(
   183→        (s) => s.setupNumber === setupNumber && (editId === null || s.id !== editId) && s.setupDate
   184→      );
   185→      if (sameSetupD.length && sameSetupD[0].setupDate !== setupDateParsed.normalized) {
   186→        errors.setupDate = `تاریخ ستاپ ${U.faNum(setupNumber)} قبلاً ${U.faJalali(sameSetupD[0].setupDate)} ثبت شده و باید یکسان باشد (ارث‌بری تاریخ ستاپ).`;
   187→      }
   188→    }
   189→
   190→    if (Object.keys(errors).length) {
   191→      return { ok: false, errors, message: 'لطفاً خطاهای فرم را برطرف کنید.' };
   192→    }
   193→
   194→    const record = {
   195→      setupNumber,
   196→      machineNumber,
   197→      width,
   198→      thickness,
   199→      standardLength,
   200→      rollCount,
   201→      cuttingPattern: String(input.cuttingPattern ?? '').trim() || null,
   202→      setupDate: setupDateParsed ? setupDateParsed.normalized : null,   // نسخهٔ ۲٫۸ — YYYY/MM/DD شمسی
   203→    };
   204→
   205→    /* --- کلید یکتا (§25 + الگو) — الگو با ترتیب بی‌اثر --- */
   206→    const duplicate = allSetups.find(
   207→      (s) => (editId === null || s.id !== editId) && Setups.uniqueKey(s) === Setups.uniqueKey(record)
   208→    );
   209→    if (duplicate) {
   210→      return {
   211→        ok: false,
   212→        errors: { machineNumber: 'ترکیب عرض + شماره ستاپ + متراژ استاندارد + دستگاه + الگوی برش تکراری است.' },
   213→        message:
   214→          `کلید «${U.faWidth(width)}-${U.faNum(setupNumber)}-${U.faNum(standardLength)}-${U.faNum(machineNumber)}-${U.escapeHtml(Setups.normalizePatternKey(record.cuttingPattern) || 'بدون الگو')}» ` +
   215→          `قبلاً ثبت شده است (ترتیب قطعات الگو در یکتایی بی‌اثر است).`,
   216→      };
   217→    }
   218→
   219→    return { ok: true, record };
   220→  };
   221→
   222→  /* ---------------- عملیات CRUD ---------------- */
   223→
   224→  /** تاریخ ثبت‌شدهٔ اولین رکوردِ همان شماره ستاپ (ارث‌بری تاریخ — نسخهٔ ۲٫۸)
   225→      دقیقاً مانند منطق ضخامت: یک تاریخ برای همهٔ رکوردهای همان شماره ستاپ. */
   226→  Setups.findInheritedDate = async function (setupNumber) {
   227→    if (setupNumber === null || setupNumber === undefined || setupNumber < 0) return null;
   228→    const existing = await db.setupRolls.where('setupNumber').equals(setupNumber).toArray();
   229→    const withDate = existing.find((s) => s.setupDate);
   230→    return withDate ? withDate.setupDate : null;
   231→  };
   232→
   233→  /** ایجاد رکورد جدید — تراکنشی (§14) */
   234→  Setups.create = async function (input) {
   235→    const [rules, allSetups] = await Promise.all([
   236→      db.standardLengthRules.toArray(),
   237→      db.setupRolls.toArray(),
   238→    ]);
   239→
   240→    const check = Setups.validate(input, rules, allSetups);
   241→    if (!check.ok) return check;
   242→
   243→    const record = { ...check.record, createdAt: Date.now(), updatedAt: Date.now() };
   244→    const id = await db.transaction('rw', db.setupRolls, async () => {
   245→      return db.setupRolls.add(record);
   246→    });
   247→    return { ok: true, id };
   248→  };
   249→
   250→  /**
   251→   * ویرایش رکورد (§23، §26).
   252→   * اگر ضخامت تغییر کند و رکوردهای دیگری با همین شماره ستاپ وجود داشته باشد،
   253→   * کاربر باید تأیید کند (waterfall) — این تابع فقط پس از تأیید فراخوانی شود.
   254→   * UI وظیفهٔ گرفتن تأیید و گزارش تعداد رکوردهای متأثر را دارد (cascadeInfo).
   255→   */
   256→  Setups.update = async function (id, input, options = {}) {
   257→    const [rules, allSetups] = await Promise.all([
   258→      db.standardLengthRules.toArray(),
   259→      db.setupRolls.toArray(),
   260→    ]);
   261→
   262→    // در ویرایش، تغییر ضخامت (§23) و تاریخ (نسخهٔ ۲٫۸) با آبشار مجاز است
   263→    const check = Setups.validate(input, rules, allSetups, id, {
   264→      allowThicknessChange: true,
   265→      allowDateChange: true,
   266→    });
   267→    if (!check.ok) return check;
   268→
   269→    const existing = allSetups.find((s) => s.id === id);
   270→    if (!existing) return { ok: false, message: 'رکورد موردنظر یافت نشد.' };
   271→
   272→    const newRecord = { ...check.record, createdAt: existing.createdAt, updatedAt: Date.now() };
   273→
   274→    // حذف پرچم قدیمی پس از تکمیل ویرایش
   275→    if ('legacy' in newRecord) delete newRecord.legacy;
   276→
   277→    await db.transaction('rw', db.setupRolls, async () => {
   278→      await db.setupRolls.update(id, newRecord);
   279→
   280→      // آبشار ضخامت و تاریخ به سایر رکوردهای همان شماره ستاپ (§23 + نسخهٔ ۲٫۸) — تراکنشی
   281→      const others = allSetups.filter((s) => s.id !== id && s.setupNumber === newRecord.setupNumber);
   282→      for (const other of others) {
   283→        const patched = { ...other };
   284→        let changed = false;
   285→        if (other.thickness !== newRecord.thickness) {
   286→          patched.thickness = newRecord.thickness;
   287→          changed = true;
   288→        }
   289→        if ((other.setupDate || null) !== (newRecord.setupDate || null)) {
   290→          patched.setupDate = newRecord.setupDate;
   291→          changed = true;
   292→        }
   293→        if (changed) {
   294→          patched.updatedAt = Date.now();
   295→          if ('legacy' in patched) delete patched.legacy;
   296→          await db.setupRolls.update(other.id, patched);
   297→        }
   298→      }
   299→    });
   300→
   301→    return {
   302→      ok: true, id,
   303→      cascaded: options.cascadedCount || 0,
   304→      cascadedDate: options.cascadedDateCount || 0,
   305→    };
   306→  };
   307→
   308→  /** اطلاعات آبشار تغییر ضخامت و تاریخ: چند رکورد دیگر تغییر می‌کنند؟
   309→      (برای پیام تأیید §23 + نسخهٔ ۲٫۸ — UI وظیفهٔ گرفتن تأیید را دارد) */
   310→  Setups.cascadeInfo = function (id, newThickness, newDate) {
   311→    return db.transaction('r', db.setupRolls, async () => {
   312→      const all = await db.setupRolls.toArray();
   313→      const target = all.find((s) => s.id === id);
   314→      if (!target) return { affected: 0, affectedDate: 0, setupNumber: null };
   315→
   316→      const others = all.filter((s) => s.id !== id && s.setupNumber === target.setupNumber);
   317→      const affected = others.filter((s) => s.thickness !== newThickness).length;
   318→      const affectedDate = others.filter((s) => (s.setupDate || null) !== (newDate || null)).length;
   319→      return { affected, affectedDate, setupNumber: target.setupNumber };
   320→    });
   321→  };
   322→
   323→  /** حذف رکورد (تأیید در UI — §70) */
   324→  Setups.remove = async function (id) {
   325→    await db.transaction('rw', db.setupRolls, async () => {
   326→      await db.setupRolls.delete(id);
   327→    });
   328→  };
   329→
   330→  /** مجموع تعداد رول‌های همهٔ ستاپ‌ها (§57) */
   331→  Setups.totalRollCount = function (setups) {
   332→    return setups.reduce((sum, s) => sum + (s.rollCount || 0), 0);
   333→  };

   335→  /* ================================================================
   336→     ثبت گروهی از Clipboard — Bulk Paste (TSV)
   337→     ----------------------------------------------------------------
   338→     ساختار ورودی — هر «جفت خط» یک ست دست تولید می‌کند:
   339→
   340→       خط فرد (الگو):   عرض‌های الگو ... [سلول‌های خالی] ... تعداد رول
   341→       خط زوج (مادر):   عرض رول‌های مادر — هر سلول غیرخالی، گروه جدید
   342→                        شروع می‌کند؛ سلول خالی یعنی ادامهٔ مادر قبلی.
   343→                        هر قطعهٔ الگو به مادر متناظر خودش متصل می‌شود.
   344→
   345→     الگوها می‌توانند یک‌تکه، دو‌تکه یا چندتکه باشند.
   346→
   347→     جریان (قابل ویرایش پس از تحلیل):
   348→       ۱) buildBulkDraft    — تجزیهٔ متن + تجمیع هم‌کلیدها → پیش‌نویس رکوردها
   349→       ۲) پیش‌نمایش UI      — کاربر فیلدهای سراسری (شماره ستاپ/ضخامت) و
   350→                              فیلدهای هر رکورد (دستگاه/متراژ استاندارد/
   351→                              عرض مادر/الگو/تعداد) را ویرایش می‌کند
   352→       ۳) validateBulkDraft — اعتبارسنجی با همان منطق ثبت دستی (§26)
   353→       ۴) commitBulkDraft   — ثبت همهٔ رکوردها در یک تراکنش واحد
   354→     ================================================================ */
   355→
   356→  /**
   357→   * تجزیهٔ متن Paste شده به رکوردهای ساختاری (بدون اعتبارسنجی منطقی).
   358→   * خروجی: { pairs, structureErrors }
   359→   *   pairs: [{ pairIndex, lineNos, rollCountToken, rollCount, groups: [{ motherToken, parts: [token] }] }]
   360→   */
   361→  Setups.parseBulkText = function (text) {
   362→    const rawLines = String(text ?? '').replace(/\r/g, '').split('\n');
   363→    const cells = rawLines.map((l) => l.split('\t'));
   364→
   365→    // اندیس خطوط دارای محتوا (خط کاملاً خالی نادیده گرفته می‌شود)
   366→    const contentIdx = [];
   367→    cells.forEach((row, i) => {
   368→      if (row.some((c) => String(c).trim() !== '')) contentIdx.push(i);
   369→    });
   370→
   371→    const pairs = [];
   372→    const structureErrors = [];
   373→
   374→    for (let p = 0; p < contentIdx.length; p += 2) {
   375→      const patternLine = contentIdx[p];
   376→      const motherLine = contentIdx[p + 1];          // undefined → خط مادر غایب
   377→      const patternCells = cells[patternLine];
   378→      const motherCells = motherLine !== undefined ? cells[motherLine] : [];
   379→      const pairIndex = pairs.length + 1;
   380→
   381→      // --- آخرین سلول غیرخالیِ خط الگو = تعداد رول ---
   382→      let countIdx = -1;
   383→      for (let i = patternCells.length - 1; i >= 0; i--) {
   384→        if (String(patternCells[i] ?? '').trim() !== '') { countIdx = i; break; }
   385→      }
   386→      if (countIdx <= 0) {
   387→        structureErrors.push({ pairIndex, message: `جفت ${U.faNum(pairIndex)}: خط الگو ساختار معتبری ندارد (عرض الگو یا تعداد رول یافت نشد).` });
   388→        continue;
   389→      }
   390→
   391→      const rollCountToken = String(patternCells[countIdx]).trim();
   392→
   393→      // --- سلول‌های مادر غیرخالی (فقط تا اندیس تعداد) — شروع گروه ---
   394→      const motherStarts = new Map();   // اندیس ستون → توکن عرض مادر
   395→      const limit = Math.min(motherCells.length, countIdx);
   396→      for (let i = 0; i < limit; i++) {
   397→        const t = String(motherCells[i] ?? '').trim();
   398→        if (t !== '') motherStarts.set(i, t);
   399→      }
   400→
   401→      // --- گروه‌بندی قطعات الگو زیر مادر متناظر ---
   402→      const groups = [];
   403→      let current = null;
   404→      let orphanParts = 0;
   405→
   406→      for (let i = 0; i < countIdx; i++) {
   407→        const t = String(patternCells[i] ?? '').trim();
   408→        if (t === '') continue;
   409→
   410→        if (motherStarts.has(i)) {
   411→          current = { motherToken: motherStarts.get(i), parts: [] };
   412→          groups.push(current);
   413→        }
   414→        if (current === null) {
   415→          orphanParts++;
   416→          continue;
   417→        }
   418→        current.parts.push(t);
   419→      }
   420→
   421→      /* --- نسخهٔ ۲٫۴: بدون ادغام گروه‌های یکسان — همهٔ عرض‌ها حتی تکراری
   422→             به‌صورت رکورد جداگانه نمایش داده می‌شوند تا کاربر بتواند هر
   423→             نسخه را به دستگاه/متراژ متفاوت تخصیص دهد (ادغام باعث از دست
   424→             رفتن این امکان بود). تکراری‌بودنِ نهایی در اعتبارسنجی (§26)
   425→             با کلید یکتا بررسی و رد می‌شود. --- */
   426→      for (const g of groups) {
   427→        if (g.occurrences == null) g.occurrences = 1;
   428→      }
   429→      const mergedGroups = groups;
   430→
   431→      if (motherLine === undefined) {
   432→        structureErrors.push({
   433→          pairIndex,
   434→          message: `جفت ${U.faNum(pairIndex)}: خط رول مادر متناظر برای خط الگو وجود ندارد.`,
   435→        });
   436→      }
   437→      if (orphanParts > 0) {
   438→        structureErrors.push({
   439→          pairIndex,
   440→          message: `جفت ${U.faNum(pairIndex)}: ${U.faNum(orphanParts)} قطعهٔ الگو در ابتدای خط مادر قرار ندارند و رول مادر متناظری ندارند.`,
   441→        });
   442→      }
   443→
   444→      pairs.push({
   445→        pairIndex,
   446→        lineNos: { pattern: patternLine + 1, mother: motherLine !== undefined ? motherLine + 1 : null },
   447→        rollCountToken,
   448→        rollCount: U.parseInt(rollCountToken),
   449→        /* همهٔ سلول‌های غیرخالی ردیف مادر (تا ستون تعداد) — مبنای مرجع
   450→           «تعداد مجاز»: هر سلول = یک تکرار عرض در آن ردیف، حتی اگر سلول
   451→           الگوی متناظرش خالی باشد و رکوردی نسازد (کنترل باید آن را گزارش کند). */
   452→        motherCells: [...motherStarts.values()],
   453→        groups: mergedGroups,
   454→      });
   455→    }
   456→
   457→    return { pairs, structureErrors };
   458→  };
   459→
   460→  /** فیلدهای سراسری Batch — خطاهایشان یک‌بار در سطح Batch گزارش می‌شوند */
   461→  Setups.BULK_GLOBAL_FIELDS = ['setupNumber', 'thickness', 'setupDate'];
   462→
   463→  /**
   464→   * ساخت پیش‌نویس رکوردهای Bulk از متن Paste‌شده (بدون اعتبارسنجی منطقی).
   465→   *
   466→   * نسخهٔ ۲٫۴ — بدون تجمیع: هر گروه (مادر + الگو) در هر خط Paste، حتی با
   467→   * عرض/الگوی مشابه، یک رکورد جداگانهٔ پیش‌نمایش می‌سازد؛ بنابراین کاربر
   468→   * می‌تواند عرض تکراری ردیف ۱ را به دستگاه ۲ و عرض تکراری ردیف ۳ را به
   469→   * دستگاه ۱ تخصیص دهد یا متراژ استاندارد متفاوت انتخاب کند. اگر در پایان
   470→   * (عرض مادر + شماره ستاپ + متراژ استاندارد + دستگاه + الگوی برش نرمال)
   471→   * تکراری بود، اعتبارسنجی (§26) ردیف را خطا می‌دهد و ثبت انجام نمی‌شود.
   472→   *
   473→   * commonInit: مقادیر اولیهٔ دستگاه/متراژ استاندارد ردیف‌ها (از فرم دستی).
   474→   * خروجی: { drafts, structureErrors, allowedMap }
   475→   *   drafts: [{ motherToken, patternText, rollCount, occurrences, pairs,
   476→   *              machineNumber, standardLength }]
   477→   *
   478→   * allowedMap (کنترل «تعداد مجاز» — تغییر جدید):
   479→   *   مرجعِ مقایسهٔ عرض‌های مادر دیتای پیست‌شده. برای هر عرض یکتای مادر:
   480→   *     تعداد مجاز (دیتای پیست‌شده) = عرض × تعداد تکرار عرض در ردیف مادر
   481→   *                                      × عدد انتهایی ردیف الگوی بالایی
   482→   *   مقدار ذخیره‌شده در نقشه فقط Σ(تکرار × عدد انتهایی) است (بدون ضرب عرض)
   483→   *   و عرض در لحظهٔ مقایسه/نمایش ضرب می‌شود (طبق تعریف کاربر:
   484→   *   مثلاً 1180 → 1180×8 + 1180×2 = 11800).
   485→   */
   486→  Setups.buildBulkDraft = function (commonInit, bulkText) {
   487→    const parsed = Setups.parseBulkText(bulkText);
   488→    const initMachine = String(commonInit.machineNumber ?? '');
   489→    const initStandard = String(commonInit.standardLength ?? '');
   490→
   491→    const drafts = [];
   492→    const allowedMap = {};    // "عرض(کلید عددی)" → Σ (تکرار × عدد انتهایی ردیف الگو)
   493→    for (const pair of parsed.pairs) {
   494→      const multiplier = pair.rollCount === null ? 0 : pair.rollCount;
   495→      for (const group of pair.groups) {
   496→        drafts.push({
   497→          motherToken: group.motherToken,
   498→          patternText: group.parts.join('-'),
   499→          rollCount: pair.rollCount,
   500→          occurrences: group.occurrences || 1,
   501→          pairs: [pair.pairIndex],
   502→          machineNumber: initMachine,      // قابل ویرایش در هر رکورد
   503→          standardLength: initStandard,    // قابل ویرایش در هر رکورد
   504→        });
   505→      }
   506→
   507→      /* --- مرجع تعداد مجاز عرض‌های مادر (دیتای پیست‌شده) —
   508→           «همهٔ سلول‌های ردیف مادر» شمرده می‌شوند (هر سلول = یک تکرار
   509→           عرض در همان ردیف × عدد انتهایی ردیف الگوی بالایی)، حتی اگر
   510→           سلول الگوی متناظر خالی باشد و رکوردی نسازد — در آن صورت
   511→           پیش‌نمایش آن عرض را ندارد و کنترل خطای «کمتر» می‌دهد. --- */
   512→      for (const token of (pair.motherCells || [])) {
   513→        const w = U.parseNumber(token);
   514→        if (w !== null && w > 0) {
   515→          const key = String(Math.round(w * 100) / 100);
   516→          allowedMap[key] = (allowedMap[key] || 0) + multiplier;
   517→        }
   518→      }
   519→    }
   520→
   521→    return { drafts, structureErrors: parsed.structureErrors, allowedMap };
   522→  };
   523→
   524→  /**
   525→   * اعتبارسنجی پیش‌نویس Bulk — دقیقاً همان منطق ثبت دستی (§26).
   526→   * @param {object} common   { setupNumber, thickness } — سراسری برای کل Batch
   527→   * @param {Array}  drafts   پیش‌نویس‌های (احتمالاً ویرایش‌شدهٔ) رکوردها
   528→   * @param {object} options  { allowedMap } — مرجع «تعداد مجاز» عرض‌های مادر
   529→   *                          دیتای پیست‌شده (از buildBulkDraft — اختیاری)
   530→   * خروجی: { ok, globalErrors, records, total, allowedRows }
   531→   *   globalErrors: خطاهای فیلدهای سراسری (یک‌بار)
   532→   *   records[i]:   { ...draft, errors: [msgs], record? } — record وقتی معتبر
   533→   *   allowedRows: ردیف‌های کنترل «تعداد مجاز» هر عرض یکتای مادر:
   534→   *     { width, key, pastedCount, previewCount, pasted, preview, status }
   535→   *     status ∈ 'ok' | 'less' | 'more' — عرض × (تعداد × ست) دو طرف باید برابر باشد.
   536→   */
   537→  Setups.validateBulkDraft = async function (common, drafts, options = {}) {
   538→    const [rules, allSetups] = await Promise.all([
   539→      db.standardLengthRules.toArray(),
   540→      db.setupRolls.toArray(),
   541→    ]);
   542→
   543→    const working = [...allSetups];
   544→    const globalErrors = {};
   545→    const records = [];
   546→
   547→    // کلیدهای موجود در DB و کلیدهای قبول‌شدهٔ همین Batch — برای پیغام خطای
   548→    // دقیقِ «تکراری» (نسخهٔ ۲٫۴): تکراریِ داخل Batch یا تکراری با رکورد ثبت‌شده
   549→    const dbKeys = new Set(allSetups.map((s) => Setups.uniqueKey(s)));
   550→    const batchKeys = new Set();
   551→
   552→    // ضخامت معتبر اما بدون قانون متراژ → خطای سراسری (نه تکرار در همهٔ ردیف‌ها)
   553→    const commonThickness = U.parseNumber(common.thickness);
   554→    if (commonThickness !== null && !R.standardLengthsFor(rules, commonThickness).length) {
   555→      globalErrors.thickness = 'برای این ضخامت قانون متراژی تعریف نشده — ابتدا در تنظیمات قانون اضافه کنید.';
   556→    }
   557→
   558→    /* --- تاریخ ستاپ Batch (نسخهٔ ۲٫۸): یک فیلد تاریخ برای کل دیتای پیست‌شده ---
   559→       اگر ستاپِ سراسری قبلاً تاریخ ثبت‌شده داشته باشد، همان تاریخ به‌ارث می‌رسد
   560→       و قفل است (دقیقاً مانند منطق ضخامت — یک تاریخ برای همهٔ رکوردهای همان
   561→       شماره ستاپ)؛ در غیر این صورت تاریخ Batch الزامی و باید شمسیِ معتبر باشد. */
   562→    const commonSetupNo = U.parseInt(common.setupNumber);
   563→    const inheritedDate = commonSetupNo !== null
   564→      ? (allSetups.find((s) => s.setupNumber === commonSetupNo && s.setupDate) || null)
   565→      : null;
   566→    let batchDate = null;
   567→    if (inheritedDate) {
   568→      batchDate = U.parseJalaliDate(inheritedDate.setupDate);
   569→    } else {
   570→      batchDate = U.parseJalaliDate(common.setupDate || '');
   571→      if (!batchDate) {
   572→        globalErrors.setupDate = 'تاریخ ستاپ (شمسی) برای کل رکوردهای همین Batch الزامی است — مثال: 1405/06/18.';
   573→      }
   574→    }
   575→    const batchDateStr = batchDate ? batchDate.normalized : null;
   576→
   577→    for (const draft of drafts) {
   578→      const input = {
   579→        setupNumber: common.setupNumber,
   580→        machineNumber: draft.machineNumber,
   581→        thickness: common.thickness,
   582→        standardLength: draft.standardLength,
   583→        width: draft.motherToken,
   584→        rollCount: draft.rollCount === null ? '' : String(draft.rollCount),
   585→        cuttingPattern: draft.patternText,
   586→        setupDate: batchDateStr || common.setupDate || '',   // نسخهٔ ۲٫۸ — تاریخ کل Batch
   587→      };
   588→
   589→      // کلید یکتای این ردیف (نمایشی — ترتیب قطعات الگو بی‌اثر)
   590→      const rowKey = Setups.uniqueKey({
   591→        width: U.parseNumber(draft.motherToken),
   592→        setupNumber: U.parseInt(common.setupNumber),
   593→        standardLength: U.parseNumber(draft.standardLength),
   594→        machineNumber: U.parseInt(draft.machineNumber),
   595→        cuttingPattern: draft.patternText,
   596→      });
   597→
   598→      const check = Setups.validate(input, rules, working);
   599→      const row = { ...draft, errors: [] };
   600→
   601→      if (check.ok) {
   602→        working.push(check.record);
   603→        batchKeys.add(rowKey);
   604→        row.record = check.record;
   605→      } else {
   606→        const isDuplicate = /تکراری/.test(String(check.errors.machineNumber || ''));
   607→        for (const [field, msg] of Object.entries(check.errors)) {
   608→          if (Setups.BULK_GLOBAL_FIELDS.includes(field)) {
   609→            if (!globalErrors[field]) globalErrors[field] = msg;   // یک‌بار در سطح Batch
   610→          } else if (field === 'standardLength' && globalErrors.thickness) {
   611→            // ریشهٔ خطای متراژ، ضخامت نامعتبر سراسری است — همان گزارش می‌شود
   612→            continue;
   613→          } else if (isDuplicate && field === 'machineNumber') {
   614→            // پیغام دقیق تکراری (نسخهٔ ۲٫۴): تفکیک تکراریِ داخل Batch از ثبت‌شده
   615→            row.errors.push(
   616→              (batchKeys.has(rowKey) || dbKeys.has(rowKey))
   617→                ? `ترکیب تکراری — «عرض مادر ${U.faWidth(U.parseNumber(draft.motherToken) ?? '?')} + شماره ستاپ ${U.faNum(U.parseInt(common.setupNumber) ?? '?')} + متراژ استاندارد ${U.faNum(U.parseNumber(draft.standardLength) ?? '?')} + دستگاه ${U.faNum(U.parseInt(draft.machineNumber) ?? '?')} + الگوی برش ${U.escapeHtml(Setups.normalizePatternKey(draft.patternText) || 'بدون الگو')}» ${dbKeys.has(rowKey) && !batchKeys.has(rowKey) ? 'قبلاً در پایگاه‌داده ثبت شده است' : 'در همین Batch تکراری است'} (ترتیب قطعات الگو بی‌اثر). ثبت مجدد مجاز نیست — دستگاه، متراژ استاندارد یا الگو را تغییر دهید.`
   618→                : msg
   619→            );
   620→          } else {
   621→            row.errors.push(msg);
   622→          }
   623→        }
   624→      }
   625→      records.push(row);
   626→    }
   627→
   628→    /* --- کنترل «تعداد مجاز» عرض‌های مادر (تغییر جدید — ۱) -------------------
   629→       مرجع: دیتای پیست‌شده (allowedMap از buildBulkDraft)؛
   630→       پیش‌نمایش: Σ (تعداد ست × تعداد رول) روی رکوردهای همان عرض.
   631→       تعداد ست هر رکورد = متراژ استاندارد ÷ کوچکترین استاندارد ضخامت
   632→       (موتور قوانین — رکورد بدون متراژ ۰ حساب می‌شود).
   633→       دو مقدار باید برای هر عرض یکتا «برابر» باشد؛ در غیر این صورت خطا:
   634→         پیش‌نمایش < پیست‌شده → «تعداد کمتر از دیتای ستاپ است»
   635→         پیش‌نمایش > پیست‌شده → «تعداد رول‌های آن عرض بیشتر از دیتای ستاپ است»
   636→       خطا روی «همهٔ» رکوردهای همان عرض نمایش داده می‌شود (عرض‌های تکراری).
   637→       -------------------------------------------------------------------- */
   638→    let allowedRows = null;
   639→    const allowedMap = options.allowedMap || null;
   640→    if (allowedMap && Object.keys(allowedMap).length) {
   641→      const commonThickness = U.parseNumber(common.thickness);
   642→
   643→      /* ۱) مجموع «تعداد × ست» رکوردهای پیش‌نمایش به تفکیک عرض مادر */
   644→      const previewSums = new Map();   // کلید عددی → { width, sum }
   645→      for (const row of records) {
   646→        const w = U.parseNumber(row.motherToken);
   647→        if (w === null || !(w > 0)) continue;
   648→        const key = String(Math.round(w * 100) / 100);
   649→        if (!previewSums.has(key)) previewSums.set(key, { width: w, sum: 0 });
   650→
   651→        const std = U.parseNumber(row.standardLength);
   652→        const sc = R.setCount(rules, commonThickness, std);
   653→        const setCount = sc !== null && sc > 0 ? sc : 0;
   654→        const count = row.rollCount === null ? 0 : (Number(row.rollCount) || 0);
   655→        previewSums.get(key).sum += setCount * count;
   656→      }
   657→
   658→      /* ۲) ساخت ردیف‌های کنترل برای اجتماع عرض‌های دو طرف */
   659→      const keys = new Set([...Object.keys(allowedMap), ...previewSums.keys()]);
   660→      allowedRows = [];
   661→      for (const key of keys) {
   662→        const pastedCount = Number(allowedMap[key]) || 0;
   663→        const pv = previewSums.get(key) || { width: Number(key), sum: 0 };
   664→        allowedRows.push({
   665→          key,
   666→          width: pv.width,
   667→          pastedCount,                                   // Σ (تکرار × عدد انتهایی)
   668→          previewCount: pv.sum,                          // Σ (ست × تعداد)
   669→          pasted: pastedCount * pv.width,                // «تعداد مجاز» کامل (× عرض)
   670→          preview: pv.sum * pv.width,
   671→          status: pv.sum === pastedCount ? 'ok' : (pv.sum < pastedCount ? 'less' : 'more'),
   672→        });
   673→      }
   674→      allowedRows.sort((a, b) => a.width - b.width);
   675→
   676→      /* ۳) الصاق خطا به همهٔ رکوردهای عرض‌های ناموفق */
   677→      const failed = new Map(
   678→        allowedRows.filter((r) => r.status !== 'ok').map((r) => [r.key, r])
   679→      );
   680→      if (failed.size) {
   681→        for (const row of records) {
   682→          const w = U.parseNumber(row.motherToken);
   683→          if (w === null || !(w > 0)) continue;
   684→          const info = failed.get(String(Math.round(w * 100) / 100));
   685→          if (!info) continue;
   686→          row.errors.push(
   687→            info.status === 'less'
   688→              ? `کنترل تعداد مجاز عرض ${U.faWidth(info.width)} — «تعداد مجاز» پیش‌نمایش ${U.faNum(info.preview)} کمتر از دیتای پیست‌شده ${U.faNum(info.pasted)} است — تعداد کمتر از دیتای ستاپ است.`
   689→              : `کنترل تعداد مجاز عرض ${U.faWidth(info.width)} — «تعداد مجاز» پیش‌نمایش ${U.faNum(info.preview)} بیشتر از دیتای پیست‌شده ${U.faNum(info.pasted)} است — تعداد رول‌های آن عرض بیشتر از دیتای ستاپ است.`
   690→          );
   691→        }
   692→        if (!globalErrors.allowed) {
   693→          globalErrors.allowed =
   694→            `کنترل «تعداد مجاز» برای ${U.faNum(failed.size)} عرض مادر پاس نشد — جزئیات در پنل «کنترل تعداد مجاز» و وضعیت ردیف‌ها.`;
   695→        }
   696→      }
   697→    }
   698→
   699→    const ok =
   700→      records.length > 0 &&
   701→      Object.keys(globalErrors).length === 0 &&
   702→      records.every((r) => !r.errors.length && r.record);
   703→
   704→    return {
   705→      ok, globalErrors, records, total: records.length, allowedRows,
   706→      dateInherited: !!inheritedDate,                          // نسخهٔ ۲٫۸ — تاریخ از رکورد موجود همان ستاپ به‌ارث رسیده
   707→      inheritedDateValue: inheritedDate ? inheritedDate.setupDate : null,
   708→      batchDate: batchDateStr,                                 // تاریخ مؤثر کل Batch (نرمال‌شده)
   709→    };
   710→  };

   712→  /**
   713→   * Commit پیش‌نویس — همه یا هیچ: اگر حتی یک رکورد خطا داشته باشد
   714→   * هیچ رکوردی ثبت نمی‌شود؛ در غیر این صورت همه در یک تراکنش واحد
   715→   * ذخیره می‌شوند (بدون Batch نیمه‌کاره). options.allowedMap نیز مثل
   716→   * validateBulkDraft به کنترل «تعداد مجاز» پاس داده می‌شود (تغییر جدید — ۱).
   717→   */
   718→  Setups.commitBulkDraft = async function (common, drafts, options = {}) {
   719→    const check = await Setups.validateBulkDraft(common, drafts, options);
   720→    if (!check.ok) return check;
   721→
   722→    const now = Date.now();
   723→    const ids = await db.transaction('rw', db.setupRolls, async () => {
   724→      return db.setupRolls.bulkAdd(
   725→        check.records.map((r) => ({ ...r.record, createdAt: now, updatedAt: now }))
   726→      );
   727→    });
   728→
   729→    return { ok: true, ids, count: check.records.length, records: check.records };
   730→  };
   731→
   732→  RM.setupEngine = Setups;
   733→})();
