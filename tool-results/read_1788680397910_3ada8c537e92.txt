    87→  /** ارث‌بری ضخامت (§22): اگر ستاپ موجود باشد → قفل خودکار */
    88→  SetupView.applyThicknessInheritance = async function () {
    89→    const { setupNumber, thickness, standardLength } = F();
    90→    const value = U.parseInt(setupNumber.value);
    91→    const hintEl = document.getElementById('hint-thickness');
    92→
    93→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
    94→
    95→    if (value === null || value < 0) {
    96→      SetupView.thicknessLocked = false;
    97→      thickness.disabled = false;
    98→      hintEl.hidden = true;
    99→      return;
   100→    }
   101→
   102→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   103→    const withThickness = existing.find((s) => s.thickness !== null);
   104→
   105→    if (withThickness) {
   106→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   107→      SetupView.thicknessLocked = true;
   108→      thickness.value = String(withThickness.thickness);
   109→      thickness.disabled = true;
   110→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   111→      hintEl.hidden = false;
   112→    } else {
   113→      SetupView.thicknessLocked = false;
   114→      thickness.disabled = false;
   115→      hintEl.hidden = true;
   116→    }
   117→
   118→    SetupView.refreshStandardLengthOptions(standardLength.value);
   119→  };

   121→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   122→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   123→    const { thickness, standardLength } = F();
   124→    const thicknessValue = U.parseNumber(thickness.value);
   125→    const errEl = document.getElementById('err-standard');
   126→
   127→    const options = thicknessValue === null
   128→      ? []
   129→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   130→
   131→    let html = '<option value="">— انتخاب کنید —</option>';
   132→    for (const opt of options) {
   133→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   134→    }
   135→    standardLength.innerHTML = html;
   136→
   137→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   138→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   139→      standardLength.value = keepValue;
   140→    } else {
   141→      standardLength.value = '';
   142→    }
   143→
   144→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   145→    if (thicknessValue !== null && !options.length) {
   146→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   147→      errEl.hidden = false;
   148→    } else {
   149→      errEl.textContent = '';
   150→      errEl.hidden = true;
   151→    }
   152→  };

   154→  /** بازخورد زندهٔ الگوی برش (§29) */
   155→  SetupView.updatePatternFeedback = function () {
   156→    const { width, cuttingPattern } = F();
   157→    const box = document.getElementById('pattern-feedback');
   158→
   159→    const result = Setups.validatePattern(cuttingPattern.value, U.parseNumber(width.value));
   160→
   161→    if (!cuttingPattern.value.trim()) {
   162→      box.className = 'pattern-feedback';
   163→      box.innerHTML = '<span class="muted">اختیاری — مثال: 1100-1170</span>';
   164→      return;
   165→    }
   166→
   167→    if (!result.ok) {
   168→      box.className = 'pattern-feedback invalid';
   169→      box.innerHTML = `<span class="pf-status">✕</span> ${result.message}`;
   170→      return;
   171→    }
   172→
   173→    const motherWidth = U.parseNumber(width.value);
   174→    const remaining = motherWidth !== null ? motherWidth - result.sum : null;
   175→
   176→    box.className = 'pattern-feedback valid';
   177→    box.innerHTML =
   178→      `<span class="pf-status">✓</span> مجموع: <b>${U.faWidth(result.sum)}</b>` +
   179→      (remaining !== null && remaining >= 0
   180→        ? ` · باقی‌مانده (پرت عرض): <b>${U.faWidth(remaining)}</b>`
   181→        : '') +
   182→      ` · بخش‌ها: ${U.faNum(result.parts.length)}`;
   183→  };

   189→  SetupView.submit = async function () {
   190→    const form = document.getElementById('setup-form');
   191→    const f = F();
   192→    UI.clearFormErrors(form);
   193→
   194→    const input = {
   195→      setupNumber: f.setupNumber.value,

   201→      cuttingPattern: f.cuttingPattern.value,
   202→    };
   203→

   205→    if (SetupView.editingId !== null) {
   206→      const newThickness = U.parseNumber(input.thickness);
   207→      const info = await Setups.cascadeInfo(SetupView.editingId, newThickness);
   208→
   209→      if (info.affected > 0) {
   210→        const ok = await UI.confirm({
   211→          title: 'تأیید تغییر ضخامت',
   212→          html: `
   213→            <div class="confirm-stats">
   214→              <div>تغییر ضخامت ستاپ <b>${U.faNum(info.setupNumber)}</b> باعث تغییر ضخامت
   215→              <b>${U.faNum(info.affected)}</b> رکورد دیگر نیز خواهد شد.</div>
   216→            </div>`,
   217→          message: 'این عملیات تراکنشی است و در صورت خطا به حالت قبل برمی‌گردد. ادامه می‌دهید؟',
   218→          confirmText: 'بله، ضخامت همه را تغییر بده',
   219→        });
   220→        if (!ok) {
   221→          UI.toast('عملیات ویرایش لغو شد.', 'info');
   222→          return;
   223→        }
   224→      }
   225→
   226→      const result = await Setups.update(SetupView.editingId, input, { cascadedCount: info.affected });
   227→      if (!result.ok) {
   228→        SetupView.showValidationErrors(result);
   229→        return;
   230→      }
   231→      UI.toast(
   232→        (info.affected > 0
   233→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   234→          : 'رکورد با موفقیت ویرایش شد.'),
   235→        'success'
   236→      );
   237→    } else {
   238→      /* --- ثبت جدید --- */
   239→      const result = await Setups.create(input);
   240→      if (!result.ok) {
   241→        SetupView.showValidationErrors(result);
   242→        return;
   243→      }
   244→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   245→    }
   246→
   247→    SetupView.resetForm();
   248→    await RM.refreshAll();
   249→    await SetupView.render();
   250→  };

   253→  SetupView.showValidationErrors = function (result) {
   254→    const form = document.getElementById('setup-form');
   255→    const f = F();
   256→    const fieldMap = {
   257→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   258→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   259→      width: [f.width, document.getElementById('err-width')],
   260→      thickness: [f.thickness, document.getElementById('err-thickness')],
   261→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   262→      rollCount: [f.rollCount, document.getElementById('err-count')],
   263→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   264→    };
   265→
   266→    let firstInvalid = null;
   267→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   268→      const msg = result.errors && result.errors[field];
   269→      UI.setFieldError(input, errEl, msg);
   270→      if (msg && !firstInvalid) firstInvalid = input;
   271→    }
   272→
   273→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   274→    if (firstInvalid) firstInvalid.focus();
   275→  };

   367→  /** رندر پیش‌نمایش Batch — خطاهای هر رکورد به‌صورت واضح */
   368→  SetupView.renderBulkPreview = function (result) {
   369→    const wrap = document.getElementById('bulk-preview');
   370→    const commitBtn = document.getElementById('btn-bulk-commit');
   371→    const summaryEl = document.getElementById('bulk-summary');
   372→
   373→    if (!result || (!result.records && !result.structureErrors)) {
   374→      wrap.innerHTML = '';
   375→      summaryEl.innerHTML = '';
   376→      commitBtn.disabled = true;
   377→      return;
   378→    }
   379→
   380→    const rows = (result.records || []);
   381→    const structureErrors = result.structureErrors || [];
   382→    const commonErrors = result.commonErrors || {};
   383→    const allValid = result.ok === true && rows.length > 0;
   384→
   385→    // --- خطاهای فیلدهای مشترک (یک‌بار) ---
   386→    const commonHtml = Object.entries(commonErrors).map(([field, msg]) => {
   387→      const isDuplicate = /تکراری/.test(msg);
   388→      const label = isDuplicate ? 'کلید یکتای ستاپ' : ({
   389→        setupNumber: 'شماره ستاپ', machineNumber: 'شماره دستگاه',
   390→        thickness: 'ضخامت', standardLength: 'متراژ استاندارد',
   391→      }[field] || field);
   392→      return `<div class="bulk-common-error">✕ <b>${label}:</b> ${U.escapeHtml(msg)}</div>`;
   393→    }).join('');
   394→
   395→    const structureHtml = structureErrors.map((e) =>
   396→      `<div class="bulk-common-error">✕ <b>ساختار:</b> ${U.escapeHtml(e.message)}</div>`
   397→    ).join('');
   398→
   399→    // --- ردیف‌های رکوردها (تجمیع‌شده بر اساس کلید یکتا) ---
   400→    const rowsHtml = rows.map((r, i) => {
   401→      const parsed = Setups.parsePattern(r.patternText);
   402→      const sum = parsed.ok ? parsed.sum : null;
   403→      const mother = U.parseNumber(r.motherToken);
   404→      const trim = parsed.ok && mother !== null ? mother - sum : null;
   405→      const valid = !r.errors.length;
   406→      const pairsTxt = (r.pairs && r.pairs.length > 1)
   407→        ? ` (جفت ${r.pairs.map((p) => U.faNum(p)).join('،')})`
   408→        : '';
   409→
   410→      return `
   411→        <tr class="${valid ? '' : 'bulk-row-error'}">
   412→          <td>${U.faNum(i + 1)}<span class="muted" style="font-size:.62rem">${pairsTxt}</span></td>
   413→          <td class="count-cell">${U.faWidth(mother)}</td>
   414→          <td class="ltr" dir="ltr">${U.escapeHtml(r.patternText || '—')}${(r.occurrences || 1) > 1 ? ` <span class="muted">×${U.faNum(r.occurrences)}</span>` : ''}</td>
   415→          <td>${sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>'}</td>
   416→          <td>${trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>'}</td>
   417→          <td>${r.rollCount === null ? '<span class="muted">—</span>' : U.faNum(r.rollCount)}</td>
   418→          <td>${valid
   419→            ? '<span class="status-pill pill-ok">✓ معتبر</span>'
   420→            : `<span class="status-pill pill-bad">✕ خطا</span>
   421→               <ul class="bulk-errors">${r.errors.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`}</td>
   422→        </tr>`;
   423→    }).join('');
   424→
   425→    const errorCount = rows.filter((r) => r.errors.length).length;
   426→
   427→    wrap.innerHTML = `
   428→      ${commonHtml || structureHtml ? `<div class="bulk-alert">${commonHtml}${structureHtml}</div>` : ''}
   429→      <div class="table-wrap">
   430→        <table class="data-table bulk-table">
   431→          <thead><tr>
   432→            <th>#</th><th>عرض مادر</th><th>الگوی برش</th><th>مجموع الگو</th><th>پرت عرض</th><th>تعداد رول</th><th>وضعیت</th>
   433→          </tr></thead>
   434→          <tbody>${rowsHtml || `<tr><td colspan="7" class="muted">رکوردی از دادهٔ Paste‌شده استخراج نشد.</td></tr>`}</tbody>
   435→        </table>
   436→      </div>`;
   437→
   438→    summaryEl.innerHTML = `
   439→      ${U.faNum(rows.length)} رکورد استخراج‌شده
   440→      ${allValid
   441→        ? ' — <span class="pill-ok status-pill">همه معتبر — آمادهٔ ثبت یکجا</span>'
   442→        : ` — <span class="pill-bad status-pill">${U.faNum(errorCount)} رکورد خطا</span>
   443→           ${commonHtml || structureHtml ? ' · خطاهای ساختاری/مشترک' : ''}`}
   444→      ${allValid ? '' : ' — تا رفع همهٔ خطاها، ثبت کل Batch انجام نمی‌شود.'}
   445→    `;
   446→
   447→    commitBtn.disabled = !allValid;
   448→  };

   450→  /** Commit نهایی — یک تراکنش واحد؛ در صورت خطا هیچ رکوردی ثبت نمی‌شود */
   451→  SetupView.commitBulk = async function () {
   452→    const text = SetupView._bulkText || document.getElementById('bulk-text').value;
   453→    if (!text.trim()) return;
   454→
   455→    // اعتبارسنجی مجدد (fresh) + Commit — همه یا هیچ
   456→    const result = await Setups.createBulk(SetupView.bulkCommonInput(), text);
   457→
   458→    if (!result.ok) {
   459→      SetupView.renderBulkPreview(result);
   460→      UI.toast('Batch ثبت نشد — خطاها را در پیش‌نمایش برطرف کنید (هیچ رکوردی ذخیره نشده است).', 'error', 6000);
   461→      return;
   462→    }
   463→
   464→    UI.toast(
   465→      `${U.faNum(result.count)} رکورد ستاپ در یک تراکنش واحد ثبت شد (Bulk Paste).`,
   466→      'success',
   467→      5000
   468→    );
   469→    SetupView.resetBulk();
   470→    await RM.refreshAll();
   471→    await SetupView.render();
   472→  };

   523→  /** کارت‌های الگوی برش (§30) */
   524→  SetupView.renderPatternCards = function (setup) {
   525→    const parsed = Setups.parsePattern(setup.cuttingPattern);
   526→    if (!setup.cuttingPattern) return '<span class="muted">—</span>';
   527→    if (!parsed.ok) return `<span class="film-bad">${U.escapeHtml(setup.cuttingPattern)}</span>`;
   528→
   529→    const cards = parsed.parts
   530→      .map((p) => `<span class="pattern-card">${U.faWidth(p)}</span>`)
   531→      .join('');
   532→
   533→    const trim = setup.width - parsed.sum;
   534→    return `
   535→      <div class="pattern-cards">${cards}</div>
   536→      <div class="pattern-sum">
   537→        Σ ${U.faWidth(parsed.sum)}
   538→        ${trim > 0 ? ` · پرت عرض: ${U.faWidth(trim)}` : ''}
   539→      </div>`;
   540→  };

   619→  SetupView.showAudit = function (id, state) {
   620→    const r = state.setupReport.find((x) => x.id === id);
   621→    if (!r) return;
   622→
   623→    // یافتن سایر ستاپ‌های دارای همان کلید متالایز برای گزارش تخصیص
   624→    const metPeers = state.setupReport.filter(
   625→      (x) => x.id !== id && x.audit.metGroup && r.audit.metGroup && x.audit.metGroup.key === r.audit.metGroup.key
   626→    );
   627→
   628→    const peersHtml = metPeers.length
   629→      ? `<p class="muted" style="font-size:.78rem">سایر دستگاه‌های همین کلید متالایز:</p>
   630→         <ul class="audit-peers">${metPeers.map((p) =>
   631→          `<li>دستگاه ${U.faNum(p.machineNumber)}: ظرفیت ${U.faNum(p.audit.metCapacity)} — تخصیص ${U.faNum(p.metallized)}</li>`
   632→        ).join('')}</ul>`
   633→      : '';
   634→
   635→    const rawGroupInfo = r.audit.rawGroup
   636→      ? `${U.faNum(r.audit.rawGroup.count)} رول خام برای کلید <code>${U.escapeHtml(r.audit.rawGroup.key)}</code> شناسایی شد؛
   637→         ${U.faNum(r.rawExisting)} رول تا سقف ظرفیت این ستاپ تخصیص یافت.`
   638→      : 'هیچ گروه رول خام منطبقی (عرض + متراژ استاندارد + ستاپ) یافت نشد.';
   639→
   640→    const metGroupInfo = r.audit.metGroup
   641→      ? `${U.faNum(r.audit.metGroup.count)} رول متالایز یونیک برای کلید <code>${U.escapeHtml(r.audit.metGroup.key)}</code> شناسایی شد؛
   642→         ${U.faNum(r.metallized)} رول به این دستگاه تخصیص یافت.`
   643→      : 'هیچ رول متالایز منطبقی (عرض + ستاپ) یافت نشد.';
   644→
   645→    UI.infoModal(
   646→      `جزئیات محاسبه — ستاپ ${U.faNum(r.setupNumber)} / دستگاه ${U.faNum(r.machineNumber)}`,
   647→      `
   648→      <div class="audit-stats">
   649→        <div>تعداد ستاپ: <b>${U.faNum(r.rollCount)}</b></div>
   650→        <div>خام موجود: <b>${U.faNum(r.rawExisting)}</b></div>
   651→        <div>متالایز: <b>${U.faNum(r.metallized)}</b></div>
   652→        <div>برش‌نشده: <b>${U.faNum(r.uncut)}</b></div>
   653→        ${r.conflict > 0 ? `<div class="audit-conflict">⚠ مغایرت ظرفیت: <b>${U.faNum(r.conflict)} رول</b><br>
   654→          <span style="font-size:.75rem">${U.faNum(r.rawExisting)} + ${U.faNum(r.metallized)} > ظرفیت ${U.faNum(r.rollCount)}</span></div>` : ''}
   655→      </div>
   656→      <div class="audit-section"><h4>ریشهٔ رول‌های خام (کلید ۳فاکتوری)</h4><p>${rawGroupInfo}</p></div>
   657→      <div class="audit-section"><h4>ریشهٔ متالایزها (کلید ۲فاکتوری)</h4><p>${metGroupInfo}</p></div>
   658→      ${peersHtml}
   659→      <div class="audit-formula">
   660→        برش‌نشده = ${U.faNum(r.rollCount)} − (${U.faNum(r.rawExisting)} + ${U.faNum(r.metallized)}) =
   661→        <b>${U.faNum(r.uncutCalculated)}</b>${r.conflict > 0 ? ` → نمایش ${U.faNum(r.uncut)} + مغایرت ${U.faNum(r.conflict)}` : ''}
   662→      </div>
   663→      `
   664→    );
   665→  };
