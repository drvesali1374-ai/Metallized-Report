   156→   156→  /**
   157→   157→   * رندر جدول صفحه‌بندی‌شده:
   158→   158→   * @param wrap       عنصر نگهدارندهٔ جدول
   159→   159→   * @param columns    [{key, label, render?, className?, sortable?}]
   160→   160→   * @param rows       داده‌ها
   161→   161→   * @param opts       { page, pageSize, emptyText, footerInfo }
   162→   162→   */
   163→   163→  UI.renderTable = function (wrap, columns, rows, opts = {}) {
   164→   164→    const pageSize = opts.pageSize || RM.config.PAGE_SIZE;
   165→   165→    const page = opts.page || 1;
   166→   166→    const emptyText = opts.emptyText || 'داده‌ای برای نمایش وجود ندارد.';
   167→   167→
   168→   168→    if (!rows.length) {
   169→   169→      wrap.innerHTML = `
   170→   170→        <div class="empty-state">
   171→   171→          ${UI.icons.alert}
   172→   172→          <p>${emptyText}</p>
   173→   173→        </div>`;
   174→   174→      return { page: 1, pages: 0 };
   175→   175→    }
   176→   176→
   177→   177→    const pages = Math.max(1, Math.ceil(rows.length / pageSize));
   178→   178→    const safePage = Math.min(page, pages);
   179→   179→    const slice = rows.slice((safePage - 1) * pageSize, safePage * pageSize);
   180→   180→
   181→   181→    const thead = '<thead><tr>' +
   182→   182→      columns.map((c) => `<th>${c.label}</th>`).join('') +
   183→   183→      '</tr></thead>';
   184→   184→
   185→   185→    const tbody = '<tbody>' + slice.map((row) => {
   186→   186→      const cells = columns.map((c) => {
   187→   187→        const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   188→   188→        return `<td class="${c.className || ''}">${content}</td>`;
   189→   189→      }).join('');
   190→   190→      return `<tr>${cells}</tr>`;
   191→   191→    }).join('') + '</tbody>';
   192→   192→
   193→   193→    // نوار صفحه‌بندی (اگر بیش از یک صفحه)
   194→   194→    const pager = pages > 1
   195→   195→      ? `<div class="pager" role="navigation" aria-label="صفحه‌بندی جدول">
   196→   196→           <button type="button" class="btn btn-ghost btn-sm" data-page="1" ${safePage === 1 ? 'disabled' : ''}>ابتدا</button>
   197→   197→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage - 1}" ${safePage === 1 ? 'disabled' : ''}>قبلی</button>
   198→   198→           <span class="pager-info">صفحهٔ ${U.faNum(safePage)} از ${U.faNum(pages)} — ${U.faNum(rows.length)} ردیف</span>
   199→   199→           <button type="button" class="btn btn-ghost btn-sm" data-page="${safePage + 1}" ${safePage === pages ? 'disabled' : ''}>بعدی</button>
   200→   200→           <button type="button" class="btn btn-ghost btn-sm" data-page="${pages}" ${safePage === pages ? 'disabled' : ''}>انتها</button>
   201→   201→         </div>`
   202→   202→      : (opts.footerInfo ? `<div class="pager"><span class="pager-info">${opts.footerInfo(rows.length)}</span></div>` : '');
   203→   203→
   204→   204→    wrap.innerHTML = `
   205→   205→      <div class="table-wrap table-scroll">
   206→   206→        <table class="data-table">${thead}${tbody}</table>
   207→   207→      </div>
   208→   208→      ${pager}`;
   209→   209→
   210→   210→    // رویداد صفحه‌بندی
   211→   211→    wrap.querySelectorAll('[data-page]').forEach((btn) => {
   212→   212→      btn.addEventListener('click', () => {
   213→   213→        const nextPage = Number(btn.dataset.page);
   214→   214→        if (opts.onPageChange) opts.onPageChange(nextPage);
   215→   215→      });
   216→   216→    });
   217→   217→
   218→   218→    return { page: safePage, pages };
   219→   219→  };

   395→   395→  /**
   396→   396→   * نرمال‌سازی متن جست‌وجو — ارقام فارسی/عربی → لاتین + حذف جداکننده‌های
   397→   397→   * هزارگان و فاصله‌ها. هم روی «مقدار جست‌وجو» و هم روی «متن سلول» اعمال
   398→   398→   * می‌شود تا «۸» با 8 و «۲۳» با 23 هم‌ارزش شود.
   399→   399→   */
   400→   400→  UI.filterNormalize = function (text) {
   401→   401→    return U.toLatinDigits(String(text ?? ''))
   402→   402→      .replace(/[,\u066C\u2019\u066B\s]/g, '')
   403→   403→      .trim()
   404→   404→      .toLowerCase();
   405→   405→  };
   406→   406→
   407→   407→  /**
   408→   408→   * منطق تطبیق «شامل بودن» (نسخهٔ ۲٫۳):
   409→   409→   *   - عدد و متن هر دو با «شامل بودن» فیلتر می‌شوند (نه تطابق دقیق)
   410→   410→   *   - invert = true → برعکس: «شامل نشود»
   411→   411→   * @param {string[]} tokens   متن‌های قابل‌جست‌وجوی سلول (نمایش + مقدار خام)
   412→   412→   * @param {string}   value    مقدار واردشده در باکس فیلتر
   413→   413→   * @param {boolean}  invert   منطق برعکس فعال است؟
   414→   414→   */
   415→   415→  UI.filterContains = function (tokens, value, invert) {
   416→   416→    const needle = UI.filterNormalize(value);
   417→   417→    if (needle === '') return true;      // فیلتر خالی → همه
   418→   418→    const hay = (tokens || [])
   419→   419→      .map((t) => UI.filterNormalize(t))
   420→   420→      .filter(Boolean)
   421→   421→      .join(' ');
   422→   422→    const contains = hay.includes(needle);
   423→   423→    return invert ? !contains : contains;
   424→   424→  };

   426→   426→  /** خواندن/ساخت وضعیت فیلتر یک ستون ({ value, invert }) از state */
   427→   427→  UI.filterStateOf = function (state, key) {
   428→   428→    if (!state.filters[key] || typeof state.filters[key] !== 'object') {
   429→   429→      state.filters[key] = { value: '', invert: false };
   430→   430→    }
   431→   431→    return state.filters[key];
   432→   432→  };

   437→   437→  /**
   438→   438→   * جدول بلند (حداقل ۵۰ رکورد) + فیلتر پیشرفتهٔ زیر سرستون‌ها — مولفهٔ
   439→   439→   * مشترک جدول‌های «رول‌های برش‌خورده»، «رول‌های خام» و «گزارش ستاپ‌ها».
   440→   440→   *
   441→   441→   * منطق نسخهٔ ۲٫۳:
   442→   442→   *   - فیلتر همهٔ ستون‌ها (عدد و متن) با «شامل بودن» + ارقام فارسی/لاتین
   443→   443→   *   - دکمهٔ کوچک «≠» کنار هر باکس → سوییچ «شامل نشود»

   445→   445→   *     سرستون/باکس‌ها بازسازی نمی‌شوند و فوکوس جابه‌جا نمی‌شود)
   446→   446→   *   - سرستون + ردیف فیلتر چسبان هنگام اسکرول عمودی
   447→   447→   *
   448→   448→   * منطق نسخهٔ ۲٫۴:
   449→   449→   *   - باکس «ردیف‌های قابل نمایش» در نوار پایین → ارتفاع جدول داینامیک
   450→   450→   *     (فقط استایل ارتفاع به‌روز می‌شود؛ جدول/فیلترها/فوکوس دست‌نخورده)
   451→   451→   *   - پشتیبانی عرض ستون (c.width → colgroup با table-layout:fixed)
   452→   452→   *
   453→   453→   * منطق نسخهٔ ۲٫۵:
   454→   454→   *   - Wrap Text هدرها: عنوان ستون‌ها با توجه به عرض ستون چندخط می‌شود؛
   455→   455→   *     ارتفاع ردیف سرستون اندازه‌گیری و در متغیر --ft-head-h ثبت می‌شود تا
   456→   456→   *     ردیف فیلتر چسبان دقیقاً زیر آن بنشیند (ResizeObserver → هم‌گام با تغییر عرض)
   457→   457→   *   - چیدمان باکس فیلتر: آیکون «≠» کوچک در سمت راست + باکس فیلتر در ادامه
   458→   458→   *     که عرضش با عرض ستون تنظیم می‌شود (min-width: 0 → هرگز ناقص نمی‌شود)
   459→   459→   *
   460→   460→   * @param {HTMLElement} wrap     نگهدارندهٔ جدول
   461→   461→   * @param {Array}  columns       [{key, label, render(row)→html, tokens(row)→string[],
   462→   462→   *                                className?, filterable? (پیش‌فرض true), width? (px — نسخهٔ ۲٫۴)}]

   465→   465→   *                                visibleRows, rowHeight, tableClass, emptyText,

   467→   467→   *                                onRowsChange?(n) (ذخیرهٔ ماندگار),

   469→   469→   *                                onRendered(visibleRows) }
   470→   470→   */
   471→   471→  UI.filterTable = function (wrap, columns, rows, opts = {}) {

   474→   474→    state.filters = state.filters || {};
   475→   475→    // تعداد ردیف قابل نمایش (نسخهٔ ۲٫۴): اولویت با state کاربر (ماندگار)، بعد opts

   499→   499→                  aria-pressed="${String(!!f.invert)}"

   501→   501→                  aria-label="برعکس‌کردن منطق فیلتر ستون ${U.escapeHtml(String(c.label))}">
   502→   502→            ${UI.icons.notContains}

   523→   523→        ? visible.map((row) => {

   525→   525→              const content = c.render ? c.render(row) : U.escapeHtml(row[c.key] ?? '—');
   526→   526→              return `<td class="${c.className || ''}">${content}</td>`;
   527→   527→            }).join('');
   528→   528→            return `<tr>${cells}</tr>`;
   529→   529→          }).join('')
   530→   530→        : `<tr><td colspan="${columns.length}" class="muted" style="text-align:center;padding:18px">${

   532→   532→              ? (opts.filteredEmptyText || 'هیچ رکوردی با فیلترهای فعلی مطابقت ندارد — فیلترها را پاک کنید.')

   535→   535→
   536→   536→      if (statusEl) {
   537→   537→        statusEl.innerHTML = activeCount
   538→   538→          ? `<span class="chip chip-strong">${U.faNum(visible.length)} از ${U.faNum(rows.length)} رکورد (فیلترشده)</span>`
   539→   539→          : `<span class="chip">${U.faNum(rows.length)} رکورد</span>`;
   540→   540→      }
   541→   541→
   542→   542→      if (typeof opts.onRendered === 'function') opts.onRendered(visible, rows);
   543→   543→      return visible;
   544→   544→    };
   545→   545→
   546→   546→    /* ---------- اسکلت جدول (یک‌بار) ---------- */
   547→   547→    if (!rows.length && !Object.values(state.filters).some((f) => f && String(f.value ?? '').trim() !== '')) {

   550→   550→          ${UI.icons.alert}

   644→   644→        f.value = '';
   645→   645→        f.invert = false;
   646→   646→      }
   647→   647→      wrap.querySelectorAll('[data-ft-filter]').forEach((el) => { el.value = ''; });
   648→   648→      wrap.querySelectorAll('[data-ft-invert]').forEach((b) => {
   649→   649→        b.classList.remove('on');
   650→   650→        b.setAttribute('aria-pressed', 'false');
   651→   651→        b.title = 'برعکس‌کردن منطق: شامل نشود';
   652→   652→      });
   653→   653→      renderBody();
   654→   654→      UI.toast('فیلترها پاک شد.', 'info');
   655→   655→    });
   656→   656→
   657→   657→    // باکس «ردیف‌های قابل نمایش» (نسخهٔ ۲٫۴) — فقط ارتفاع اسکرول تغییر می‌کند؛
   658→   658→    // جدول/فیلترها/فوکوس دست‌نخورده می‌مانند. ماندگاری با onRowsChange.
   659→   659→    const rowsInput = wrap.querySelector('[data-ft-rows]');

   681→   681→  /* ---------------- پنل ترتیب ستون‌ها (مشترک — چیپ‌های کشیدنی + باکس عرض) ---------------- */
   682→   682→
   683→   683→  /**
   684→   684→   * پنل «ترتیب ستون‌ها» — نسخهٔ ۲٫۴:
   685→   685→   *   - ترتیب با کشیدن‌ورها (drag) تغییر می‌کند (فلش‌های ↑/↓ حذف شدند)
   686→   686→   *   - بجای فلش‌ها، هر ستون یک باکس عددی «عرض (px)» دارد؛ کاربر عرض دلخواه
   687→   687→   *     هر ستون را وارد می‌کند (صفر یا خالی → عرض خودکار)
   688→   688→   * برای جدول «گزارش تولید ستاپ‌ها».
   689→   689→   * @param {HTMLElement} wrap      عنصر نگهدارندهٔ چیپ‌ها
   690→   690→   * @param {Array}        items    [{key, label, width}] به ترتیب فعلی
   691→   691→   * @param {object}       handlers { onMove(from, to), onWidth(key, width), emptyText? }
   692→   692→   */
   693→   693→  UI.orderPanel = function (wrap, items, handlers) {
   694→   694→    if (!wrap) return;
   695→   695→    wrap.innerHTML = items.map((it, i) => `
   696→   696→      <div class="sort-chip order-chip" draggable="true" data-index="${i}" title="برای تغییر ترتیب بکشید">
   697→   697→        <span class="grip">${UI.icons.grip}</span>
   698→   698→        <span class="sort-priority">${U.faNum(i + 1)}</span>
   699→   699→        <span class="sort-label">${U.escapeHtml(it.label)}</span>
   700→   700→        <label class="order-width" title="عرض این ستون در جدول (پیکسل) — صفر یا خالی = خودکار">
   701→   701→          <span class="order-width-unit">px</span>
   702→   702→          <input type="number" class="order-width-input" data-order-width="${U.escapeHtml(it.key)}"
   703→   703→                 min="0" max="1200" step="1" inputmode="numeric" aria-label="عرض ستون ${U.escapeHtml(it.label)}"
   704→   704→                 value="${it.width != null ? Number(it.width) : ''}" placeholder="خودکار">
   705→   705→        </label>
   706→   706→      </div>`).join('') ||
   707→   707→      `<span class="muted" style="font-size:.78rem">${U.escapeHtml(handlers.emptyText || 'بدون ستون')}</span>`;
   708→   708→
   709→   709→    // باکس عرض هر ستون (نسخهٔ ۲٫۴) — debounce سبک؛ خالی/۰ = عرض خودکار
   710→   710→    let widthTimer = null;
   711→   711→    wrap.querySelectorAll('[data-order-width]').forEach((inp) => {
   712→   712→      inp.addEventListener('input', () => {
   713→   713→        clearTimeout(widthTimer);
   714→   714→        const key = inp.dataset.orderWidth;
   715→   715→        widthTimer = setTimeout(() => {
   716→   716→          const v = U.parseInt(inp.value);
   717→   717→          handlers.onWidth(key, v === null ? 0 : Math.min(Math.max(v, 0), 1200));
   718→   718→        }, 320);
   719→   719→      });
   720→   720→    });
   721→   721→
   722→   722→    // کشیدن و رها کردن
   723→   723→    let dragSrcIndex = null;
   724→   724→    wrap.querySelectorAll('.order-chip').forEach((chip) => {
   725→   725→      chip.addEventListener('dragstart', (e) => {
   726→   726→        dragSrcIndex = Number(chip.dataset.index);
   727→   727→        chip.classList.add('dragging');
   728→   728→        e.dataTransfer.effectAllowed = 'move';
   729→   729→        e.dataTransfer.setData('text/plain', String(dragSrcIndex));
   730→   730→      });
   731→   731→      chip.addEventListener('dragend', () => {
   732→   732→        chip.classList.remove('dragging');
   733→   733→        wrap.querySelectorAll('.order-chip').forEach((c) => c.classList.remove('drag-over'));
   734→   734→        dragSrcIndex = null;
   735→   735→      });
   736→   736→      chip.addEventListener('dragover', (e) => {
   737→   737→        e.preventDefault();
   738→   738→        e.dataTransfer.dropEffect = 'move';
   739→   739→        if (!chip.classList.contains('drag-over')) chip.classList.add('drag-over');
   740→   740→      });
   741→   741→      chip.addEventListener('dragleave', () => chip.classList.remove('drag-over'));
   742→   742→      chip.addEventListener('drop', (e) => {
   743→   743→        e.preventDefault();
   744→   744→        const from = dragSrcIndex;
   745→   745→        const to = Number(chip.dataset.index);
   746→   746→        if (from === null || from === to) return;
   747→   747→        handlers.onMove(from, to);
   748→   748→      });
   749→   749→    });
   750→   750→  };
   751→   751→
   752→   752→  RM.ui = UI;
   753→   753→})();
