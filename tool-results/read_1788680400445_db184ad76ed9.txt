   275→   275→  };

   368→   368→  SetupView.renderBulkPreview = function (result) {
   369→   369→    const wrap = document.getElementById('bulk-preview');
   370→   370→    const commitBtn = document.getElementById('btn-bulk-commit');
   371→   371→    const summaryEl = document.getElementById('bulk-summary');
   372→   372→
   373→   373→    if (!result || (!result.records && !result.structureErrors)) {
   374→   374→      wrap.innerHTML = '';
   375→   375→      summaryEl.innerHTML = '';
   376→   376→      commitBtn.disabled = true;
   377→   377→      return;
   378→   378→    }
   379→   379→
   380→   380→    const rows = (result.records || []);
   381→   381→    const structureErrors = result.structureErrors || [];
   382→   382→    const commonErrors = result.commonErrors || {};
   383→   383→    const allValid = result.ok === true && rows.length > 0;
   384→   384→
   385→   385→    // --- خطاهای فیلدهای مشترک (یک‌بار) ---
   386→   386→    const commonHtml = Object.entries(commonErrors).map(([field, msg]) => {
   387→   387→      const isDuplicate = /تکراری/.test(msg);
   388→   388→      const label = isDuplicate ? 'کلید یکتای ستاپ' : ({
   389→   389→        setupNumber: 'شماره ستاپ', machineNumber: 'شماره دستگاه',
   390→   390→        thickness: 'ضخامت', standardLength: 'متراژ استاندارد',
   391→   391→      }[field] || field);
   392→   392→      return `<div class="bulk-common-error">✕ <b>${label}:</b> ${U.escapeHtml(msg)}</div>`;
   393→   393→    }).join('');
   394→   394→
   395→   395→    const structureHtml = structureErrors.map((e) =>
   396→   396→      `<div class="bulk-common-error">✕ <b>ساختار:</b> ${U.escapeHtml(e.message)}</div>`
   397→   397→    ).join('');
