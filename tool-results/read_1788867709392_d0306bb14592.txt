   130→  SetupView.applyThicknessInheritance = async function () {
   131→    const { setupNumber, thickness, standardLength } = F();
   132→    const value = U.parseInt(setupNumber.value);
   133→    const hintEl = document.getElementById('hint-thickness');
   134→
   135→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   136→
   137→    if (value === null || value < 0) {
   138→      SetupView.thicknessLocked = false;
   139→      thickness.disabled = false;
   140→      hintEl.hidden = true;
   141→      return;
   142→    }
   143→
   144→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   145→    const withThickness = existing.find((s) => s.thickness !== null);
   146→
   147→    if (withThickness) {
   148→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   149→      SetupView.thicknessLocked = true;
   150→      thickness.value = String(withThickness.thickness);
   151→      thickness.disabled = true;
   152→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   153→      hintEl.hidden = false;
   154→    } else {
   155→      SetupView.thicknessLocked = false;
   156→      thickness.disabled = false;
   157→      hintEl.hidden = true;
   158→    }
   159→
   160→    SetupView.refreshStandardLengthOptions(standardLength.value);
   161→  };

   163→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   164→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   165→    const { thickness, standardLength } = F();
   166→    const thicknessValue = U.parseNumber(thickness.value);
   167→    const errEl = document.getElementById('err-standard');
   168→
   169→    const options = thicknessValue === null
   170→      ? []
   171→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   172→
   173→    let html = '<option value="">— انتخاب کنید —</option>';
   174→    for (const opt of options) {
   175→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   176→    }
   177→    standardLength.innerHTML = html;
   178→
   179→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   180→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   181→      standardLength.value = keepValue;
   182→    } else {
   183→      standardLength.value = '';
   184→    }
   185→
   186→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   187→    if (thicknessValue !== null && !options.length) {
   188→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   189→      errEl.hidden = false;
   190→    } else {
   191→      errEl.textContent = '';
   192→      errEl.hidden = true;
   193→    }
   194→  };

   225→  };

   260→          confirmText: 'بله، ضخامت همه را تغییر بده',

   263→          UI.toast('عملیات ویرایش لغو شد.', 'info');
   264→          return;
   265→        }
   266→      }
   267→
   268→      const result = await Setups.update(SetupView.editingId, input, { cascadedCount: info.affected });
   269→      if (!result.ok) {
   270→        SetupView.showValidationErrors(result);
   271→        return;
   272→      }
   273→      UI.toast(
   274→        (info.affected > 0
   275→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   276→          : 'رکورد با موفقیت ویرایش شد.'),
   277→        'success'
   278→      );
   279→    } else {
   280→      /* --- ثبت جدید --- */
   281→      const result = await Setups.create(input);
   282→      if (!result.ok) {
   283→        SetupView.showValidationErrors(result);
   284→        return;
   285→      }
   286→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   287→    }
   288→
   289→    SetupView.resetForm();
   290→    await RM.refreshAll();
   291→    await SetupView.render();
   292→  };

   294→  /** نمایش خطاهای Field-Level (§69): حاشیهٔ قرمز + پیام + فوکوس */
   295→  SetupView.showValidationErrors = function (result) {
   296→    const form = document.getElementById('setup-form');
   297→    const f = F();
   298→    const fieldMap = {
   299→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   300→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   301→      width: [f.width, document.getElementById('err-width')],
   302→      thickness: [f.thickness, document.getElementById('err-thickness')],
   303→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   304→      rollCount: [f.rollCount, document.getElementById('err-count')],
   305→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   306→    };
   307→
   308→    let firstInvalid = null;
   309→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   310→      const msg = result.errors && result.errors[field];
   311→      UI.setFieldError(input, errEl, msg);
   312→      if (msg && !firstInvalid) firstInvalid = input;
   313→    }
   314→
   315→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   316→    if (firstInvalid) firstInvalid.focus();
   317→  };

   434→  /**
   435→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   436→   * بلافاصله به‌روز می‌شود (روش محاسبهٔ جدید):
   437→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   438→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   439→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست همان)
   440→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   441→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه):
   442→   *   ۱۶ / ۳ / ۴۸ / ۵
   443→   */
   444→  SetupView.bulkSummaryStats = function (drafts) {
   445→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   446→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست) — «جمع عرض الگوی برش»
   447→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   448→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   449→
   450→    for (const d of drafts) {
   451→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   452→      motherRollsTotal += rollCount;
   453→
   454→      const w = U.parseNumber(d.motherToken);
   455→      if (w !== null) motherWidthSet.add(w);
   456→
   457→      const parsed = Setups.parsePattern(d.patternText);
   458→      if (parsed.ok && parsed.parts.length) {
   459→        patternWidthsTotal += parsed.parts.length * rollCount;
   460→        for (const part of parsed.parts) patternWidthSet.add(part);
   461→      }
   462→    }
   463→
   464→    return {
   465→      count: drafts.length,
   466→      motherRollsTotal,                            // ۱۶
   467→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   468→      patternWidthsTotal,                          // ۴۸
   469→      uniquePatternWidths: patternWidthSet.size,   // ۵
   470→    };
   471→  };

   661→  SetupView.updateBulkRowComputed = function (rowIdx = null) {
   662→    const b = SetupView._bulk;
   663→    if (!b) return;
   664→    const rowsIdx = rowIdx === null ? b.drafts.map((_, i) => i) : [rowIdx];
   665→
   666→    for (const i of rowsIdx) {
   667→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   668→      if (!tr) continue;
   669→      const d = b.drafts[i];
   670→
   671→      const parsed = Setups.parsePattern(d.patternText);
   672→      const mother = U.parseNumber(d.motherToken);
   673→      const sum = parsed.ok ? parsed.sum : null;
   674→      const trim = parsed.ok && mother !== null ? mother - sum : null;
   675→
   676→      tr.querySelector('[data-bulk-cell="sum"]').innerHTML =
   677→        sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>';
   678→      tr.querySelector('[data-bulk-cell="trim"]').innerHTML =
   679→        trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>';
   680→    }

   716→  SetupView.revalidateBulk = async function () {

   723→  /** اعمال نتیجهٔ اعتبارسنجی روی UI پیش‌نمایش (خطاهای سراسری + وضعیت ردیف‌ها) */
   724→  SetupView.applyBulkCheck = function (check) {
   725→    SetupView._bulkCheck = check;
   726→    const b = SetupView._bulk;
   727→    if (!b) return;
   728→
   729→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   730→    const globalMap = {
   731→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   732→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   733→    };
   734→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {
   735→      const errEl = document.getElementById(errId);
   736→      const inputEl = document.getElementById(inputId);
   737→      const msg = check.globalErrors[field] || '';
   738→      if (errEl) { errEl.textContent = msg; errEl.hidden = !msg; }
   739→      if (inputEl) inputEl.classList.toggle('invalid', !!msg);
   740→    }
   741→
   742→    // وضعیت هر ردیف
   743→    b.drafts.forEach((d, i) => {
   744→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   745→      if (!tr) return;
   746→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   747→      const rec = check.records[i];
   748→      const errs = rec ? rec.errors : [];
   749→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   750→      cell.innerHTML = errs.length
   751→        ? `<span class="status-pill pill-bad">✕ خطا</span>
   752→           <ul class="bulk-errors">${errs.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`
   753→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   754→    });
   755→
   756→    const commitBtn = document.getElementById('btn-bulk-commit');
   757→    const hasStructure = b.structureErrors.length > 0;
   758→    commitBtn.disabled = !(check.ok === true && !hasStructure);
   759→  };

   774→    UI.toast(
   775→      `${U.faNum(result.count)} رکورد ستاپ در یک تراکنش واحد ثبت شد (Bulk Paste).`,

   926→  SetupView.applySetupSort = function (rows) {
   927→    const sorted = [...rows];
   928→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
   929→      const { field, direction } = SetupView.sortPipeline[i];
   930→      sorted.sort((a, b) => {
   931→        if (field === 'cuttingPattern') {
   932→          const ka = Setups.normalizePatternKey(a.cuttingPattern) || '';
   933→          const kb = Setups.normalizePatternKey(b.cuttingPattern) || '';
   934→          const r = ka.localeCompare(kb, 'fa', { numeric: true });
   935→          return direction === 'desc' ? -r : r;
   936→        }
   937→        return U.compareNumeric(a[field], b[field], direction);
   938→      });
   939→    }
   940→    return sorted;
   941→  };
