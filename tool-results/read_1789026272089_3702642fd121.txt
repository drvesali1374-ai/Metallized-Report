   141→  /** ارث‌بری ضخامت (§22): اگر ستاپ موجود باشد → قفل خودکار */
   142→  SetupView.applyThicknessInheritance = async function () {
   143→    const { setupNumber, thickness, standardLength } = F();
   144→    const value = U.parseInt(setupNumber.value);
   145→    const hintEl = document.getElementById('hint-thickness');
   146→
   147→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   148→
   149→    if (value === null || value < 0) {
   150→      SetupView.thicknessLocked = false;
   151→      thickness.disabled = false;
   152→      hintEl.hidden = true;
   153→      return;
   154→    }
   155→
   156→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   157→    const withThickness = existing.find((s) => s.thickness !== null);
   158→
   159→    if (withThickness) {
   160→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   161→      SetupView.thicknessLocked = true;
   162→      thickness.value = String(withThickness.thickness);
   163→      thickness.disabled = true;
   164→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   165→      hintEl.hidden = false;
   166→    } else {
   167→      SetupView.thicknessLocked = false;
   168→      thickness.disabled = false;
   169→      hintEl.hidden = true;
   170→    }
   171→
   172→    SetupView.refreshStandardLengthOptions(standardLength.value);
   173→  };

   175→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   176→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   177→    const { thickness, standardLength } = F();
   178→    const thicknessValue = U.parseNumber(thickness.value);
   179→    const errEl = document.getElementById('err-standard');
   180→
   181→    const options = thicknessValue === null
   182→      ? []
   183→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   184→
   185→    let html = '<option value="">— انتخاب کنید —</option>';
   186→    for (const opt of options) {
   187→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   188→    }
   189→    standardLength.innerHTML = html;
   190→
   191→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   192→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   193→      standardLength.value = keepValue;
   194→    } else {
   195→      standardLength.value = '';
   196→    }
   197→
   198→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   199→    if (thicknessValue !== null && !options.length) {
   200→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   201→      errEl.hidden = false;
   202→    } else {
   203→      errEl.textContent = '';
   204→      errEl.hidden = true;
   205→    }
   206→  };

   209→  SetupView.updatePatternFeedback = function () {
   210→    const { width, cuttingPattern } = F();
   211→    const box = document.getElementById('pattern-feedback');
   212→
   213→    const result = Setups.validatePattern(cuttingPattern.value, U.parseNumber(width.value));
   214→

   216→      box.className = 'pattern-feedback';
   217→      box.innerHTML = '<span class="muted">اختیاری — مثال: 1100-1170</span>';
   218→      return;
   219→    }
   220→
   221→    if (!result.ok) {
   222→      box.className = 'pattern-feedback invalid';
   223→      box.innerHTML = `<span class="pf-status">✕</span> ${result.message}`;
   224→      return;
   225→    }
   226→
   227→    const motherWidth = U.parseNumber(width.value);
   228→    const remaining = motherWidth !== null ? motherWidth - result.sum : null;
   229→
   230→    box.className = 'pattern-feedback valid';
   231→    box.innerHTML =
   232→      `<span class="pf-status">✓</span> مجموع: <b>${U.faWidth(result.sum)}</b>` +
   233→      (remaining !== null && remaining >= 0
   234→        ? ` · باقی‌مانده (پرت عرض): <b>${U.faWidth(remaining)}</b>`
   235→        : '') +
   236→      ` · بخش‌ها: ${U.faNum(result.parts.length)}`;
   237→  };

   243→  SetupView.submit = async function () {
   244→    const form = document.getElementById('setup-form');
   245→    const f = F();
   246→    UI.clearFormErrors(form);
   247→
   248→    const input = {
   249→      setupNumber: f.setupNumber.value,
   250→      machineNumber: f.machineNumber.value,
   251→      width: f.width.value,
   252→      thickness: f.thickness.value,
   253→      standardLength: f.standardLength.value,
   254→      rollCount: f.rollCount.value,
   255→      cuttingPattern: f.cuttingPattern.value,
   256→    };
   257→
   258→    /* --- ویرایش: بررسی آبشار ضخامت (§23) --- */
   259→    if (SetupView.editingId !== null) {
   260→      const newThickness = U.parseNumber(input.thickness);
   261→      const info = await Setups.cascadeInfo(SetupView.editingId, newThickness);
   262→
   263→      if (info.affected > 0) {
   264→        const ok = await UI.confirm({
   265→          title: 'تأیید تغییر ضخامت',
   266→          html: `
   267→            <div class="confirm-stats">
   268→              <div>تغییر ضخامت ستاپ <b>${U.faNum(info.setupNumber)}</b> باعث تغییر ضخامت
   269→              <b>${U.faNum(info.affected)}</b> رکورد دیگر نیز خواهد شد.</div>
   270→            </div>`,
   271→          message: 'این عملیات تراکنشی است و در صورت خطا به حالت قبل برمی‌گردد. ادامه می‌دهید؟',
   272→          confirmText: 'بله، ضخامت همه را تغییر بده',
   273→        });
   274→        if (!ok) {
   275→          UI.toast('عملیات ویرایش لغو شد.', 'info');
   276→          return;
   277→        }
   278→      }
   279→
   280→      const result = await Setups.update(SetupView.editingId, input, { cascadedCount: info.affected });
   281→      if (!result.ok) {
   282→        SetupView.showValidationErrors(result);
   283→        return;
   284→      }
   285→      UI.toast(
   286→        (info.affected > 0
   287→          ? `ویرایش انجام شد — ضخامت ${U.faNum(info.affected)} رکورد دیگر نیز به‌روزرسانی شد.`
   288→          : 'رکورد با موفقیت ویرایش شد.'),
   289→        'success'
   290→      );
   291→    } else {
   292→      /* --- ثبت جدید --- */
   293→      const result = await Setups.create(input);
   294→      if (!result.ok) {
   295→        SetupView.showValidationErrors(result);
   296→        return;
   297→      }
   298→      UI.toast('رکورد ستاپ جدید ذخیره شد.', 'success');
   299→    }
   300→
   301→    SetupView.resetForm();
   302→    await RM.refreshAll();
   303→    await SetupView.render();
   304→  };

   306→  /** نمایش خطاهای Field-Level (§69): حاشیهٔ قرمز + پیام + فوکوس */
   307→  SetupView.showValidationErrors = function (result) {
   308→    const form = document.getElementById('setup-form');
   309→    const f = F();
   310→    const fieldMap = {
   311→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   312→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   313→      width: [f.width, document.getElementById('err-width')],
   314→      thickness: [f.thickness, document.getElementById('err-thickness')],
   315→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   316→      rollCount: [f.rollCount, document.getElementById('err-count')],
   317→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   318→    };
   319→
   320→    let firstInvalid = null;
   321→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   322→      const msg = result.errors && result.errors[field];
   323→      UI.setFieldError(input, errEl, msg);
   324→      if (msg && !firstInvalid) firstInvalid = input;
   325→    }
   326→
   327→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   328→    if (firstInvalid) firstInvalid.focus();
   329→  };

   419→  /** تحلیل + پیش‌نمایش — بدون ثبت (dry-run) */
   420→  SetupView.analyzeBulk = async function () {
   421→    const textarea = document.getElementById('bulk-text');
   422→    const text = textarea.value;
   423→
   424→    if (!text.trim()) {
   425→      UI.toast('متن Paste شده خالی است — داده‌ها را در کادر بالا وارد کنید.', 'warning');
   426→      return;
   427→    }
   428→
   429→    // قوانین تازه برای گزینه‌های متراژ استاندارد
   430→    SetupView.currentRules = await db.standardLengthRules.toArray();
   431→
   432→    const commonInit = SetupView.bulkCommonInput();
   433→    const { drafts, structureErrors } = Setups.buildBulkDraft(commonInit, text);
   434→
   435→    SetupView._bulk = {
   436→      common: { setupNumber: commonInit.setupNumber, thickness: commonInit.thickness },
   437→      drafts,
   438→      structureErrors,

   446→  /**
   447→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   448→   * بلافاصله به‌روز می‌شود (روش محاسبهٔ جدید):
   449→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   450→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   451→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست همان)
   452→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   453→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه):
   454→   *   ۱۶ / ۳ / ۴۸ / ۵
   455→   */
   456→  SetupView.bulkSummaryStats = function (drafts) {
   457→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   458→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست) — «جمع عرض الگوی برش»
   459→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   460→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   461→
   462→    for (const d of drafts) {
   463→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   464→      motherRollsTotal += rollCount;
   465→
   466→      const w = U.parseNumber(d.motherToken);
   467→      if (w !== null) motherWidthSet.add(w);
   468→
   469→      const parsed = Setups.parsePattern(d.patternText);
   470→      if (parsed.ok && parsed.parts.length) {
   471→        patternWidthsTotal += parsed.parts.length * rollCount;
   472→        for (const part of parsed.parts) patternWidthSet.add(part);
   473→      }
   474→    }
   475→
   476→    return {
   477→      count: drafts.length,
   478→      motherRollsTotal,                            // ۱۶
   479→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   480→      patternWidthsTotal,                          // ۴۸
   481→      uniquePatternWidths: patternWidthSet.size,   // ۵
   482→    };
   483→  };

   702→  /** سلول‌های محاسباتی ردیف‌ها (مجموع الگو / پرت عرض) */
   703→  SetupView.updateBulkRowComputed = function (rowIdx = null) {
   704→    const b = SetupView._bulk;
   705→    if (!b) return;
   706→    const rowsIdx = rowIdx === null ? b.drafts.map((_, i) => i) : [rowIdx];
   707→
   708→    for (const i of rowsIdx) {
   709→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   710→      if (!tr) continue;
   711→      const d = b.drafts[i];
   712→
   713→      const parsed = Setups.parsePattern(d.patternText);
   714→      const mother = U.parseNumber(d.motherToken);
   715→      const sum = parsed.ok ? parsed.sum : null;
   716→      const trim = parsed.ok && mother !== null ? mother - sum : null;
   717→
   718→      tr.querySelector('[data-bulk-cell="sum"]').innerHTML =
   719→        sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>';
   720→      tr.querySelector('[data-bulk-cell="trim"]').innerHTML =
   721→        trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>';
   722→    }
   723→  };

   758→  SetupView.revalidateBulk = async function () {
   759→    const b = SetupView._bulk;
   760→    if (!b) return;

   765→  /** اعمال نتیجهٔ اعتبارسنجی روی UI پیش‌نمایش (خطاهای سراسری + وضعیت ردیف‌ها) */
   766→  SetupView.applyBulkCheck = function (check) {
   767→    SetupView._bulkCheck = check;
   768→    const b = SetupView._bulk;
   769→    if (!b) return;
   770→
   771→    // خطاهای فیلدهای سراسری زیر ورودی‌های خودشان
   772→    const globalMap = {
   773→      setupNumber: ['bulk-g-err-setup', 'bulk-g-setup'],
   774→      thickness: ['bulk-g-err-thickness', 'bulk-g-thickness'],
   775→    };
   776→    for (const [field, [errId, inputId]] of Object.entries(globalMap)) {
   777→      const errEl = document.getElementById(errId);
   778→      const inputEl = document.getElementById(inputId);
   779→      const msg = check.globalErrors[field] || '';
   780→      if (errEl) { errEl.textContent = msg; errEl.hidden = !msg; }
   781→      if (inputEl) inputEl.classList.toggle('invalid', !!msg);
   782→    }
   783→
   784→    // وضعیت هر ردیف
   785→    b.drafts.forEach((d, i) => {
   786→      const tr = document.querySelector(`[data-bulk-tr="${i}"]`);
   787→      if (!tr) return;
   788→      const cell = tr.querySelector('[data-bulk-cell="status"]');
   789→      const rec = check.records[i];
   790→      const errs = rec ? rec.errors : [];
   791→      tr.classList.toggle('bulk-row-error', errs.length > 0);
   792→      cell.innerHTML = errs.length
   793→        ? `<span class="status-pill pill-bad">✕ خطا</span>
   794→           <ul class="bulk-errors">${errs.map((e) => `<li>${U.escapeHtml(e)}</li>`).join('')}</ul>`
   795→        : '<span class="status-pill pill-ok">✓ معتبر</span>';
   796→    });
   797→
   798→    const commitBtn = document.getElementById('btn-bulk-commit');
   799→    const hasStructure = b.structureErrors.length > 0;
   800→    // نسخهٔ ۲٫۶: با حذف همهٔ رکوردها، دکمهٔ Commit هم غیرفعال می‌ماند
   801→    commitBtn.disabled = !(b.drafts.length > 0 && check.ok === true && !hasStructure);
   802→  };
   803→
   804→  /** Commit نهایی — یک تراکنش واحد از وضعیت ویرایش‌شده؛ در صورت خطا هیچ رکوردی ثبت نمی‌شود */
   805→  SetupView.commitBulk = async function () {

   969→  SetupView.applySetupSort = function (rows) {
   970→    const sorted = [...rows];
   971→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
   972→      const { field, direction } = SetupView.sortPipeline[i];
   973→      sorted.sort((a, b) => {
   974→        if (field === 'cuttingPattern') {
   975→          const ka = Setups.normalizePatternKey(a.cuttingPattern) || '';
   976→          const kb = Setups.normalizePatternKey(b.cuttingPattern) || '';
   977→          const r = ka.localeCompare(kb, 'fa', { numeric: true });
   978→          return direction === 'desc' ? -r : r;
   979→        }
   980→        return U.compareNumeric(a[field], b[field], direction);
   981→      });
   982→    }
   983→    return sorted;
