   130→   130→  SetupView.applyThicknessInheritance = async function () {
   131→   131→    const { setupNumber, thickness, standardLength } = F();
   132→   132→    const value = U.parseInt(setupNumber.value);
   133→   133→    const hintEl = document.getElementById('hint-thickness');
   134→   134→
   135→   135→    if (SetupView.editingId !== null) return;   // در ویرایش، ضخامت آزاد است (§23)
   136→   136→
   137→   137→    if (value === null || value < 0) {
   138→   138→      SetupView.thicknessLocked = false;
   139→   139→      thickness.disabled = false;
   140→   140→      hintEl.hidden = true;
   141→   141→      return;
   142→   142→    }
   143→   143→
   144→   144→    const existing = await db.setupRolls.where('setupNumber').equals(value).toArray();
   145→   145→    const withThickness = existing.find((s) => s.thickness !== null);
   146→   146→
   147→   147→    if (withThickness) {
   148→   148→      // ارث‌بری: ضخامت تعیین‌شدهٔ اولین رکورد همان ستاپ (§22)
   149→   149→      SetupView.thicknessLocked = true;
   150→   150→      thickness.value = String(withThickness.thickness);
   151→   151→      thickness.disabled = true;
   152→   152→      hintEl.textContent = `ضخامت ستاپ ${U.faNum(value)} قبلاً ${U.faNum(withThickness.thickness)} تعیین شده و به‌صورت خودکار به ارث رسید (§ ارث‌بری ضخامت).`;
   153→   153→      hintEl.hidden = false;
   154→   154→    } else {
   155→   155→      SetupView.thicknessLocked = false;
   156→   156→      thickness.disabled = false;
   157→   157→      hintEl.hidden = true;
   158→   158→    }
   159→   159→
   160→   160→    SetupView.refreshStandardLengthOptions(standardLength.value);
   161→   161→  };

   163→   163→  /** پرکردن دراپ‌داون متراژ استاندارد فقط با قوانینِ ضخامت فعلی (§24) */
   164→   164→  SetupView.refreshStandardLengthOptions = function (keepValue = null) {
   165→   165→    const { thickness, standardLength } = F();
   166→   166→    const thicknessValue = U.parseNumber(thickness.value);
   167→   167→    const errEl = document.getElementById('err-standard');
   168→   168→
   169→   169→    const options = thicknessValue === null
   170→   170→      ? []
   171→   171→      : RM.rulesEngine.standardLengthsFor(SetupView.currentRules, thicknessValue);
   172→   172→
   173→   173→    let html = '<option value="">— انتخاب کنید —</option>';
   174→   174→    for (const opt of options) {
   175→   175→      html += `<option value="${opt}">${U.faNum(opt)}</option>`;
   176→   176→    }
   177→   177→    standardLength.innerHTML = html;
   178→   178→
   179→   179→    // نگهداری مقدار در صورت تعلق به همین ضخامت
   180→   180→    if (keepValue !== null && options.includes(U.parseNumber(keepValue))) {
   181→   181→      standardLength.value = keepValue;
   182→   182→    } else {
   183→   183→      standardLength.value = '';
   184→   184→    }
   185→   185→
   186→   186→    // راهنمای کاربر وقتی قوانین این ضخامت تعریف نشده
   187→   187→    if (thicknessValue !== null && !options.length) {
   188→   188→      errEl.textContent = 'برای این ضخامت قانون متراژی تعریف نشده — از تب «تنظیمات» قانون اضافه کنید.';
   189→   189→      errEl.hidden = false;
   190→   190→    } else {
   191→   191→      errEl.textContent = '';
   192→   192→      errEl.hidden = true;
   193→   193→    }
   194→   194→  };

   197→   197→  SetupView.updatePatternFeedback = function () {
   198→   198→    const { width, cuttingPattern } = F();
   199→   199→    const box = document.getElementById('pattern-feedback');
   200→   200→
   201→   201→    const result = Setups.validatePattern(cuttingPattern.value, U.parseNumber(width.value));
   202→   202→
   203→   203→    if (!cuttingPattern.value.trim()) {
   204→   204→      box.className = 'pattern-feedback';
   205→   205→      box.innerHTML = '<span class="muted">اختیاری — مثال: 1100-1170</span>';
   206→   206→      return;

   208→   208→
   209→   209→    if (!result.ok) {
   210→   210→      box.className = 'pattern-feedback invalid';
   211→   211→      box.innerHTML = `<span class="pf-status">✕</span> ${result.message}`;
   212→   212→      return;
   213→   213→    }
   214→   214→
   215→   215→    const motherWidth = U.parseNumber(width.value);
   216→   216→    const remaining = motherWidth !== null ? motherWidth - result.sum : null;
   217→   217→
   218→   218→    box.className = 'pattern-feedback valid';
   219→   219→    box.innerHTML =
   220→   220→      `<span class="pf-status">✓</span> مجموع: <b>${U.faWidth(result.sum)}</b>` +
   221→   221→      (remaining !== null && remaining >= 0
   222→   222→        ? ` · باقی‌مانده (پرت عرض): <b>${U.faWidth(remaining)}</b>`
   223→   223→        : '') +
   224→   224→      ` · بخش‌ها: ${U.faNum(result.parts.length)}`;
   225→   225→  };

   277→   277→        'success'

   280→   280→      /* --- ثبت جدید --- */
   281→   281→      const result = await Setups.create(input);

   283→   283→        SetupView.showValidationErrors(result);

   294→   294→  /** نمایش خطاهای Field-Level (§69): حاشیهٔ قرمز + پیام + فوکوس */
   295→   295→  SetupView.showValidationErrors = function (result) {
   296→   296→    const form = document.getElementById('setup-form');
   297→   297→    const f = F();
   298→   298→    const fieldMap = {
   299→   299→      setupNumber: [f.setupNumber, document.getElementById('err-setup')],
   300→   300→      machineNumber: [f.machineNumber, document.getElementById('err-machine')],
   301→   301→      width: [f.width, document.getElementById('err-width')],
   302→   302→      thickness: [f.thickness, document.getElementById('err-thickness')],
   303→   303→      standardLength: [f.standardLength, document.getElementById('err-standard')],
   304→   304→      rollCount: [f.rollCount, document.getElementById('err-count')],
   305→   305→      cuttingPattern: [f.cuttingPattern, document.getElementById('err-pattern')],
   306→   306→    };
   307→   307→
   308→   308→    let firstInvalid = null;
   309→   309→    for (const [field, [input, errEl]] of Object.entries(fieldMap)) {
   310→   310→      const msg = result.errors && result.errors[field];
   311→   311→      UI.setFieldError(input, errEl, msg);
   312→   312→      if (msg && !firstInvalid) firstInvalid = input;
   313→   313→    }
   314→   314→
   315→   315→    if (result.message) UI.toast(U.escapeHtml(result.message), 'error');
   316→   316→    if (firstInvalid) firstInvalid.focus();
   317→   317→  };

   434→   434→  /**
   435→   435→   * خلاصهٔ زندهٔ رکوردهای تشخیص‌داده‌شده — با هر تغییری در رکوردها
   436→   436→   * بلافاصله به‌روز می‌شود (روش محاسبهٔ جدید):
   437→   437→   *   جمع عرض رول مادر   = Σ ستون تعداد (مجموع تعداد رول‌های مادر)
   438→   438→   *   یونیک رول مادر     = تعداد عرض‌های یکتای مادر (مثل ۲۲۸۰/۲۳۸۰/۲۳۰۰ = ۳)
   439→   439→   *   جمع عرض الگوی برش = Σ (تعداد عرض‌های الگوی هر رکورد × تعداد ست همان)
   440→   440→   *   یونیک عرض الگو     = تعداد عرض‌های یکتای برش (مثل ۷۸۰/۷۰۰/۸۰۰/۹۰۰/۷۴۰ = ۵)
   441→   441→   * مثال مرجع (۴ رکورد با تعداد ۲/۷/۵/۲ و الگوهای سه‌تکه):
   442→   442→   *   ۱۶ / ۳ / ۴۸ / ۵
   443→   443→   */
   444→   444→  SetupView.bulkSummaryStats = function (drafts) {
   445→   445→    let motherRollsTotal = 0;      // Σ تعداد — «جمع عرض رول مادر»
   446→   446→    let patternWidthsTotal = 0;    // Σ (تعداد عرض الگو × تعداد ست) — «جمع عرض الگوی برش»
   447→   447→    const motherWidthSet = new Set();   // عرض‌های یکتای مادر
   448→   448→    const patternWidthSet = new Set();  // عرض‌های یکتای الگو
   449→   449→
   450→   450→    for (const d of drafts) {
   451→   451→      const rollCount = d.rollCount === null ? 0 : (d.rollCount || 0);
   452→   452→      motherRollsTotal += rollCount;
   453→   453→
   454→   454→      const w = U.parseNumber(d.motherToken);
   455→   455→      if (w !== null) motherWidthSet.add(w);
   456→   456→
   457→   457→      const parsed = Setups.parsePattern(d.patternText);
   458→   458→      if (parsed.ok && parsed.parts.length) {
   459→   459→        patternWidthsTotal += parsed.parts.length * rollCount;
   460→   460→        for (const part of parsed.parts) patternWidthSet.add(part);
   461→   461→      }
   462→   462→    }
   463→   463→
   464→   464→    return {
   465→   465→      count: drafts.length,
   466→   466→      motherRollsTotal,                            // ۱۶
   467→   467→      uniqueMotherWidths: motherWidthSet.size,     // ۳
   468→   468→      patternWidthsTotal,                          // ۴۸
   469→   469→      uniquePatternWidths: patternWidthSet.size,   // ۵
   470→   470→    };
   471→   471→  };

   662→   662→    const b = SetupView._bulk;

   664→   664→    const rowsIdx = rowIdx === null ? b.drafts.map((_, i) => i) : [rowIdx];

   672→   672→      const mother = U.parseNumber(d.motherToken);
   673→   673→      const sum = parsed.ok ? parsed.sum : null;
   674→   674→      const trim = parsed.ok && mother !== null ? mother - sum : null;
   675→   675→
   676→   676→      tr.querySelector('[data-bulk-cell="sum"]').innerHTML =
   677→   677→        sum !== null ? U.faWidth(sum) : '<span class="muted">—</span>';

   679→   679→        trim !== null && trim >= 0 ? U.faWidth(trim) : '<span class="muted">—</span>';
   680→   680→    }

   927→   927→    const sorted = [...rows];
   928→   928→    for (let i = SetupView.sortPipeline.length - 1; i >= 0; i--) {
   929→   929→      const { field, direction } = SetupView.sortPipeline[i];
   930→   930→      sorted.sort((a, b) => {

   935→   935→          return direction === 'desc' ? -r : r;
   936→   936→        }
   937→   937→        return U.compareNumeric(a[field], b[field], direction);
   938→   938→      });
   939→   939→    }
   940→   940→    return sorted;
